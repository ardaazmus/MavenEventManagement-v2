    62→  PENDING: "Bekliyor",

    64→  ...EXPENSE_STATUS,

    66→
    67→// kategori → Chip tonu (TECH için sky yok, en yakın teal)
    68→const CATEGORY_TONE: Record<string, ChipTone> = {
    69→  FIELD_EXPENSE: "amber",
    70→  TECH: "teal",
    71→  CATERING: "violet",
    72→  LOGISTICS: "neutral",
    73→};
    74→
    75→const categoryTone = (key: string): ChipTone => CATEGORY_TONE[key] ?? "neutral";
    76→
    77→// durum rengini STATUS_TONE'dan Chip tonuna çevir (kırılımda tutarlı renk)
    78→const toneFromStatus = (status: string): ChipTone => {
    79→  const cls = STATUS_TONE[status] ?? "";
    80→  if (cls.includes("emerald")) return "emerald";
    81→  if (cls.includes("amber")) return "amber";
    82→  if (cls.includes("rose")) return "rose";
    83→  if (cls.includes("sky")) return "teal";
    84→  if (cls.includes("purple")) return "violet";
    85→  return "neutral";
    86→};
    87→
    88→const todayStr = (): string => {
    89→  const d = new Date();
    90→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
    91→};
    92→
    93→// benzersiz gider kodu: GSN-YYYY-XXX (sayaç = mevcut adet + 1; çakışırsa +1 dene)
    94→const nextExpenseCode = (existing: ExpenseRow[]): string => {
    95→  const year = new Date().getFullYear();
    96→  const taken = new Set(existing.map((e) => e.code));
    97→  let n = existing.length + 1;
    98→  let code = `GSN-${year}-${String(n).padStart(3, "0")}`;
    99→  while (taken.has(code)) {
   100→    n += 1;
   101→    code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   102→  }
   103→  return code;
   104→};

   106→// kırılım barları için ortak satır
   107→function BreakdownBar({ text, count, total, max, barClass }: { text: string; count: number; total: number; max: number; barClass: string }) {
   108→  return (
   109→    <div>
   110→      <div className="flex items-baseline justify-between gap-2 text-xs">
   111→        <span className="min-w-0 truncate font-medium">{text}</span>
   112→        <span className="shrink-0 tabular-nums text-muted-foreground">
   113→          {count} adet · <span className="font-semibold text-foreground">{fmtMoney(total)}</span>
   114→        </span>
   115→      </div>
   116→      <div className="mt-1 h-2 overflow-hidden rounded-full bg-muted">
   117→        <div className={`h-full rounded-full ${barClass}`} style={{ width: `${Math.max(4, Math.round((total / max) * 100))}%` }} />
   118→      </div>
   119→    </div>
   120→  );
   121→}
