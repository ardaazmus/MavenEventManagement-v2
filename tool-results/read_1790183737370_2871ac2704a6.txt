    93→    93→// benzersiz gider kodu: GSN-YYYY-XXX (sayaç = mevcut adet + 1; çakışırsa +1 dene)
    94→    94→const nextExpenseCode = (existing: ExpenseRow[]): string => {
    95→    95→  const year = new Date().getFullYear();
    96→    96→  const taken = new Set(existing.map((e) => e.code));
    97→    97→  let n = existing.length + 1;
    98→    98→  let code = `GSN-${year}-${String(n).padStart(3, "0")}`;
    99→    99→  while (taken.has(code)) {
   100→   100→    n += 1;
   101→   101→    code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   102→   102→  }
   103→   103→  return code;
   104→   104→};
