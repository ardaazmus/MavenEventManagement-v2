    93→export function PeopleView() {
    94→  const { tenant, bump, currentEditionId, refreshKey } = useApp();
    95→  const { toast } = useToast();
    96→  const [q, setQ] = useState("");
    97→  const [selected, setSelected] = useState<PersonRow | null>(null);
    98→  const [detail, setDetail] = useState<Person360 | null>(null);
    99→  const [detailLoading, setDetailLoading] = useState(false);
   100→  const [createOpen, setCreateOpen] = useState(false);
   101→  const [form, setForm] = useState({ firstName: "", lastName: "", email: "", company: "", title: "" });
   102→  const [mergeSug, setMergeSug] = useState<DuplicateSuggestion | null>(null);
   103→  const [mergeTarget, setMergeTarget] = useState<string | null>(null); // korunacak (hedef) kişi id
   104→  const [mergeBusy, setMergeBusy] = useState(false);
   105→  const [mergePreview, setMergePreview] = useState<MergePreview | null>(null);
   106→  const [previewLoading, setPreviewLoading] = useState(false);
   107→  const [resolutions, setResolutions] = useState<Record<string, "target" | "source">>({});
   108→
   109→  const { data, error, reload, loading } = useApi<PersonRow[]>(() => listEntity<PersonRow>("people", { q, limit: 300 }), [q]);
   110→
   111→  // Olası mükerrerler — tenant geneli (edisyon-bağımsız, null-guard gerekmez); edisyon değişince tazelensin
   112→  const { data: dupData, error: dupError, reload: dupReload } = useApi<DuplicatesData>(() => apiGet<DuplicatesData>("/api/people/duplicates"), [currentEditionId, refreshKey]);
   113→
   114→  const open360 = async (p: PersonRow) => {
   115→    setSelected(p);
   116→    setDetailLoading(true);
   117→    try {
   118→      const d = await apiGet<Person360>(`/api/people/${p.id}`);
   119→      console.log("[360] loaded", p.id, d?.person?.id, Object.keys(d ?? {}).length);
   120→      setDetail(d);
   121→    } catch (e) {
   122→      console.error("[360] failed", p.id, e);
   123→      setDetail(null);
   124→    } finally {
   125→      setDetailLoading(false);
   126→    }
   127→  };
   128→
   129→  const createPerson = async () => {
   130→    try {
   131→      await apiSend("/api/people", "POST", { ...form, tenantId: tenant?.id });
   132→      toast({ title: "Kişi oluşturuldu", description: `${form.firstName} ${form.lastName} tenant içine eklendi.` });
   133→      setCreateOpen(false);
   134→      setForm({ firstName: "", lastName: "", email: "", company: "", title: "" });
   135→      reload(); bump();
   136→    } catch (e) {
   137→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Kişi oluşturulamadı", variant: "destructive" });
   138→    }
   139→  };
   140→
   141→  // Birleştirme diyaloğunu aç — varsayılan hedef: daha eski kayıt (API'nin olderId önerisi)
   142→  const openMerge = (s: DuplicateSuggestion) => {
   143→    const def = s.olderId && s.persons.some((p) => p.id === s.olderId) ? s.olderId : s.persons[0]?.id ?? null;
   144→    setMergeSug(s);
   145→    setMergeTarget(def);
   146→    setMergePreview(null);
   147→    setResolutions({});
   148→  };
   149→
   150→  // hedef seçilince çakışma önizlemesi + varsayılan çözümler
   151→  useEffect(() => {
   152→    if (!mergeSug || !mergeTarget) { setMergePreview(null); return; }
   153→    const source = mergeSug.persons.find((p) => p.id !== mergeTarget);
   154→    if (!source) { setMergePreview(null); return; }
   155→    let alive = true;
   156→    setPreviewLoading(true);
   157→    apiGet<MergePreview>(`/api/people/merge-preview?sourceId=${source.id}&targetId=${mergeTarget}`)
   158→      .then((d) => {
   159→        if (!alive) return;
   160→        setMergePreview(d);
   161→        const def: Record<string, "target" | "source"> = {};
   162→        for (const c of d.conflictingEditions) def[c.editionId] = c.defaultWinner;
   163→        setResolutions(def);
   164→      })
   165→      .catch(() => { if (alive) setMergePreview(null); })
   166→      .finally(() => { if (alive) setPreviewLoading(false); });
   167→    return () => { alive = false; };
   168→  }, [mergeSug, mergeTarget]);
   169→
   170→  // Onaylı birleştirme — çakışma çözümleriyle (sunucu tek işlemde taşır)
   171→  const confirmMerge = async () => {
   172→    const sug = mergeSug;
   173→    if (!sug || !mergeTarget) return;
   174→    const target = sug.persons.find((p) => p.id === mergeTarget) ?? null;
   175→    const source = sug.persons.find((p) => p.id !== mergeTarget) ?? null;
   176→    if (!target || !source) {
   177→      toast({ title: "Hata", description: "Kaynak ve hedef aynı olamaz.", variant: "destructive" });
   178→      return;
   179→    }
   180→    setMergeBusy(true);
   181→    try {
   182→      const r = await apiSend<{ mergedRegistrations?: number; resolvedEditions?: number }>("/api/flows", "POST", {
   183→        action: "person.merge", sourceId: source.id, targetId: target.id, resolutions, fillProfile: true,
   184→      });
   185→      toast({
   186→        title: "Kişiler birleştirildi — geçmiş korundu",
   187→        description: `${source.fullName} → ${target.fullName}${r.mergedRegistrations ? ` · ${r.mergedRegistrations} kayıt taşındı` : ""}${r.resolvedEditions ? ` · ${r.resolvedEditions} edisyonda çakışma çözüldü` : ""}`,
   188→      });
   189→      setMergeSug(null);
   190→      setMergeTarget(null);
   191→      setMergePreview(null);
   192→      reload(); dupReload(); bump(); // kişi listesi + mükerrer listesi + global sayaçlar
   193→    } catch (e) {
   194→      toast({ title: "Birleştirme başarısız", description: e instanceof Error ? e.message : "Kişiler birleştirilemedi", variant: "destructive" });
   195→    } finally {
   196→      setMergeBusy(false);
   197→    }
   198→  };
   199→
   200→  return (
   201→    <div>
   202→      <PageHeader title="Kişiler" desc="Tenant içinde tekil kimlik — e-posta güçlü işaret, kesin kimlik değil (birleştirme onaylı yapılır)">
   203→        {dupData && dupData.suggestions.length === 0 && (
   204→          <Chip tone="emerald">
   205→            <span className="inline-flex items-center gap-1"><Icons.CheckCircle2 className="size-3" aria-hidden />Mükerrer yok</span>
   206→          </Chip>
   207→        )}
   208→        <Input placeholder="Ad, e-posta, kurum ara…" value={q} onChange={(e) => setQ(e.target.value)} className="h-9 w-56" />
   209→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   210→          <DialogTrigger asChild><Button size="sm"><Icons.UserPlus className="size-4" /> Kişi Ekle</Button></DialogTrigger>
   211→          <DialogContent>
   212→            <DialogHeader><DialogTitle>Yeni Kişi</DialogTitle></DialogHeader>
   213→            <div className="grid gap-3 sm:grid-cols-2">
   214→              <div><Label>Ad</Label><Input value={form.firstName} onChange={(e) => setForm({ ...form, firstName: e.target.value })} /></div>
   215→              <div><Label>Soyad</Label><Input value={form.lastName} onChange={(e) => setForm({ ...form, lastName: e.target.value })} /></div>
   216→              <div className="sm:col-span-2"><Label>E-posta</Label><Input type="email" value={form.email} onChange={(e) => setForm({ ...form, email: e.target.value })} /></div>
   217→              <div><Label>Kurum</Label><Input value={form.company} onChange={(e) => setForm({ ...form, company: e.target.value })} /></div>
   218→              <div><Label>Unvan</Label><Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} /></div>
   219→            </div>
   220→            <DialogFooter><Button onClick={createPerson} disabled={!form.firstName || !form.lastName}>Oluştur</Button></DialogFooter>
   221→          </DialogContent>
   222→        </Dialog>
   223→      </PageHeader>
   224→
   225→      {/* ── Olası Mükerrerler (R2-b) — öneri bağlamaz, birleştirme onaylı yapılır ── */}
   226→      {dupError && !dupData && (
   227→        <p className="mb-3 flex items-center gap-1.5 text-xs text-muted-foreground">
   228→          <Icons.TriangleAlert className="size-3.5 shrink-0 text-amber-500" aria-hidden />
   229→          Mükerrer taraması yüklenemedi — kişi listesi etkilenmedi.
   230→        </p>
   231→      )}
   232→      {dupData && dupData.suggestions.length > 0 && (
   233→        <SectionCard
   234→          title="Olası Mükerrerler"
   235→          desc="Tenant genelinde kimlik eşleşmesi — öneriler bağlamadan incelenmelidir"
   236→          action={<Chip tone="amber">{dupData.suggestions.length} öneri</Chip>}
   237→          className="mb-4"
   238→        >
   239→          <div className="maven-scroll max-h-96 space-y-3 overflow-y-auto">
   240→            {dupData.suggestions.map((s) => {
   241→              const [a, b] = s.persons;
   242→              return (
   243→                <div key={s.key} className="rounded-lg border bg-muted/20 p-3">
   244→                  <div className="grid grid-cols-1 items-stretch gap-2 md:grid-cols-[1fr_auto_1fr] md:gap-3">
   245→                    <DupPersonCard p={a} />
   246→                    <div className="flex items-center justify-center gap-2 md:w-40 md:flex-col">
   247→                      <Icons.ArrowLeftRight className="size-4 shrink-0 text-muted-foreground" aria-hidden />
   248→                      <Chip tone={REASON_TONE[s.reason] ?? "neutral"}>{dupData.reasonLabels[s.reason] ?? s.reason}</Chip>
   249→                      <Button size="sm" variant="outline" onClick={() => openMerge(s)}>
   250→                        <Icons.Merge className="size-4" /> Birleştir
   251→                      </Button>
   252→                    </div>
   253→                    <DupPersonCard p={b} />
   254→                  </div>
   255→                </div>
   256→              );
   257→            })}
   258→          </div>
   259→        </SectionCard>
   260→      )}
   261→
   262→      {loading ? <Loading /> : error ? <ErrorState message={error} onRetry={reload} /> : (data ?? []).length === 0 ? (
   263→        <EmptyState title="Kişi bulunamadı" desc="Filtre sonucu yok — aramayı temizleyin veya yeni kişi ekleyin." />
   264→      ) : (
   265→        <div className="grid gap-2">
   266→          {(data ?? []).map((p) => (
   267→            <button key={p.id} onClick={() => open360(p)} className="flex min-w-0 items-center gap-3 rounded-xl border bg-card p-3 text-left transition hover:border-primary/40 hover:shadow-sm">
   268→              <span className="grid size-9 shrink-0 place-items-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   269→                {p.firstName[0]}{p.lastName[0]}
   270→              </span>
   271→              <div className="min-w-0 flex-1">
   272→                <p className="truncate text-sm font-medium">
   273→                  {p.firstName} {p.lastName}
   274→                  {p.status === "MERGED" && <Chip tone="rose">birleştirildi</Chip>}
   275→                </p>
   276→                <p className="truncate text-xs text-muted-foreground">{p.title ? `${p.title} · ` : ""}{p.company ?? "—"} · {p.email ?? "e-posta yok"}</p>
   277→              </div>
   278→              <Icons.ChevronRight className="size-4 shrink-0 text-muted-foreground" />
   279→            </button>
   280→          ))}
   281→        </div>
   282→      )}
   283→
   284→      {/* ── Birleştirme onay diyaloğu — hedef seçimi + çakışma çözümü (R7) ── */}
   285→      <Dialog open={Boolean(mergeSug)} onOpenChange={(o) => { if (!o) setMergeSug(null); }}>
   286→        <DialogContent className="max-h-[85vh] overflow-y-auto maven-scroll sm:max-w-xl">
   287→          <DialogHeader>
   288→            <DialogTitle>Kişileri Birleştir</DialogTitle>
   289→            <DialogDescription>{mergeSug?.note ?? "Kaynak kişi hedefe taşınır; hedef kayıt korunur."}</DialogDescription>
   290→          </DialogHeader>
   291→
   292→          <div role="radiogroup" aria-label="Korunacak kişi" className="grid gap-2 sm:grid-cols-2">
   293→            {mergeSug?.persons.map((p, i) => {
   294→              const checked = mergeTarget === p.id;
   295→              return (
   296→                <button
   297→                  key={p.id}
   298→                  type="button"
   299→                  role="radio"
   300→                  aria-checked={checked}
   301→                  onClick={() => setMergeTarget(p.id)}
   302→                  className={cn(
   303→                    "flex items-start gap-2.5 rounded-lg border p-3 text-left transition focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary/40",
   304→                    checked ? "border-primary bg-primary/5 ring-1 ring-primary/30" : "hover:border-primary/40"
   305→                  )}
   306→                >
   307→                  <span aria-hidden className={cn("mt-0.5 grid size-4 shrink-0 place-items-center rounded-full border", checked ? "border-primary" : "border-muted-foreground/40")}>
   308→                    {checked && <span className="size-2 rounded-full bg-primary" />}
   309→                  </span>
   310→                  <span className="min-w-0">
   311→                    <span className="block text-sm font-medium">
   312→                      {i === 0 ? "A'yı koru" : "B'yi koru"} <span className="font-normal text-muted-foreground">({i === 0 ? "B" : "A"} birleşir)</span>
   313→                    </span>
   314→                    <span className="mt-0.5 block truncate text-xs text-muted-foreground">{p.fullName} · {p.email ?? "e-posta yok"} · {p.company ?? "—"}</span>
   315→                  </span>
   316→                </button>
   317→              );
   318→            })}
   319→          </div>
   320→
   321→          {/* ── Çakışma önizlemesi — hedef seçilince yüklenir ── */}
   322→          {previewLoading && (
   323→            <p className="flex items-center gap-2 py-2 text-xs text-muted-foreground"><Icons.Loader2 className="size-3.5 animate-spin" /> Çakışma analizi yapılıyor…</p>
   324→          )}
   325→          {mergePreview && !previewLoading && (
   326→            <div className="maven-portal-enter space-y-3">
   327→              {/* profil alan farkları */}
   328→              {mergePreview.fieldDiffs.length > 0 && (
   329→                <div className="rounded-lg border bg-muted/20 p-3">
   330→                  <p className="mb-2 flex items-center gap-1.5 text-xs font-semibold"><Icons.SlidersHorizontal className="size-3.5 text-primary" /> Profil alanları</p>
   331→                  <ul className="space-y-1">
   332→                    {mergePreview.fieldDiffs.map((f) => (
   333→                      <li key={f.field} className="flex flex-wrap items-center gap-x-2 gap-y-0.5 text-[11px]">
   334→                        <span className="w-14 shrink-0 font-medium text-muted-foreground">{MERGE_FIELD_LABELS[f.field] ?? f.field}</span>
   335→                        {f.kind === "fill" ? (
   336→                          <Chip tone="teal">hedef boş → "{f.source}" kaynaktan doldurulur</Chip>
   337→                        ) : (
   338→                          <Chip tone="amber">farklı: hedef "{f.target}" · kaynak "{f.source}" → hedef korunur</Chip>
   339→                        )}
   340→                      </li>
   341→                    ))}
   342→                  </ul>
   343→                </div>
   344→              )}
   345→
   346→              {/* aynı edisyonda iki katılım — çözüm zorunlu */}
   347→              {mergePreview.conflictingEditions.length > 0 ? (
   348→                <div className="space-y-2">
   349→                  <p className="flex items-center gap-1.5 rounded-lg border border-amber-300 bg-amber-50 px-2.5 py-1.5 text-xs font-semibold text-amber-800">
   350→                    <Icons.TriangleAlert className="size-3.5" aria-hidden />
   351→                    {mergePreview.conflictingEditions.length} edisyonda iki katılım var — kazanan seçilmeli
   352→                  </p>
   353→                  {mergePreview.conflictingEditions.map((c) => {
