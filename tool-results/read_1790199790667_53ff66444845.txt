   214→export function ProgramView() {
   215→  const { currentEditionId, bump, refreshKey, editions } = useApp();
   216→  const { toast } = useToast();
   217→  const [dayFilter, setDayFilter] = useState("ALL");
   218→  const [publishTarget, setPublishTarget] = useState<SessionRow | null>(null);
   219→  const [busy, setBusy] = useState(false);
   220→
   221→  // CME Kredi Defteri — yalnız edisyonun CME_CREDITS yeteneği açıksa görünür
   222→  const edition = editions.find((e) => e.id === currentEditionId);
   223→  const cmeEnabled = hasCapability(edition, "CME_CREDITS");
   224→  const [creditInputs, setCreditInputs] = useState<Record<string, string>>({});
   225→  const [savingId, setSavingId] = useState<string | null>(null);
   226→  const [bulkDefaults, setBulkDefaults] = useState<Record<string, string>>({});
   227→  const [bulkBusy, setBulkBusy] = useState(false);
   228→  const [reportOpen, setReportOpen] = useState(false);
   229→
   230→  const { data: sessions, error, reload, loading } = useApi<SessionRow[]>(() => listEntity<SessionRow>("sessions", { editionId: currentEditionId ?? undefined, limit: 200 }), [currentEditionId, refreshKey]);
   231→  const { data: rooms } = useApi<{ id: string; name: string; capacity: number }[]>(() => listEntity("rooms", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey]);
   232→  const { data: cme, error: cmeError, reload: reloadCme, loading: cmeLoading } = useApi<CmeData | null>(() => {
   233→    if (!currentEditionId || !cmeEnabled) return Promise.resolve(null);
   234→    return apiGet<CmeData>("/api/cme?editionId=" + currentEditionId);
   235→  }, [currentEditionId, refreshKey]);
   236→
   237→  // dış veri reload'unda satır kredi girişlerini sunucu değeriyle senkronize et
   238→  useEffect(() => {
   239→    if (!cme) return;
   240→    const next: Record<string, string> = {};
   241→    for (const s of cme.sessions) next[s.id] = s.cmeCredits != null ? String(s.cmeCredits) : "";
   242→    setCreditInputs(next);
   243→  }, [cme]);
   244→
   245→  const days = Array.from(new Set((sessions ?? []).map((s) => s.startTime.slice(0, 10)))).sort();
   246→  const filtered = (sessions ?? []).filter((s) => dayFilter === "ALL" || s.startTime.slice(0, 10) === dayFilter);
   247→  // çakışma denetimi: aynı salonda zaman üstüste
   248→  const clashes = new Set<string>();
   249→  for (let i = 0; i < filtered.length; i++) {
   250→    for (let j = i + 1; j < filtered.length; j++) {
   251→      const a = filtered[i], b = filtered[j];
   252→      if (a.room?.id && a.room.id === b.room?.id && new Date(a.startTime) < new Date(b.endTime) && new Date(b.startTime) < new Date(a.endTime)) {
   253→        clashes.add(a.id); clashes.add(b.id);
   254→      }
   255→    }
   256→  }
   257→
   258→  // toplu atama butonu: en az bir tür girişi doluysa aktif
   259→  const bulkFilled = CME_SESSION_TYPES.some((t) => (bulkDefaults[t] ?? "").trim() !== "");
   260→
   261→  const saveCredits = async (s: CmeSession) => {
   262→    const raw = (creditInputs[s.id] ?? "").trim();
   263→    if (raw === "") return;
   264→    const credits = Number(raw);
   265→    if (Number.isNaN(credits) || credits < 0 || credits > 99) {
   266→      toast({ title: "Geçersiz kredi", description: "Kredi 0–99 arasında olmalı.", variant: "destructive" });
   267→      return;
   268→    }
   269→    setSavingId(s.id);
   270→    try {
   271→      await apiSend("/api/cme", "POST", { action: "set-credits", sessionId: s.id, credits });
   272→      toast({ title: `${credits} kredi atandı: ${s.title}` });
   273→      reloadCme(); bump();
   274→    } catch (e) {
   275→      toast({ title: "Kredi atanamadı", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   276→    } finally { setSavingId(null); }
   277→  };
   278→
   279→  const applyBulk = async () => {
   280→    if (!currentEditionId) return;
   281→    const defaults: Record<string, number> = {};
   282→    for (const t of CME_SESSION_TYPES) {
   283→      const raw = (bulkDefaults[t] ?? "").trim();
   284→      if (raw === "") continue;
   285→      const n = Number(raw);
   286→      if (!Number.isNaN(n) && n >= 0 && n <= 99) defaults[t] = n;
   287→    }
   288→    if (Object.keys(defaults).length === 0) return;
   289→    setBulkBusy(true);
   290→    try {
   291→      const res = await apiSend<{ ok: boolean; updated: number }>("/api/cme", "POST", { action: "bulk-apply", editionId: currentEditionId, defaults });
   292→      if (res.updated > 0) toast({ title: `${res.updated} oturuma kredi atandı`, description: "Yalnız kredisi olmayan oturumlar güncellendi." });
   293→      else toast({ title: "Atanacak kredisiz oturum yok", description: "Seçilen türlere ait tüm oturumlar zaten kredili." });
   294→      reloadCme(); bump();
   295→    } catch (e) {
   296→      toast({ title: "Toplu atama başarısız", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   297→    } finally { setBulkBusy(false); }
   298→  };
   299→
   300→  const publish = async () => {
   301→    if (!publishTarget) return;
   302→    setBusy(true);
   303→    try {
   304→      await apiSend(`/api/sessions/${publishTarget.id}`, "PUT", { status: "PUBLISHED", isVisible: true });
   305→      toast({ title: "Oturum yayınlandı", description: "Kişisel programlarda görünür; etkilenen konuşmacılar bilgilendirilir." });
   306→      setPublishTarget(null); reload(); bump();
   307→    } catch (e) {
   308→      toast({ title: "Yayınlanamadı", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   309→    } finally { setBusy(false); }
   310→  };
   311→
   312→  return (
   313→    <div className="space-y-5">
   314→      <PageHeader title="Program" desc="Salon, zaman, görevli — çakışmalı yayın engellenir; bilimsel kararı program modülü değiştirmez">
   315→        <Button variant="ghost" size="sm" onClick={() => { reload(); reloadCme(); }} aria-label="Yenile"><Icons.RefreshCw className="size-4" /></Button>
   316→      </PageHeader>
   317→
   318→      <Tabs defaultValue="sessions">
   319→        <TabsList className="h-auto flex-wrap">
   320→          <TabsTrigger value="sessions">Oturumlar</TabsTrigger>
   321→          {cmeEnabled && <TabsTrigger value="cme"><Icons.GraduationCap className="size-4" /> CME Kredi</TabsTrigger>}
   322→        </TabsList>
   323→
   324→        <TabsContent value="sessions" className="mt-4 space-y-4">
   325→          <Select value={dayFilter} onValueChange={setDayFilter}>
   326→            <SelectTrigger className="h-9 w-44"><SelectValue /></SelectTrigger>
   327→            <SelectContent>
   328→              <SelectItem value="ALL">Tüm günler</SelectItem>
   329→              {days.map((d) => <SelectItem key={d} value={d}>{fmtDate(d)}</SelectItem>)}
   330→            </SelectContent>
   331→          </Select>
   332→
   333→          {clashes.size > 0 && (
   334→            <div className="flex items-start gap-2 rounded-lg border border-rose-200 bg-rose-50/60 p-3 text-sm text-rose-800">
   335→              <Icons.OctagonAlert className="mt-0.5 size-4 shrink-0" />
   336→              Aynı salonda çakışan oturumlar var ({clashes.size}) — çakışmalı yayın engellenir veya yetkili gerekçeyle işaretler.
   337→            </div>
   338→          )}
   339→
   340→          {loading ? <Loading /> : error ? <ErrorState message={error} onRetry={reload} /> : filtered.length === 0 ? (
   341→            <EmptyState title="Bu gün için oturum yok" desc="Oturum oluşturun veya başka gün seçin." />
   342→          ) : (
   343→            <div className="space-y-2.5">
   344→              {filtered.map((s) => (
   345→                <div key={s.id} className={cn("rounded-xl border bg-card p-4", clashes.has(s.id) && "border-rose-300")}>
   346→                  <div className="flex flex-wrap items-center gap-2">
   347→                    <span className="rounded-md bg-primary/10 px-2 py-1 text-xs font-semibold tabular-nums text-primary">
   348→                      {new Date(s.startTime).toLocaleTimeString("tr-TR", { hour: "2-digit", minute: "2-digit" })}–{new Date(s.endTime).toLocaleTimeString("tr-TR", { hour: "2-digit", minute: "2-digit" })}
   349→                    </span>
   350→                    <p className="min-w-0 flex-1 truncate font-semibold">{s.title}</p>
   351→                    <Chip tone={s.type === "KEYNOTE" ? "violet" : s.type === "BREAK" ? "neutral" : "teal"}>{s.type}</Chip>
   352→                    {s.room && <Chip>{s.room.name}</Chip>}
   353→                    <StatusBadge map={SESSION_STATUS} value={s.status} />
   354→                    {clashes.has(s.id) && <Chip tone="rose">çakışma</Chip>}
   355→                    {!s.isVisible && s.status === "PUBLISHED" && <Chip tone="amber">gizli</Chip>}
   356→                  </div>
   357→                  {s.submission && <p className="mt-1 text-xs text-muted-foreground">kaynak bildiri: {s.submission.code} — {s.submission.title}</p>}
   358→                  <div className="mt-2 flex flex-wrap gap-1.5">
   359→                    {s.assignments.map((a) => {
   360→                      const name = a.person ? `${a.person.firstName} ${a.person.lastName}` : a.participation ? `${a.participation.person.firstName} ${a.participation.person.lastName}` : "—";
   361→                      return <Chip key={a.id} tone={a.status === "CONFIRMED" ? "emerald" : "amber"}>{label2(EVENT_ROLES, a.role)}: {name}</Chip>;
   362→                    })}
   363→                  </div>
   364→                  {s.status !== "PUBLISHED" && (
   365→                    <Button size="sm" variant="outline" className="mt-3" onClick={() => setPublishTarget(s)} disabled={clashes.has(s.id)}>
   366→                      <Icons.Upload className="size-4" /> Yayınla
   367→                    </Button>
   368→                  )}
   369→                </div>
   370→              ))}
   371→            </div>
   372→          )}
   373→        </TabsContent>
   374→
   375→        {cmeEnabled && (
   376→          <TabsContent value="cme" className="mt-4 space-y-4">
   377→            {cmeLoading ? <Loading rows={5} /> : cmeError ? <ErrorState message={cmeError} onRetry={reloadCme} /> : !cme ? (
   378→              <EmptyState title="CME verisi yok" desc="Veri yüklenemedi — yenile düğmesiyle tekrar deneyin." />

   380→              <>
   381→                {/* Üst KPI satırı */}
   382→                <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   383→                  <KpiCard label="Kredili Oturum" value={cme.summary.sessionsWithCredits} sub={`${cme.summary.sessionsTotal} oturum`} icon={<Icons.GraduationCap className="size-4" />} />
   384→                  <KpiCard label="Kredi Potansiyeli" value={cme.summary.creditsPotential} sub="kredili oturumların toplamı" tone="violet" icon={<Icons.Sigma className="size-4" />} />
   385→                  <KpiCard label="Kredi Kazanan" value={cme.summary.attendees} sub="katılımcı" tone="emerald" icon={<Icons.UserCheck className="size-4" />} />
   386→                  <KpiCard label="Dağıtılan Kredi" value={cme.summary.creditsIssued} sub={`ort. ${cme.summary.avgCredits} kredi/kişi`} tone="amber" icon={<Icons.Award className="size-4" />} />
   387→                </div>
   388→
   389→                {/* Kapsam satırı + resmî rapor aksiyonları */}
   390→                <div className="flex flex-wrap items-center gap-2 gap-y-2 rounded-xl border bg-card px-4 py-3 shadow-sm sm:gap-3">
   391→                  <span className="text-xs font-medium text-muted-foreground">Kapsam</span>
   392→                  <div className="h-1.5 min-w-24 flex-1 overflow-hidden rounded bg-muted">
   393→                    <div className="h-full rounded bg-teal-500 transition-all duration-300" style={{ width: `${cme.summary.coveragePercent}%` }} />
   394→                  </div>
   395→                  <span className="whitespace-nowrap text-xs text-muted-foreground">{`Oturumların %${cme.summary.coveragePercent}'i kredili`}</span>
   396→                  <Button size="sm" variant="outline" className="ml-auto h-8 shrink-0" onClick={() => setReportOpen(true)} disabled={!currentEditionId}>
   397→                    <Icons.FileBadge className="size-3.5" /> Resmî Rapor
   398→                  </Button>
   399→                  <Button size="sm" variant="ghost" className="h-8 shrink-0" onClick={() => window.open(`/api/cme/report?editionId=${encodeURIComponent(currentEditionId ?? "")}&format=csv`, "_blank")}>
   400→                    <Icons.Download className="size-3.5" /> CSV
   401→                  </Button>
   402→                </div>
   403→
   404→                {/* Oturum kredi editörü */}
   405→                <SectionCard title="Oturum Kredileri" desc="Oturuma atanacak CME kredisini girin — oturum taraması gerçekleştiğinde kişiye işlenir">
   406→                  <div className="maven-scroll max-h-96 overflow-auto">
   407→                    <table className="w-full min-w-[560px] text-sm">
   408→                      <thead className="sticky top-0 z-10 bg-card text-left text-muted-foreground">
   409→                        <tr className="border-b">
   410→                          <th className="px-3 py-2 text-xs font-medium">Oturum</th>
   411→                          <th className="px-3 py-2 text-xs font-medium">Saat</th>
   412→                          <th className="px-3 py-2 text-xs font-medium">Katılım</th>
   413→                          <th className="px-3 py-2 text-right text-xs font-medium">Kredi</th>
   414→                        </tr>
   415→                      </thead>
   416→                      <tbody>
   417→                        {cme.sessions.map((s) => (
   418→                          <tr key={s.id} className="border-b last:border-0 hover:bg-muted/30">
   419→                            <td className="px-3 py-2.5">
   420→                              <p className="max-w-64 truncate font-semibold">{s.title}</p>
   421→                              <div className="mt-1"><Chip tone={cmeTypeTone(s.type)}>{s.type}</Chip></div>
   422→                            </td>
   423→                            <td className="whitespace-nowrap px-3 py-2.5 tabular-nums text-muted-foreground">{cmeTime(s.startTime)}</td>
   424→                            <td className="whitespace-nowrap px-3 py-2.5 tabular-nums">{s.attendanceCount} kişi</td>
   425→                            <td className="px-3 py-2.5">
   426→                              <div className="flex items-center justify-end gap-1.5">
   427→                                <Input
   428→                                  type="number" inputMode="decimal" min={0} max={99} step={0.5}
   429→                                  value={creditInputs[s.id] ?? ""} placeholder="—"
   430→                                  onChange={(e) => setCreditInputs((prev) => ({ ...prev, [s.id]: e.target.value }))}
   431→                                  className="h-8 w-20" aria-label={`${s.title} — CME kredisi`}
   432→                                />
   433→                                <Button
   434→                                  size="sm" variant="outline" className="h-8"
   435→                                  disabled={savingId !== null || (creditInputs[s.id] ?? "").trim() === ""}
   436→                                  onClick={() => saveCredits(s)}
   437→                                >
   438→                                  {savingId === s.id ? <Icons.Loader2 className="size-3.5 animate-spin" /> : <Icons.Check className="size-3.5" />}
   439→                                  Kaydet
   440→                                </Button>
   441→                              </div>
   442→                            </td>
   443→                          </tr>
   444→                        ))}
   445→                      </tbody>
   446→                    </table>
   447→                  </div>
   448→                </SectionCard>
   449→
   450→                {/* Tür bazlı toplu atama */}
   451→                <SectionCard title="Tür Bazlı Toplu Ata" desc="Tür şablonunu oturumlara uygula">
   452→                  <div className="flex flex-wrap items-end gap-3">
   453→                    {CME_SESSION_TYPES.map((t) => (
   454→                      <div key={t} className="space-y-1">
   455→                        <Label className="text-xs text-muted-foreground">{t}</Label>
   456→                        <Input
   457→                          type="number" min={0} max={99} step={0.5} placeholder="0"
   458→                          value={bulkDefaults[t] ?? ""}
   459→                          onChange={(e) => setBulkDefaults((prev) => ({ ...prev, [t]: e.target.value }))}
   460→                          className="h-8 w-20" aria-label={`${t} oturumları için varsayılan kredi`}
   461→                        />
   462→                      </div>
   463→                    ))}
   464→                    <div className="ml-auto flex flex-wrap items-center gap-2">
   465→                      <span className="text-xs text-muted-foreground">Yalnız kredisiz oturumlara uygulanır</span>
   466→                      <Button size="sm" onClick={applyBulk} disabled={bulkBusy || !bulkFilled}>
   467→                        {bulkBusy && <Icons.Loader2 className="size-4 animate-spin" />}
   468→                        Boş Kredilere Uygula
   469→                      </Button>
   470→                    </div>
   471→                  </div>
   472→                </SectionCard>
   473→
   474→                {/* Kişi bazlı kredi defteri */}
   475→                <SectionCard title="Kredi Defteri" desc="Kişi bazlı CME birikimi — krediye göre sıralı">
   476→                  {cme.ledger.length === 0 ? (
   477→                    <EmptyState title="Defter boş" desc="Oturum taraması ve kredi bekleniyor" />
   478→                  ) : (
   479→                    <div className="maven-scroll max-h-96 overflow-auto">
   480→                      <table className="w-full min-w-[640px] text-sm">
   481→                        <thead className="sticky top-0 z-10 bg-card text-left text-muted-foreground">
   482→                          <tr className="border-b">
   483→                            <th className="px-3 py-2 text-xs font-medium">Kişi</th>
   484→                            <th className="px-3 py-2 text-xs font-medium">Roller</th>
   485→                            <th className="px-3 py-2 text-xs font-medium">Katılım</th>
   486→                            <th className="px-3 py-2 text-right text-xs font-medium">Kredi</th>
   487→                            <th className="px-3 py-2 text-xs font-medium">İlerleme</th>
   488→                            <th className="px-3 py-2 text-xs font-medium">Son Etkinlik</th>
   489→                          </tr>
   490→                        </thead>
   491→                        <tbody>
   492→                          {cme.ledger.map((l) => (
   493→                            <tr key={l.participationId} className={cn("border-b last:border-0", l.attendedCount === 0 && "opacity-60")}>
   494→                              <td className="px-3 py-2.5">
   495→                                <p className="font-semibold">{l.person.fullName}</p>
   496→                                {(l.person.company || l.person.title) && (
   497→                                  <p className="max-w-48 truncate text-xs text-muted-foreground">{[l.person.company, l.person.title].filter(Boolean).join(" · ")}</p>
   498→                                )}
   499→                              </td>
   500→                              <td className="px-3 py-2.5">
   501→                                <div className="flex flex-wrap gap-1">
   502→                                  {l.roles.slice(0, 2).map((r) => <Chip key={r} tone="teal">{label(EVENT_ROLES, r)}</Chip>)}
   503→                                  {l.roles.length > 2 && <Chip tone="neutral">+{l.roles.length - 2}</Chip>}
   504→                                  {l.roles.length === 0 && <span className="text-xs text-muted-foreground">—</span>}
   505→                                </div>
   506→                              </td>
   507→                              <td className="whitespace-nowrap px-3 py-2.5 tabular-nums">{l.attendedCount}/{l.eligibleCount} oturum</td>
   508→                              <td className="px-3 py-2.5 text-right font-semibold tabular-nums">{l.credits}</td>
   509→                              <td className="px-3 py-2.5">
   510→                                <div className="flex items-center gap-2">
   511→                                  <div className="h-1.5 w-16 overflow-hidden rounded bg-muted">
   512→                                    <div className="h-full rounded bg-teal-500" style={{ width: `${l.percent}%` }} />
   513→                                  </div>
   514→                                  <span className="text-xs tabular-nums text-muted-foreground">%{l.percent}</span>
   515→                                </div>
   516→                              </td>
   517→                              <td className="whitespace-nowrap px-3 py-2.5 text-xs text-muted-foreground">{l.lastActivity ? fmtDateTime(l.lastActivity) : "—"}</td>
   518→                            </tr>
   519→                          ))}
   520→                        </tbody>
   521→                      </table>
   522→                    </div>
   523→                  )}
   524→                </SectionCard>
   525→              </>
   526→            )}
   527→          </TabsContent>
   528→        )}
   529→      </Tabs>
   530→
   531→      <Dialog open={Boolean(publishTarget)} onOpenChange={(o) => !o && setPublishTarget(null)}>
   532→        <DialogContent className="sm:max-w-sm">
   533→          <DialogHeader><DialogTitle>Oturumu Yayınla</DialogTitle><DialogDescription>{publishTarget?.title}</DialogDescription></DialogHeader>
   534→          <p className="text-xs text-muted-foreground">Yayın düğmesi etkilenen konuşmacı ve katılımcı sayısını gösterir; kişisel programlarda görünür olur.</p>
   535→          <DialogFooter>
   536→            <Button variant="outline" onClick={() => setPublishTarget(null)}>Vazgeç</Button>
   537→            <Button onClick={publish} disabled={busy}>{busy ? "Yayınlanıyor…" : "Yayınla"}</Button>
   538→          </DialogFooter>
   539→        </DialogContent>
   540→      </Dialog>
   541→
   542→      {reportOpen && currentEditionId && (
   543→        <CmeReportOverlay editionId={currentEditionId} onClose={() => setReportOpen(false)} />
   544→      )}
   545→    </div>
   546→  );
   547→}
