     1→"use client";
     2→// Konaklama — gecelik stok (§32), rezervasyon teyidi (§09-E), oda-gece metriği
     3→// R9-d genişletmesi: occupancy/rate/gece düzenleme, no-show akışı, misafir hiyerarşisi (bağımlı kişi + refakatçi).
     4→import { useState } from "react";
     5→import { listEntity, apiSend } from "@/lib/client";
     6→import { useApp } from "@/lib/store";
     7→import { SectionCard, EmptyState, Loading, ErrorState, useApi, PageHeader, StatusBadge, Chip, KpiCard } from "../bits";
     8→import { ACCOMMODATION_STATUS, OCCUPANCY_TYPE, RELATION_TYPE, fmtDate, fmtMoney, label } from "@/lib/constants";
     9→import { Button } from "@/components/ui/button";
    10→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from "@/components/ui/dialog";
    11→import { Input } from "@/components/ui/input";
    12→import { Label } from "@/components/ui/label";
    13→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    14→import { Checkbox } from "@/components/ui/checkbox";
    15→import { Switch } from "@/components/ui/switch";
    16→import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from "@/components/ui/tooltip";
    17→import { useToast } from "@/hooks/use-toast";
    18→import * as Icons from "lucide-react";
    19→import { cn } from "@/lib/utils";
    20→
    21→interface HotelRow {
    22→  id: string; name: string; city?: string | null; district?: string | null; contactName?: string | null; contactPhone?: string | null;
    23→  roomTypes: { id: string; name: string; capacity: number; pricePerNight: number; currency: string }[];
    24→  blocks: { id: string; name: string; releaseDate?: string | null; cancellationPolicy?: string | null; roomType: { name: string; id: string }; inventoryNights: { id: string; date: string; totalRooms: number; reservedRooms: number }[] }[];
    25→}
    26→interface ReservationRow {
    27→  id: string; guestName: string; checkIn: string; checkOut: string; status: string; payerType: string; payerName?: string | null;
    28→  occupancyType?: string | null; ratePerNight?: number; nights?: number; noShow?: boolean; noShowFee?: number;
    29→  block?: { name: string; hotel: { name: string }; roomType: { name: string } } | null;
    30→  primaryGuest?: { id: string; person: { id: string; firstName: string; lastName: string } } | null;
    31→  occupancySlots: { id: string; guestName?: string | null; position: number }[];
    32→}
    33→interface PersonRow {
    34→  id: string; firstName: string; lastName: string; email?: string | null; parentPersonId?: string | null; relationType?: string | null;
    35→}
    36→interface CompanionRow {
    37→  id: string; participationId: string; name: string; type: string; notes?: string | null;
    38→}
    39→
    40→const NO_SHOW_NOTE = "Gerçekleşmeyen konaklama ücreti faturaya no-show kalemi olarak yansır — sistemden düşme kaydıdır.";
    41→const COMPANION_AGE = { ADULT: "Yetişkin", CHILD: "Çocuk", INFANT: "Bebek (0-2)" } as const;
    42→
    43→export function AccommodationView() {
    44→  const { currentEditionId, tenant, bump, refreshKey } = useApp();
    45→  const { toast } = useToast();
    46→  const [busyId, setBusyId] = useState<string | null>(null);
    47→
    48→  const { data: hotels, error, reload, loading } = useApi<HotelRow[]>(() => listEntity<HotelRow>("hotels", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey]);
    49→  const { data: reservations, reload: reloadRes } = useApi<ReservationRow[]>(() => listEntity<ReservationRow>("reservations", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey]);
    50→  const { data: people, reload: reloadPeople } = useApi<PersonRow[]>(() => listEntity<PersonRow>("people", { tenantId: tenant?.id ?? undefined, limit: 500 }), [tenant?.id, refreshKey]);
    51→
    52→  // ── düzenleme / no-show / misafir diyaloğu durumları
    53→  const [editRes, setEditRes] = useState<ReservationRow | null>(null);
    54→  const [editForm, setEditForm] = useState({ occupancyType: "__none__", ratePerNight: 0, nights: 0, noShow: false, noShowFee: 0 });
    55→  const [noShowRes, setNoShowRes] = useState<ReservationRow | null>(null);
    56→  const [noShowFee, setNoShowFee] = useState(0);
    57→  const [guestRes, setGuestRes] = useState<ReservationRow | null>(null);
    58→  const [addGuestOpen, setAddGuestOpen] = useState(false);
    59→  const [guestForm, setGuestForm] = useState({ firstName: "", lastName: "", relationType: "SPOUSE", ageType: "ADULT", makeCompanion: true, addSlot: true });
    60→  const [busy, setBusy] = useState(false);
    61→  const [guestTick, setGuestTick] = useState(0);
    62→
    63→  const { data: companions, reload: reloadCompanions } = useApi<CompanionRow[]>(
    64→    () => listEntity<CompanionRow>("companions", { participationId: guestRes?.primaryGuest?.id ?? "__none__" }),
    65→    [guestRes?.primaryGuest?.id, refreshKey, guestTick],
    66→  );
    67→
    68→  const nightsOf = (r: ReservationRow) => Math.max(1, Math.round((new Date(r.checkOut).getTime() - new Date(r.checkIn).getTime()) / 86400000));
    69→  const nightsSaved = (r: ReservationRow) => (r.nights && r.nights > 0 ? r.nights : nightsOf(r));
    70→  const totalOf = (r: ReservationRow) => (r.ratePerNight ?? 0) * nightsSaved(r);
    71→  const roomNightsSold = (reservations ?? []).filter((r) => ["RESERVED", "CONFIRMED", "CHECKED_IN"].includes(r.status)).reduce((s, r) => s + nightsSaved(r), 0);
    72→  const pendingRes = (reservations ?? []).filter((r) => ["REQUESTED", "WAITLIST"].includes(r.status));
    73→  const confirmedCount = (reservations ?? []).filter((r) => ["CONFIRMED", "CHECKED_IN"].includes(r.status)).length;
    74→  const noShowList = (reservations ?? []).filter((r) => r.noShow);
    75→  const noShowFeeTotal = noShowList.reduce((s, r) => s + (r.noShowFee ?? 0), 0);
    76→  const occupancyBreakdown = (["SINGLE", "DOUBLE", "FAMILY_SHARED"] as const).map((t) => ({
    77→    type: t,
    78→    count: (reservations ?? []).filter((r) => r.occupancyType === t && !r.noShow && r.status !== "CANCELLED").length,
    79→  }));
    80→
    81→  const confirm = async (r: ReservationRow) => {
    82→    setBusyId(r.id);
    83→    try {
    84→      await apiSend("/api/flows", "POST", { action: "reservation.confirm", reservationId: r.id });
    85→      toast({ title: "Rezervasyon teyit edildi", description: `${nightsOf(r)} oda-gece stoktan tüketildi.` });
    86→      reload(); bump();
    87→    } catch (e) {
    88→      toast({ title: "Teyit engellendi", description: e instanceof Error ? e.message : "Bir gece eksikse 'teyitli' görünmez.", variant: "destructive" });
    89→    } finally { setBusyId(null); }
    90→  };
    91→
    92→  // ── rezervasyon düzenleme (occupancy / rate / gece / no-show)
    93→  const openEdit = (r: ReservationRow) => {
    94→    setEditForm({
    95→      occupancyType: r.occupancyType ?? "__none__",
    96→      ratePerNight: r.ratePerNight ?? 0,
    97→      nights: nightsSaved(r),
    98→      noShow: r.noShow ?? false,
    99→      noShowFee: r.noShowFee ?? 0,
   100→    });
   101→    setEditRes(r);
   102→  };
   103→
   104→  const saveEdit = async () => {
   105→    if (!editRes) return;
   106→    setBusy(true);
   107→    try {
   108→      await apiSend(`/api/reservations/${editRes.id}`, "PUT", {
   109→        occupancyType: editForm.occupancyType === "__none__" ? null : editForm.occupancyType,
   110→        ratePerNight: Number(editForm.ratePerNight) || 0,
   111→        nights: Math.max(0, Math.round(Number(editForm.nights) || 0)),
   112→        noShow: editForm.noShow,
   113→        noShowFee: editForm.noShow ? Number(editForm.noShowFee) || 0 : 0,
   114→      });
   115→      toast({ title: "Rezervasyon güncellendi", description: `${editRes.guestName} · oda tipi/fiyat/gece kaydedildi.` });
   116→      setEditRes(null);
   117→      reloadRes(); bump();
   118→    } catch (e) {
   119→      toast({ title: "Güncellenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   120→    } finally { setBusy(false); }
   121→  };
   122→
   123→  // ── no-show hızlı aksiyon (sistemden düş)
   124→  const openNoShow = (r: ReservationRow) => {
   125→    setNoShowFee((r.ratePerNight ?? 0) > 0 ? (r.ratePerNight ?? 0) * nightsSaved(r) : 0);
   126→    setNoShowRes(r);
   127→  };
   128→
   129→  const markNoShow = async () => {
   130→    if (!noShowRes) return;
   131→    setBusy(true);
   132→    try {
   133→      await apiSend(`/api/reservations/${noShowRes.id}`, "PUT", { noShow: true, noShowFee: Number(noShowFee) || 0 });
   134→      toast({ title: "No-show işaretlendi", description: `${noShowRes.guestName} sistemden düşüldü — ücret faturaya kalem olarak yansır.` });
   135→      setNoShowRes(null);
   136→      reloadRes(); bump();
   137→    } catch (e) {
   138→      toast({ title: "İşaretlenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   139→    } finally { setBusy(false); }
   140→  };
   141→
   142→  // ── misafir hiyerarşisi: bağımlı kişi + refakatçi + konuk slotu
   143→  const dependentsOf = (personId: string) => (people ?? []).filter((p) => p.parentPersonId === personId);
   144→  const primaryDependents = guestRes?.primaryGuest ? dependentsOf(guestRes.primaryGuest.person.id) : [];
   145→
   146→  const openAddGuest = () => {
   147→    setGuestForm({ firstName: "", lastName: "", relationType: "SPOUSE", ageType: "ADULT", makeCompanion: true, addSlot: true });
   148→    setAddGuestOpen(true);
   149→  };
   150→
   151→  const addGuestProfile = async () => {
   152→    const r = guestRes;
   153→    if (!r?.primaryGuest || !tenant?.id || !currentEditionId || !guestForm.firstName.trim() || !guestForm.lastName.trim()) return;
   154→    setBusy(true);
   155→    try {
   156→      // 1) Person — self-referencing parentPersonId ile (Parent_ID kuralı)
   157→      const person = await apiSend<{ id: string }>("/api/people", "POST", {
   158→        tenantId: tenant.id,
   159→        firstName: guestForm.firstName.trim(),
   160→        lastName: guestForm.lastName.trim(),
   161→        parentPersonId: r.primaryGuest.person.id,
   162→        relationType: guestForm.relationType,
   163→      });
   164→      // 2) Katılım (bu edisyonda yoksa) — @@unique([editionId, personId]) güvencesi
   165→      const existing = await listEntity<{ id: string }>("participations", { editionId: currentEditionId, personId: person.id });
   166→      let participationId = existing[0]?.id ?? null;
   167→      let participationNote = "mevcut katılım";
   168→      if (!participationId) {
   169→        const created = await apiSend<{ id: string }>("/api/participations", "POST", { editionId: currentEditionId, personId: person.id });
   170→        participationId = created.id;
   171→        participationNote = "yeni katılım";
   172→      }
   173→      // 3) Refakatçi kaydı (ana konuğun katılımına)
   174→      if (guestForm.makeCompanion) {
   175→        await apiSend("/api/companions", "POST", {
   176→          participationId: r.primaryGuest.id,
   177→          name: `${guestForm.firstName.trim()} ${guestForm.lastName.trim()}`,
   178→          type: guestForm.ageType,
   179→        });
   180→      }
   181→      // 4) Rezervasyon konuk slotu
   182→      if (guestForm.addSlot && participationId) {
   183→        await apiSend("/api/occupancy-slots", "POST", {
   184→          reservationId: r.id,
   185→          participationId,
   186→          guestName: `${guestForm.firstName.trim()} ${guestForm.lastName.trim()}`,
   187→        });
   188→      }
   189→      toast({
   190→        title: "Misafir profili eklendi",
   191→        description: `Kişi + ${participationNote}${guestForm.makeCompanion ? " + refakatçi" : ""}${guestForm.addSlot ? " + konuk slotu" : ""} kuruldu.`,
   192→      });
   193→      setAddGuestOpen(false);
   194→      reloadPeople(); reloadCompanions(); reloadRes(); setGuestTick((t) => t + 1); bump();
   195→    } catch (e) {
   196→      toast({ title: "Misafir eklenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   197→    } finally { setBusy(false); }
   198→  };
   199→
   200→  return (
   201→    <div className="space-y-5">
   202→      <PageHeader title="Konaklama & Seyahat" desc="Oda stoğu gün bazlıdır — 1 oda × 3 gece = 3 oda-gece; bir gece eksikse teyit engellenir" />
   203→
   204→      <div className="grid grid-cols-2 gap-3 md:grid-cols-3 xl:grid-cols-5">
   205→        <KpiCard label="Otel" value={(hotels ?? []).length} icon={<Icons.Building className="size-4" />} />
   206→        <KpiCard label="Rezervasyon" value={(reservations ?? []).length} sub={`${pendingRes.length} teyit bekleyen`} tone="amber" icon={<Icons.BedDouble className="size-4" />} />
   207→        <KpiCard label="Oda-Gece (satış)" value={roomNightsSold} sub="ayrılan toplam" tone="emerald" icon={<Icons.Moon className="size-4" />} />
   208→        <KpiCard label="Bekleme Listesi" value={(reservations ?? []).filter((r) => r.status === "WAITLIST").length} tone="rose" icon={<Icons.Clock className="size-4" />} />
   209→        <KpiCard
   210→          label="No-Show"
   211→          value={noShowList.length}
   212→          sub={noShowList.length ? `${fmtMoney(noShowFeeTotal)} toplam no-show ücreti` : "gerçekleşmeyen konaklama yok"}
   213→          tone={noShowList.length ? "rose" : "neutral"}
   214→          icon={<Icons.UserX className="size-4" />}
   215→        />
   216→      </div>
   217→
   218→      {/* doluluk tipi dağılımı */}
   219→      <div className="flex flex-wrap items-center gap-2 rounded-xl border bg-card px-4 py-2.5 shadow-sm">
   220→        <span className="inline-flex items-center gap-1.5 text-xs font-medium text-muted-foreground"><Icons.BedSingle className="size-3.5" /> Doluluk tipi:</span>
   221→        {occupancyBreakdown.map((o) => (
   222→          <Chip key={o.type} tone={o.count ? "teal" : "neutral"}>{label(OCCUPANCY_TYPE, o.type)}: <b className="tabular-nums">{o.count}</b></Chip>
   223→        ))}
   224→        <span className="ml-auto text-[11px] text-muted-foreground">no-show ve iptaller dağılıma katılmaz</span>
   225→      </div>
   226→
   227→      {loading ? <Loading /> : error ? <ErrorState message={error} onRetry={reload} /> : (hotels ?? []).length === 0 ? (
   228→        <EmptyState title="Henüz oda bloğu tanımlanmadı" desc="Konaklama taleplerini açmadan önce otel ve tarihleri ekleyin." />
   229→      ) : (
   230→        (hotels ?? []).map((h) => (
   231→          <SectionCard key={h.id} title={h.name} desc={`${h.district ?? h.city ?? ""} · iletişim: ${h.contactName ?? "—"}`}>
   232→            <div className="space-y-4">
   233→              {h.blocks.map((b) => {
   234→                const total = b.inventoryNights.reduce((s, n) => s + n.totalRooms, 0);
   235→                const reserved = b.inventoryNights.reduce((s, n) => s + n.reservedRooms, 0);
   236→                return (
   237→                  <div key={b.id} className="rounded-lg border p-3">
   238→                    <div className="flex flex-wrap items-center justify-between gap-2">
   239→                      <p className="text-sm font-semibold">{b.name} <span className="font-normal text-muted-foreground">· {b.roomType.name}</span></p>
   240→                      <div className="flex items-center gap-2 text-xs text-muted-foreground">
   241→                        <span>satılabilir {total} oda-gece · ayrılan {reserved} (%{total ? Math.round((reserved / total) * 100) : 0})</span>
   242→                        {b.releaseDate && <Chip tone={new Date(b.releaseDate) < new Date(Date.now() + 7 * 86400000) ? "rose" : "neutral"}>release {fmtDate(b.releaseDate)}</Chip>}
   243→                      </div>
   244→                    </div>
   245→                    {/* gecelik stok çizelgesi */}
   246→                    <div className="mt-3 flex flex-wrap gap-1.5">
   247→                      {b.inventoryNights.map((n) => {
   248→                        const free = n.totalRooms - n.reservedRooms;
   249→                        const pctFill = n.totalRooms ? (n.reservedRooms / n.totalRooms) * 100 : 0;
   250→                        return (
   251→                          <div key={n.id} className={cn("w-20 rounded-md border p-1.5 text-center text-[10px]", free === 0 ? "border-rose-300 bg-rose-50" : pctFill > 80 ? "border-amber-300 bg-amber-50" : "border-emerald-200 bg-emerald-50/60")}>
   252→                            <p className="font-semibold">{fmtDate(n.date).slice(0, 6)}</p>
   253→                            <p className="tabular-nums font-bold text-xs">{n.reservedRooms}/{n.totalRooms}</p>
   254→                            <p className="text-muted-foreground">{free === 0 ? "dolu" : `${free} boş`}</p>
   255→                          </div>
   256→                        );
   257→                      })}
   258→                    </div>
   259→                  </div>
   260→                );
   261→              })}
   262→            </div>
   263→          </SectionCard>
   264→        ))
   265→      )}
   266→
   267→      <SectionCard title="Rezervasyonlar" desc="varış/çıkış, doluluk tipi, gecelik fiyat, misafir bağlantıları — rezervasyon ile ödeyen aynı olmak zorunda değil (§35)">
   268→        {(reservations ?? []).length === 0 ? (
   269→          <EmptyState title="Rezervasyon yok" />
   270→        ) : (
   271→          <div className="maven-scroll max-h-96 space-y-2 overflow-y-auto pr-1">
   272→            {(reservations ?? []).map((r) => {
   273→              const total = totalOf(r);
   274→              return (
   275→                <div key={r.id} className={cn("flex flex-wrap items-center gap-3 rounded-lg border p-3 transition-colors hover:bg-muted/30", r.noShow && "border-rose-200 bg-rose-50/40")}>
   276→                  <div className="min-w-0 flex-1">
   277→                    <p className="flex flex-wrap items-center gap-1.5 text-sm font-medium">
   278→                      {r.guestName}
   279→                      {r.primaryGuest && <span className="font-normal text-muted-foreground">← {r.primaryGuest.person.firstName} {r.primaryGuest.person.lastName}</span>}
   280→                      {r.noShow && <Chip tone="rose"><Icons.UserX className="mr-0.5 inline size-3" />NO-SHOW</Chip>}
   281→                    </p>
   282→                    <p className="text-xs text-muted-foreground">
   283→                      {r.block ? `${r.block.hotel.name} · ${r.block.roomType.name}` : "blok yok"} · {fmtDate(r.checkIn)} → {fmtDate(r.checkOut)} ({nightsOf(r)} gece) · ödeyen: {r.payerType}{r.payerName ? ` — ${r.payerName}` : ""}
   284→                    </p>
   285→                    <p className="mt-1 flex flex-wrap items-center gap-x-2.5 gap-y-1 text-[11px] text-muted-foreground">
   286→                      {r.occupancyType && <Chip tone="teal">{label(OCCUPANCY_TYPE, r.occupancyType)}</Chip>}
   287→                      {(r.ratePerNight ?? 0) > 0 && (
   288→                        <span className="tabular-nums">
   289→                          {fmtMoney(r.ratePerNight)}/gece × {nightsSaved(r)} gece = <b className="text-foreground">{fmtMoney(total)}</b>
   290→                        </span>
   291→                      )}
   292→                      {r.noShow && <span className="font-medium tabular-nums text-rose-600">no-show ücreti: {fmtMoney(r.noShowFee)}</span>}
   293→                    </p>
   294→                  </div>
   295→                  {r.occupancySlots.length > 0 && <Chip tone="teal">{r.occupancySlots.length} konuk slotu</Chip>}
   296→                  <StatusBadge map={ACCOMMODATION_STATUS} value={r.status} />
   297→                  <div className="flex flex-wrap items-center gap-1.5">
   298→                    {r.primaryGuest && (
   299→                      <Button size="sm" variant="outline" className="h-8 text-xs" onClick={() => { setGuestRes(r); setGuestTick((t) => t + 1); }}>
   300→                        <Icons.Users className="size-3.5" /> Konuklar
   301→                      </Button>
   302→                    )}
   303→                    <Button size="sm" variant="ghost" className="h-8 text-xs" onClick={() => openEdit(r)}>
   304→                      <Icons.Pencil className="size-3.5" /> Düzenle
   305→                    </Button>
   306→                    {!r.noShow && (
   307→                      <TooltipProvider delayDuration={200}>
   308→                        <Tooltip>
   309→                          <TooltipTrigger asChild>
   310→                            <Button size="sm" variant="ghost" className="h-8 text-xs text-rose-600 hover:bg-rose-50 hover:text-rose-700" onClick={() => openNoShow(r)} disabled={busyId === r.id}>
   311→                              <Icons.UserX className="size-3.5" /> No-show işaretle
   312→                            </Button>
   313→                          </TooltipTrigger>
   314→                          <TooltipContent side="top" className="max-w-56 text-xs">{NO_SHOW_NOTE}</TooltipContent>
   315→                        </Tooltip>
   316→                      </TooltipProvider>
   317→                    )}
   318→                    {["REQUESTED", "WAITLIST"].includes(r.status) && r.block && (
   319→                      <Button size="sm" onClick={() => confirm(r)} disabled={busyId === r.id}>
   320→                        {busyId === r.id ? "Kontrol ediliyor…" : "Teyit Et"}
   321→                      </Button>
   322→                    )}
   323→                  </div>
   324→                </div>
   325→              );
   326→            })}
   327→          </div>
   328→        )}
   329→      </SectionCard>
   330→
   331→      {/* Rezervasyon düzenleme diyaloğu */}
   332→      <Dialog open={Boolean(editRes)} onOpenChange={(o) => !o && setEditRes(null)}>
   333→        <DialogContent className="maven-scroll max-h-[85vh] overflow-y-auto sm:max-w-md">
   334→          <DialogHeader>
   335→            <DialogTitle>Rezervasyon Düzenle — {editRes?.guestName}</DialogTitle>
   336→            <DialogDescription>
   337→              {editRes?.block ? `${editRes.block.hotel.name} · ${editRes.block.roomType.name}` : "blok yok"} · {editRes && `${fmtDate(editRes.checkIn)} → ${fmtDate(editRes.checkOut)}`}
   338→            </DialogDescription>
   339→          </DialogHeader>
   340→          <div className="space-y-4">
   341→            <div className="grid gap-3 sm:grid-cols-2">
   342→              <div className="space-y-1.5">
   343→                <Label>Doluluk tipi</Label>
   344→                <Select value={editForm.occupancyType} onValueChange={(v) => setEditForm({ ...editForm, occupancyType: v })}>
   345→                  <SelectTrigger aria-label="Doluluk tipi"><SelectValue /></SelectTrigger>
   346→                  <SelectContent>
   347→                    <SelectItem value="__none__">Tanımsız</SelectItem>
   348→                    {Object.entries(OCCUPANCY_TYPE).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}
