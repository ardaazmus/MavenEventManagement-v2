   141→// ── R9-c: CV zaman çizelgesi (kişi 360 içi) ─────────────────────────────────
   142→function CvPanel({ personId, editionId }: { personId: string; editionId: string | null }) {
   143→  const { toast } = useToast();
   144→  const { data: cvs, error, reload, loading } = useApi<CvEntryRow[]>(
   145→    () => listEntity<CvEntryRow>("cv-entries", { personId, limit: 100 }),
   146→    [personId],
   147→  );
   148→  const [open, setOpen] = useState(false);
   149→  const [editing, setEditing] = useState<CvEntryRow | null>(null);
   150→  const [busy, setBusy] = useState(false);
   151→  const empty = { kind: "EXPERIENCE", title: "", organization: "", city: "", startDate: "", endDate: "", isCurrent: false, description: "", order: "0" };
   152→  const [form, setForm] = useState(empty);
   153→
   154→  const openNew = () => { setEditing(null); setForm(empty); setOpen(true); };
   155→  const openEdit = (c: CvEntryRow) => {
   156→    setEditing(c);
   157→    setForm({
   158→      kind: c.kind, title: c.title, organization: c.organization ?? "", city: c.city ?? "",
   159→      startDate: c.startDate ? c.startDate.slice(0, 10) : "", endDate: c.endDate ? c.endDate.slice(0, 10) : "",
   160→      isCurrent: c.isCurrent, description: c.description ?? "", order: String(c.order ?? 0),
   161→    });
   162→    setOpen(true);
   163→  };
   164→  const save = async () => {
   165→    if (!form.title.trim()) return;
   166→    setBusy(true);
   167→    try {
   168→      const payload = {
   169→        personId, editionId,
   170→        kind: form.kind, title: form.title.trim(), organization: form.organization || null, city: form.city || null,
   171→        startDate: form.startDate || null, endDate: form.isCurrent ? null : form.endDate || null,
   172→        isCurrent: form.isCurrent, description: form.description || null, order: Number(form.order) || 0,
   173→      };
   174→      if (editing) await apiSend(`/api/cv-entries/${editing.id}`, "PUT", payload);
   175→      else await apiSend("/api/cv-entries", "POST", payload);
   176→      toast({ title: editing ? "CV kaydı güncellendi" : "CV kaydı eklendi", description: form.title });
   177→      setOpen(false); reload();
   178→    } catch (e) {
   179→      toast({ title: "Hata", description: e instanceof Error ? e.message : "CV kaydı kaydedilemedi", variant: "destructive" });
   180→    } finally { setBusy(false); }
   181→  };
   182→  const remove = async (c: CvEntryRow) => {
   183→    try {
   184→      await apiSend(`/api/cv-entries/${c.id}`, "DELETE");
   185→      toast({ title: "CV kaydı silindi", description: c.title });
   186→      reload();
   187→    } catch (e) {
   188→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Silinemedi", variant: "destructive" });
   189→    }
   190→  };
   191→
   192→  const groups = CV_KIND_ORDER.map((k) => ({ kind: k, items: (cvs ?? []).filter((c) => c.kind === k).sort((a, b) => a.order - b.order) })).filter((g) => g.items.length > 0);
   193→  const canAdd = Boolean(editionId);
   194→
   195→  return (
   196→    <SectionCard
   197→      title="CV & Deneyimler"
   198→      desc="Eğitim, deneyim, ödül, dil, yayın ve sertifikalar — kişiye bağlı zaman çizelgesi"
   199→      action={
   200→        <Button size="sm" variant="outline" onClick={openNew} disabled={!canAdd} aria-label="CV kaydı ekle">
   201→          <Icons.Plus className="size-3.5" /> Ekle
   202→        </Button>
   203→      }
   204→    >
   205→      {!canAdd && <p className="mb-2 text-[11px] text-amber-600">CV kaydı için bir edisyon seçili olmalı (etkinlik izolasyonu).</p>}
   206→      {loading ? <Loading rows={2} /> : error ? <ErrorState message={error} onRetry={reload} /> : groups.length === 0 ? (
   207→        <p className="py-2 text-xs text-muted-foreground">Henüz CV kaydı yok — &quot;Ekle&quot; ile ilk kaydı girin.</p>
   208→      ) : (
   209→        <div className="maven-scroll max-h-96 space-y-4 overflow-y-auto pr-1">
   210→          {groups.map((g, gi) => (
   211→            <div key={g.kind} className="maven-stagger-item" style={{ animationDelay: `${gi * 60}ms` }}>
   212→              <p className="mb-2 flex items-center gap-1.5 text-xs font-semibold">
   213→                <Chip tone="teal">{label(CV_KIND, g.kind)}</Chip>
   214→                <span className="font-normal tabular-nums text-muted-foreground">{g.items.length} kayıt</span>
   215→              </p>
   216→              {/* zaman çizelgesi: sol hat + nokta düğümleri */}
   217→              <ol className="relative ml-1.5 space-y-3 border-l-2 border-teal-500/25 pl-4">
   218→                {g.items.map((c) => (
   219→                  <li key={c.id} className="group relative">
   220→                    <span className="absolute -left-[21.5px] top-1.5 size-2.5 rounded-full border-2 border-teal-500/60 bg-background" aria-hidden />
   221→                    <div className="rounded-lg border bg-muted/20 p-2.5 transition-colors hover:border-teal-500/30 hover:bg-muted/40">
   222→                      <div className="flex flex-wrap items-center justify-between gap-x-2 gap-y-1">
   223→                        <p className="min-w-0 text-sm font-medium leading-tight">{c.title}</p>
   224→                        <span className="flex shrink-0 items-center gap-1">
   225→                          {c.isCurrent && <Chip tone="emerald">Devam ediyor</Chip>}
   226→                          <button onClick={() => openEdit(c)} className="rounded p-1 text-muted-foreground opacity-60 transition hover:bg-muted hover:text-foreground group-hover:opacity-100" aria-label={`${c.title} kaydını düzenle`}>
   227→                            <Icons.Pencil className="size-3.5" />
   228→                          </button>
   229→                          <button onClick={() => remove(c)} className="rounded p-1 text-muted-foreground opacity-60 transition hover:bg-rose-50 hover:text-rose-600 group-hover:opacity-100" aria-label={`${c.title} kaydını sil`}>
   230→                            <Icons.Trash2 className="size-3.5" />
   231→                          </button>
   232→                        </span>
   233→                      </div>
   234→                      <p className="mt-0.5 text-[11px] text-muted-foreground">
   235→                        {[c.organization, c.city].filter(Boolean).join(" · ")}
   236→                        {(c.startDate || c.endDate) && (
   237→                          <span className="tabular-nums"> · {fmtDate(c.startDate)} — {c.isCurrent ? "…" : fmtDate(c.endDate)}</span>
   238→                        )}
   239→                      </p>
   240→                      {c.description && <p className="mt-1 text-xs leading-relaxed text-muted-foreground">{c.description}</p>}
   241→                    </div>
   242→                  </li>
   243→                ))}
   244→              </ol>
   245→            </div>
   246→          ))}
   247→        </div>
   248→      )}
   249→
   250→      <Dialog open={open} onOpenChange={setOpen}>
   251→        <DialogContent className="max-h-[85vh] overflow-y-auto maven-scroll sm:max-w-md">
   252→          <DialogHeader>
   253→            <DialogTitle>{editing ? "CV Kaydını Düzenle" : "Yeni CV Kaydı"}</DialogTitle>
   254→            <DialogDescription>Tür, başlık ve tarih aralığı — &quot;Devam ediyor&quot; işaretliyse bitiş tarihi yok sayılır.</DialogDescription>
   255→          </DialogHeader>
   256→          <div className="grid gap-3 sm:grid-cols-2">
   257→            <div>
   258→              <Label>Tür</Label>
   259→              <Select value={form.kind} onValueChange={(v) => setForm({ ...form, kind: v })}>
   260→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   261→                <SelectContent>
   262→                  {Object.entries(CV_KIND).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}
   263→                </SelectContent>
   264→              </Select>
   265→            </div>
   266→            <div><Label>Sıra</Label><Input type="number" className="mt-1 tabular-nums" value={form.order} onChange={(e) => setForm({ ...form, order: e.target.value })} /></div>
   267→            <div className="sm:col-span-2"><Label>Başlık *</Label><Input className="mt-1" value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="Örn. Endüstri Mühendisliği — Lisans" /></div>
   268→            <div><Label>Kurum / Okul</Label><Input className="mt-1" value={form.organization} onChange={(e) => setForm({ ...form, organization: e.target.value })} /></div>
   269→            <div><Label>Şehir</Label><Input className="mt-1" value={form.city} onChange={(e) => setForm({ ...form, city: e.target.value })} /></div>
   270→            <div><Label>Başlangıç</Label><Input type="date" className="mt-1 tabular-nums" value={form.startDate} onChange={(e) => setForm({ ...form, startDate: e.target.value })} /></div>
   271→            <div>
   272→              <Label className={cn(form.isCurrent && "opacity-50")}>Bitiş</Label>
   273→              <Input type="date" className="mt-1 tabular-nums" disabled={form.isCurrent} value={form.endDate} onChange={(e) => setForm({ ...form, endDate: e.target.value })} />
   274→            </div>
   275→            <div className="flex items-center gap-2 sm:col-span-2">
   276→              <Checkbox id="cv-current" checked={form.isCurrent} onCheckedChange={(v) => setForm({ ...form, isCurrent: v === true })} />
   277→              <Label htmlFor="cv-current" className="cursor-pointer text-sm font-normal">Devam ediyor</Label>
   278→            </div>
   279→            <div className="sm:col-span-2"><Label>Açıklama</Label><Textarea className="mt-1" rows={3} value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} /></div>
   280→          </div>
   281→          <DialogFooter>
   282→            <Button variant="outline" onClick={() => setOpen(false)}>Vazgeç</Button>
   283→            <Button onClick={save} disabled={busy || !form.title.trim()}>{busy ? "Kaydediliyor…" : "Kaydet"}</Button>
   284→          </DialogFooter>
   285→        </DialogContent>
   286→      </Dialog>
   287→    </SectionCard>
   288→  );
   289→}

   291→// ── R9-c: QR VCard kartı (kişi 360 içi) ─────────────────────────────────────
   292→function VCardPanel({ personId }: { personId: string }) {
   293→  const { toast } = useToast();
   294→  const { data, error, reload, loading } = useApi<PersonVCard | null>(
   295→    () => apiGet<PersonVCard>(`/api/people/${personId}/vcard?format=json`),
   296→    [personId],
   297→  );
   298→  const copyQr = async () => {
   299→    if (!data?.qrDataUrl) return;
   300→    try {
   301→      await navigator.clipboard.writeText(data.qrDataUrl);

   415→function RolesPanel() {
   416→  const { currentEditionId, bump, refreshKey } = useApp();
   417→  const { toast } = useToast();
   418→
   419→  // özel rol tanımları — hiyerarşi seviyesine göre sıralı (registry orderBy)
   420→  const { data: roles, error, reload, loading } = useApi<CustomRoleRow[]>(
   421→    () => listEntity<CustomRoleRow>("custom-roles", { editionId: currentEditionId ?? undefined, limit: 100 }),
   422→    [currentEditionId, refreshKey],
   423→  );
   424→  // katılımlar — rol ataması köprüsü
   425→  const { data: parts } = useApi<ParticipationLite[]>(
   426→    () => listEntity<ParticipationLite>("participations", { editionId: currentEditionId ?? undefined, limit: 50 }),
   427→    [currentEditionId, refreshKey],
   428→  );
   429→
   430→  const sorted = (roles ?? []).slice().sort((a, b) => a.hierarchyLevel - b.hierarchyLevel);
   431→
   432→  // ── rol oluştur/düzenle diyaloğu ──
   433→  const [dialogOpen, setDialogOpen] = useState(false);
   434→  const [editing, setEditing] = useState<CustomRoleRow | null>(null);
   435→  const [busy, setBusy] = useState(false);
   436→  const emptyRole = { name: "", key: "", keyTouched: false, hierarchyLevel: 50, color: "teal", permissions: [] as string[], description: "" };
   437→  const [form, setForm] = useState(emptyRole);
   438→  const keyHint = form.keyTouched ? form.key : slugifyKey(form.name);
   439→  const keyClash = Boolean(keyHint && sorted.some((r) => r.key === keyHint && r.id !== editing?.id));
   440→
   441→  const openNew = () => { setEditing(null); setForm(emptyRole); setDialogOpen(true); };
   442→  const openEdit = (r: CustomRoleRow) => {
   443→    setEditing(r);
   444→    setForm({ name: r.name, key: r.key, keyTouched: true, hierarchyLevel: r.hierarchyLevel, color: r.color ?? "teal", permissions: parsePerms(r.permissions), description: r.description ?? "" });
   445→    setDialogOpen(true);
   446→  };
   447→  const saveRole = async () => {
   448→    if (!form.name.trim() || !keyHint || keyClash || !currentEditionId) return;
   449→    setBusy(true);
   450→    try {
   451→      const payload = {
   452→        editionId: currentEditionId, name: form.name.trim(), key: keyHint, color: form.color,
   453→        hierarchyLevel: form.hierarchyLevel, permissions: JSON.stringify(form.permissions), description: form.description || null,
   454→      };
   455→      if (editing) await apiSend(`/api/custom-roles/${editing.id}`, "PUT", payload);
   456→      else await apiSend("/api/custom-roles", "POST", payload);
   457→      toast({ title: editing ? "Rol güncellendi" : "Rol oluşturuldu", description: `${form.name} · seviye ${form.hierarchyLevel}` });
   458→      setDialogOpen(false); reload(); bump();
   459→    } catch (e) {
   460→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Rol kaydedilemedi", variant: "destructive" });
   461→    } finally { setBusy(false); }
   462→  };
   463→  const toggleActive = async (r: CustomRoleRow) => {
   464→    try {
   465→      await apiSend(`/api/custom-roles/${r.id}`, "PUT", { isActive: !r.isActive });
   466→      toast({ title: r.isActive ? "Rol pasifleştirildi" : "Rol aktifleştirildi", description: r.name });
   467→      reload();
   468→    } catch (e) {
   469→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Durum değiştirilemedi", variant: "destructive" });
   470→    }
   471→  };
   472→  const deleteRole = async (r: CustomRoleRow) => {
   473→    try {
   474→      await apiSend(`/api/custom-roles/${r.id}`, "DELETE");
   475→      toast({ title: "Rol silindi", description: r.name });
   476→      reload(); bump();
   477→    } catch (e) {
   478→      toast({ title: "Silinemedi", description: e instanceof Error ? e.message : "Rola bağlı atamalar olabilir.", variant: "destructive" });
   479→    }
   480→  };
   481→
   482→  // ── rol ataması köprüsü ──
   483→  const [assignPart, setAssignPart] = useState("");
   484→  const [assignRole, setAssignRole] = useState(""); // "std:SPEAKER" | "cus:<id>"

   494→      } else {
   495→        await apiSend("/api/role-assignments", "POST", { participationId: assignPart, role: value, notes: "Rol panelinden atandı" });

   498→      setAssignPart(""); setAssignRole(""); bump();
   499→    } catch (e) {
