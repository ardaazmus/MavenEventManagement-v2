    61→function fmtSize(sizeKb?: number | null): string {
    62→  if (!sizeKb) return "—";
    63→  return sizeKb >= 1024 ? `${(sizeKb / 1024).toFixed(1).replace(".", ",")} MB` : `${sizeKb} KB`;
    64→}

    66→// mime → MEDIA_KIND (uzantı yedeğiyle)
    67→function kindFromMime(mime: string, fileName: string): string {
    68→  const m = (mime || "").toLowerCase();
    69→  const ext = fileName.split(".").pop()?.toLowerCase() ?? "";
    70→  if (m.startsWith("image/")) return "IMAGE";
    71→  if (m.startsWith("video/")) return "VIDEO";
    72→  if (m.startsWith("audio/")) return "AUDIO";
    73→  if (m.includes("spreadsheet") || m.includes("excel") || m.includes("csv")) return "SPREADSHEET";
    74→  if (m.includes("pdf") || m.includes("word") || m.includes("presentation") || m.startsWith("text/")) return "DOCUMENT";
    75→  if (m.includes("zip") || m.includes("compressed") || m.includes("tar") || m.includes("rar") || m.includes("7z") || m.includes("octet-stream")) return "ARCHIVE";
    76→  if (m.startsWith("font/")) return "FONT";
    77→  if (["png", "jpg", "jpeg", "gif", "webp", "svg", "avif"].includes(ext)) return "IMAGE";
    78→  if (["mp4", "mov", "webm", "avi", "mkv"].includes(ext)) return "VIDEO";
    79→  if (["mp3", "wav", "m4a", "aac"].includes(ext)) return "AUDIO";
    80→  if (["pdf", "doc", "docx", "ppt", "pptx", "txt", "md"].includes(ext)) return "DOCUMENT";
    81→  if (["xls", "xlsx", "csv", "ods"].includes(ext)) return "SPREADSHEET";
    82→  if (["zip", "rar", "7z", "tar", "gz", "dwg"].includes(ext)) return "ARCHIVE";
    83→  if (["ttf", "otf", "woff", "woff2"].includes(ext)) return "FONT";
    84→  return "OTHER";
    85→}

   126→  const tree = useMemo<FolderNode[]>(() => {
   127→    const list = folders ?? [];
   128→    const byId = new Map<string, FolderNode>();
   129→    for (const f of list) byId.set(f.id, { folder: f, children: [], count: 0, depth: 0 });
   130→    const roots: FolderNode[] = [];
   131→    for (const node of byId.values()) {
   132→      const parent = node.folder.parentId ? byId.get(node.folder.parentId) : undefined;
   133→      if (parent) { parent.children.push(node); node.depth = parent.depth + 1; } else roots.push(node);
   134→    }
   135→    const counts = new Map<string, number>();
   136→    for (const a of assets ?? []) {
   137→      if (a.folderId) counts.set(a.folderId, (counts.get(a.folderId) ?? 0) + 1);
   138→    }
   139→    const walk = (n: FolderNode): number => {
   140→      n.count = counts.get(n.folder.id) ?? 0;
   141→      for (const c of n.children) n.count += walk(c);
   142→      return n.count;
   143→    };
   144→    roots.forEach(walk);
   145→    const sortRec = (nodes: FolderNode[]) => {
   146→      nodes.sort((a, b) => a.folder.name.localeCompare(b.folder.name, "tr"));
   147→      nodes.forEach((n) => sortRec(n.children));
   148→    };
   149→    sortRec(roots);
   150→    return roots;
   151→  }, [folders, assets]);

   158→      if (n.folder.id === selectedId) {
   159→        const collect = (m: FolderNode) => { scope.add(m.folder.id); m.children.forEach(collect); };
   160→        collect(n);
   161→        return true;
   162→      }
   163→      return n.children.some(walk);
   164→    };
   165→    tree.forEach((n) => walk(n));

   723→// dış görsel önizleme — yüklenemezse ikon yedeğine düşer
   724→function ExternalImage({ src, alt }: { src: string; alt: string }) {
   725→  const [failed, setFailed] = useState(false);
   726→  if (failed) {
   727→    return (
   728→      <div className="flex flex-col items-center gap-1.5 text-muted-foreground">
   729→        <Icons.ImageOff className="size-8" />
   730→        <span className="text-xs">Görsel önizlenemedi — bağlantıyı kontrol edin</span>
   731→      </div>
   732→    );
   733→  }
   734→  return (
   735→    <img src={src} alt={alt} onError={() => setFailed(true)} className="max-h-64 rounded object-contain" />
   736→  );
   737→}
