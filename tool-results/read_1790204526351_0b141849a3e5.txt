   280→function WaitlistTab({ editionId, categories, onChanged }: { editionId: string | null; categories: CategoryRow[]; onChanged: () => void }) {
   281→  const { refreshKey } = useApp();
   282→  const { toast } = useToast();
   283→  const [catFilter, setCatFilter] = useState("ALL");
   284→  const [busyId, setBusyId] = useState<string | null>(null);
   285→  const [autoBusy, setAutoBusy] = useState(false);
   286→  const [addOpen, setAddOpen] = useState(false);
   287→  const [addPersonId, setAddPersonId] = useState("");
   288→  const [addCategoryId, setAddCategoryId] = useState("GENERAL");
   289→  const [addPriority, setAddPriority] = useState("");
   290→  const [addNotes, setAddNotes] = useState("");
   291→
   292→  const loader = async (): Promise<WaitlistData | null> => {
   293→    if (!editionId) return null;
   294→    return apiGet<WaitlistData>(`/api/waitlist?editionId=${encodeURIComponent(editionId)}`);
   295→  };
   296→  const { data, error, reload, loading } = useApi<WaitlistData | null>(loader, [editionId, refreshKey]);
   297→
   298→  const { data: people } = useApi<PersonLite[]>(() => listEntity<PersonLite>("people", { limit: 300 }), [editionId]);
   299→
   300→  const entries = (data?.entries ?? []).filter((e) => catFilter === "ALL" || (catFilter === "GENERAL" ? !e.categoryId : e.categoryId === catFilter));
   301→  const summary = data?.summary;
   302→  const catName = (id?: string | null) => (data?.categories ?? []).find((c) => c.id === id)?.name;
   303→  const isFullCategory = (categoryId?: string | null) => {
   304→    if (!categoryId) return false;
   305→    const c = (data?.categories ?? []).find((x) => x.id === categoryId);
   306→    // açık teklifler de koltuk tutar — boş koltuk başına tek teklif kuralı
   307→    return Boolean(c && c.capacity != null && (c.seatsLeft ?? 0) - c.offeredCount <= 0);
   308→  };
   309→
   310→  // aktif sırada kategori içi pozisyon (WAITING + OFFERED)
   311→  const positionOf = (e: WaitlistEntryRow): number | null => {
   312→    if (!["WAITING", "OFFERED"].includes(e.status)) return null;
   313→    const sameQueue = (data?.entries ?? []).filter((x) => x.categoryId === e.categoryId && ["WAITING", "OFFERED"].includes(x.status));
   314→    return sameQueue.findIndex((x) => x.id === e.id) + 1;
   315→  };
   316→
   317→  const offerCountdown = (e: WaitlistEntryRow): { text: string; expired: boolean } | null => {
   318→    if (e.status !== "OFFERED" || !e.offerExpiresAt) return null;
   319→    const ms = new Date(e.offerExpiresAt).getTime() - Date.now();
   320→    if (ms <= 0) return { text: "Süresi doldu", expired: true };
   321→    const hours = Math.floor(ms / 3600000);
   322→    const minutes = Math.floor((ms % 3600000) / 60000);
   323→    return { text: hours >= 1 ? `${hours} sa ${minutes} dk kaldı` : `${minutes} dk kaldı`, expired: false };
   324→  };
   325→
   326→  const runAction = async (fn: () => Promise<{ toastTitle: string; toastDesc?: string }>, id?: string) => {
   327→    setBusyId(id ?? "global");
   328→    try {
   329→      const t = await fn();
   330→      toast({ title: t.toastTitle, description: t.toastDesc });
   331→      reload(); onChanged();
   332→    } catch (e) {
   333→      toast({ title: "İşlem başarısız", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   334→    } finally {
   335→      setBusyId(null);
   336→    }
   337→  };
   338→
   339→  const sendOffer = (e: WaitlistEntryRow) => runAction(async () => {
   340→    await apiSend("/api/waitlist", "POST", { action: "offer", entryId: e.id });
   341→    return { toastTitle: `Teklif gönderildi: ${e.person.firstName} ${e.person.lastName}`, toastDesc: "48 saat içinde yanıt beklenir — kabul edilirse kayıt açılır." };
   342→  }, e.id);
   343→
   344→  const respond = (e: WaitlistEntryRow, response: "ACCEPT" | "DECLINE") => runAction(async () => {
   345→    const res = await apiSend<{ chained?: { personName: string }[]; registration?: { confirmationNo: string } }>("/api/waitlist", "POST", { action: "respond", entryId: e.id, response });
   346→    if (response === "ACCEPT") {
   347→      return { toastTitle: `Teklif kabul edildi — kayıt açıldı: ${res.registration?.confirmationNo ?? ""}`, toastDesc: "Bekleme listesi girişi CONVERTED oldu, rozet READY." };
   348→    }
   349→    const chained = res.chained ?? [];
   350→    return {
   351→      toastTitle: `Teklif reddedildi: ${e.person.firstName} ${e.person.lastName}`,
   352→      toastDesc: chained.length > 0 ? `Sıradakine teklif gitti: ${chained.map((c) => c.personName).join(", ")}` : "Sırada bekleyen yok.",
   353→    };
   354→  }, e.id);
   355→

   385→      priority: addPriority ? Number(addPriority) : undefined,
   386→      notes: addNotes || undefined,
   387→    });
   388→    setAddOpen(false); setAddPersonId(""); setAddPriority(""); setAddNotes("");

   392→  const busyAll = busyId != null;
   393→
   394→  if (!editionId) return <EmptyState title="Edisyon seçilmedi" desc="Bekleme listesi edisyona bağlıdır." />;

   396→  if (error) return <ErrorState message={error} onRetry={reload} />;
   397→
   398→  const activeEntries = entries.filter((e) => ["WAITING", "OFFERED"].includes(e.status));
   399→  const closedEntries = entries.filter((e) => !["WAITING", "OFFERED"].includes(e.status));

   401→
   402→  return (

   404→      {/* KPI satırı */}
   405→      <div className="mb-4 grid grid-cols-2 gap-3 md:grid-cols-3 xl:grid-cols-5">

   409→        <KpiCard label="Ret / Süre Aşımı" value={summary?.declinedExpired ?? 0} sub="sıradakine geçildi" icon={<Icons.MailX className="size-4" />} tone="rose" />
   410→        <KpiCard label="Tam Dolu Kategori" value={summary?.fullCategories ?? 0} sub="koltuk bekleniyor" icon={<Icons.Lock className="size-4" />} tone="violet" />
   411→      </div>
   412→
   413→      {/* Kategori doluluk + otomatik teklif */}
   414→      <SectionCard
   415→        title="Kategori Doluluğu & Otomatik Teklif"
   416→        desc="Koltuk boşaldığında sıradaki bekleyene 48 saatlik teklif otomatik gider (§12)"

   420→            Tümünü Tara
   421→          </Button>
   422→        }
   423→        className="mb-4"
   424→      >
   425→        {categoriesWithCapacity.length === 0 ? (
   426→          <EmptyState title="Kontenjanlı kategori yok" desc="Kategoriye capacity tanımlayın; doluluk ve bekleme burada izlenir." />
   427→        ) : (
   428→          <div className="grid gap-2.5">
   429→            {categoriesWithCapacity.map((c) => {
   430→              const full = c.capacity != null && (c.seatsLeft ?? 0) <= 0;
   431→              const pct = c.fillPercent ?? 0;
   432→              return (
   433→                <div key={c.id} className="flex flex-wrap items-center gap-x-4 gap-y-2 rounded-lg border p-3 transition-colors hover:bg-muted/30">
   434→                  <div className="min-w-36 flex-1">
   435→                    <div className="flex items-center gap-2">
   436→                      <p className="text-sm font-medium">{c.name}</p>
   437→                      <Chip tone="neutral">{c.code}</Chip>
   438→                      {full && <Chip tone="rose">Dolu</Chip>}
   439→                      {c.waitingCount > 0 && <Chip tone="amber">{c.waitingCount} bekleyen</Chip>}
   440→                      {c.offeredCount > 0 && <Chip tone="teal">{c.offeredCount} teklifte</Chip>}
   441→                    </div>
   442→                    <div className="mt-1.5 flex items-center gap-2">
   443→                      <div className="h-1.5 w-full max-w-52 overflow-hidden rounded bg-muted">
   444→                        <div className={cn("h-full rounded transition-all", full ? "bg-rose-400" : pct >= 80 ? "bg-amber-400" : "bg-teal-500")} style={{ width: `${c.capacity == null ? 100 : pct}%` }} />

   453→                    variant={full ? "default" : "outline"}

   458→                  >
   459→                    {busyId === c.id ? <Icons.Loader2 className="size-4 animate-spin" /> : <Icons.Wand2 className="size-4" />}

   462→                </div>

   464→            })}
   465→          </div>
   466→        )}
   467→      </SectionCard>
   468→
   469→      {/* Sıra listesi */}
   470→      <SectionCard

   482→            </Select>
   483→            <Button size="sm" variant="outline" className="h-8 border-teal-200 text-teal-700 hover:bg-teal-50" onClick={() => setAddOpen(true)}>

   488→      >
   489→        {entries.length === 0 ? (
   490→          <EmptyState title="Bekleme listesi boş" desc="Kategori dolduğunda kayıt formu otomatik bekleme önerir; buradan da elle ekleyebilirsiniz." />

   492→          <>
   493→            {/* aktif sıra */}
   494→            <div className="overflow-hidden rounded-lg border">
   495→              <div className="overflow-x-auto maven-scroll">
   496→                <table className="w-full min-w-[820px] text-sm">
   497→                  <thead>
   498→                    <tr className="border-b bg-muted/50 text-left text-xs text-muted-foreground">

   501→                      <th className="px-3 py-2.5 font-medium">Kategori</th>

   503→                      <th className="px-3 py-2.5 font-medium">Durum</th>
   504→                      <th className="px-3 py-2.5 font-medium">Teklif</th>
   505→                      <th className="px-3 py-2.5 font-medium text-right">Eylem</th>
   506→                    </tr>
   507→                  </thead>
   508→                  <tbody>
   509→                    {activeEntries.map((e) => {
   510→                      const pos = positionOf(e);
   511→                      const cd = offerCountdown(e);
   512→                      const rowBusy = busyId === e.id;
   513→                      return (
   514→                        <tr key={e.id} className="border-b transition hover:bg-muted/40 last:border-0">
   515→                          <td className="px-3 py-2.5">
   516→                            <span className="grid size-6 place-items-center rounded-full bg-muted text-xs font-semibold tabular-nums">{pos ?? "—"}</span>
   517→                          </td>
   518→                          <td className="px-3 py-2.5">
   519→                            <p className="font-medium leading-tight">{e.person.firstName} {e.person.lastName}</p>
   520→                            <p className="text-xs text-muted-foreground">{e.person.company ?? e.person.email ?? "—"}</p>
   521→                          </td>
   522→                          <td className="px-3 py-2.5 text-xs">{e.category ? catName(e.category.id) ?? e.category.name : <span className="text-muted-foreground">Genel</span>}</td>
   523→                          <td className="px-3 py-2.5 font-mono text-xs tabular-nums">#{e.priority}</td>
   524→                          <td className="px-3 py-2.5"><StatusBadge map={WAITLIST_STATUS} value={e.status} /></td>
   525→                          <td className="px-3 py-2.5 text-xs">
   526→                            {e.status === "OFFERED" && cd ? (
   527→                              <span className={cn("inline-flex items-center gap-1.5", cd.expired ? "font-medium text-rose-600" : "text-muted-foreground")}>
   528→                                <Icons.Clock3 className={cn("size-3.5", cd.expired && "text-rose-500")} />
   529→                                {cd.expired ? "Süresi doldu" : `${fmtDateTime(e.offeredAt)} — ${cd.text}`}
   530→                              </span>
   531→                            ) : "—"}

   533→                          <td className="px-3 py-2.5">
   534→                            <div className="flex justify-end gap-1">
   535→                              {e.status === "WAITING" && (
   536→                                isFullCategory(e.categoryId) ? (
   537→                                  <Button size="sm" variant="ghost" className="h-7 text-muted-foreground/60" disabled title="Koltuk başına tek teklif — açık teklif yanıtlanınca sıradakine geçer">
   538→                                    <Icons.Lock className="size-3.5" /> Koltuk Bekleniyor
   539→                                  </Button>
   540→                                ) : (
   541→                                  <Button size="sm" variant="outline" className="h-7 border-sky-200 text-sky-700 hover:bg-sky-50" disabled={busyAll} onClick={() => sendOffer(e)}>
   542→                                    {rowBusy ? <Icons.Loader2 className="size-3.5 animate-spin" /> : <Icons.Send className="size-3.5" />} Teklif Gönder
   543→                                  </Button>
   544→                                )
   545→                              )}
   546→                              {e.status === "OFFERED" && (
   547→                                <>
   548→                                  <Button size="sm" variant="outline" className="h-7 border-emerald-200 text-emerald-700 hover:bg-emerald-50" disabled={busyAll} onClick={() => respond(e, "ACCEPT")} title="Davetli teklifi kabul etti (portal yanıtı simülasyonu)">
   549→                                    <Icons.Check className="size-3.5" /> Kabul

   557→                                onClick={() => {
   558→                                  try { sessionStorage.setItem("maven.portal.person", e.personId); } catch { /* yoksay */ }
   559→                                  useApp.getState().setModule("portals");
   560→                                }}>
   561→                                <Icons.ExternalLink className="size-3.5" />
   562→                              </Button>
   563→                              <Button size="sm" variant="ghost" className="h-7 text-muted-foreground" disabled={busyAll} onClick={() => removeEntry(e)} title="Listeden çıkar">
   564→                                <Icons.ListX className="size-3.5" />
   565→                              </Button>
   566→                            </div>
   567→                          </td>
   568→                        </tr>
   569→                      );
   570→                    })}
   571→                  </tbody>
   572→                </table>
   573→              </div>
   574→            </div>
   575→
   576→            {/* geçmiş (terminal durumlar) */}
   577→            {closedEntries.length > 0 && (

   581→                  {closedEntries.map((e) => (
   582→                    <div key={e.id} className="flex flex-wrap items-center gap-x-3 gap-y-1 rounded-lg border bg-muted/20 px-3 py-2 text-xs">
   583→                      <span className="font-medium">{e.person.firstName} {e.person.lastName}</span>
   584→                      <span className="text-muted-foreground">{e.category ? catName(e.category.id) ?? e.category.name : "Genel"}</span>
   585→                      {e.status === "CONVERTED" && e.convertedRegistration && (
   586→                        <span className="font-mono text-[11px] text-emerald-700">{e.convertedRegistration.confirmationNo}</span>
   587→                      )}
   588→                      <span className="ml-auto text-muted-foreground">{e.respondedAt ? `yanıt ${fmtDate(e.respondedAt)}` : `eklenme ${fmtDate(e.createdAt)}`}</span>
   589→                      <StatusBadge map={WAITLIST_STATUS} value={e.status} />
   590→                    </div>
   591→                  ))}
   592→                </div>
   593→              </div>
   594→            )}
   595→          </>

   597→      </SectionCard>

   599→      {/* Listeye ekle dialogu */}
   600→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   601→        <DialogContent className="sm:max-w-md">
   602→          <DialogHeader>
   603→            <DialogTitle>Bekleme listesine ekle</DialogTitle>
   604→            <DialogDescription>Kişi koltuk boşaldığında öncelik sırasına göre otomatik teklif alır.</DialogDescription>
   605→          </DialogHeader>
   606→          <div className="space-y-3">
   607→            <div>
   608→              <Label>Kişi *</Label>
   609→              <Select value={addPersonId} onValueChange={setAddPersonId}>
   610→                <SelectTrigger className="mt-1"><SelectValue placeholder="Kişi seçin…" /></SelectTrigger>
   611→                <SelectContent>
   612→                  {(people ?? []).map((p) => (
   613→                    <SelectItem key={p.id} value={p.id}>{p.firstName} {p.lastName}{p.company ? ` — ${p.company}` : ""}</SelectItem>
   614→                  ))}
   615→                </SelectContent>
   616→              </Select>
   617→            </div>
   618→            <div>
   619→              <Label>Kategori</Label>
   620→              <Select value={addCategoryId} onValueChange={setAddCategoryId}>
   621→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   622→                <SelectContent>
   623→                  <SelectItem value="GENERAL">Genel (kategorisiz sıra)</SelectItem>
   624→                  {categories.map((c) => <SelectItem key={c.id} value={c.id}>{c.name} ({c.code})</SelectItem>)}
   625→                </SelectContent>
   626→              </Select>
   627→            </div>
   628→            <div className="grid grid-cols-2 gap-3">
   629→              <div>
   630→                <Label>Öncelik</Label>

   633→              <div className="flex items-end">
