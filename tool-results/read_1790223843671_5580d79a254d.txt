    99→const todayStr = (): string => {
   100→  const d = new Date();
   101→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
   102→};
   103→
   104→// benzersiz gider kodu: GSN-YYYY-XXX (sayaç = mevcut adet + 1; çakışırsa +1 dene)
   105→const nextExpenseCode = (existing: ExpenseRow[]): string => {
   106→  const year = new Date().getFullYear();
   107→  const taken = new Set(existing.map((e) => e.code));
   108→  let n = existing.length + 1;
   109→  let code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   110→  while (taken.has(code)) {
   111→    n += 1;
   112→    code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   113→  }
   114→  return code;
   115→};

   117→// ay etiketi: "2026-09" → "Eyl 2026"
   118→const monthLabel = (m: string): string => {
   119→  const d = new Date(`${m}-01`);
   120→  return Number.isNaN(d.getTime()) ? m : d.toLocaleDateString("tr-TR", { month: "short", year: "numeric" });
   121→};
