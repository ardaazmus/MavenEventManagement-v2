    99→    99→const todayStr = (): string => {

   104→   104→// benzersiz gider kodu: GSN-YYYY-XXX (sayaç = mevcut adet + 1; çakışırsa +1 dene)
   105→   105→const nextExpenseCode = (existing: ExpenseRow[]): string => {
   106→   106→  const year = new Date().getFullYear();
   107→   107→  const taken = new Set(existing.map((e) => e.code));
   108→   108→  let n = existing.length + 1;
   109→   109→  let code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   110→   110→  while (taken.has(code)) {
   111→   111→    n += 1;
   112→   112→    code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   113→   113→  }
   114→   114→  return code;
   115→   115→};
