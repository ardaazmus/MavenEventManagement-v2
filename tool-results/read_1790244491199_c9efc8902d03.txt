   597→   597→const FORBIDDEN = new Set(["id", "createdAt"]);
   598→   598→
   599→   599→export function sanitize(data: Record<string, unknown>): Record<string, unknown> {
   600→   600→  const out: Record<string, unknown> = {};
   601→   601→  for (const [k, v] of Object.entries(data)) {
   602→   602→    if (FORBIDDEN.has(k)) continue;
   603→   603→    if (v === undefined) continue;
   604→   604→    out[k] = v === "" ? null : v;
   605→   605→  }
   606→   606→  return out;
   607→   607→}

   611→   611→// tek kiracılı kurulumda ilk tenant ile doldurulur — aksi halde Prisma
   612→   612→// "Argument `tenant` is missing" hatası yeni etkinlik/seri/kişi/kurum
   613→   613→// oluşturmayı imkânsız kılar (bug: Yeni Etkinlik Oluşturulamadı).
   614→   614→const TENANT_SCOPED = new Set(["event-series", "editions", "people", "organizations", "mail-providers"]);
   615→   615→
   616→   616→export async function withTenant(entity: string, data: Record<string, unknown>): Promise<Record<string, unknown>> {
   617→   617→  if (!TENANT_SCOPED.has(entity)) return data;
   618→   618→  const v = data.tenantId;
   619→   619→  if (typeof v === "string" && v.trim() !== "") return data;
   620→   620→  const tenant = await db.tenant.findFirst({ select: { id: true } });
   621→   621→  if (!tenant) throw new Error("Kiracı (tenant) bulunamadı — önce demo verisini yükleyin");
   622→   622→  return { ...data, tenantId: tenant.id };
   623→   623→}
