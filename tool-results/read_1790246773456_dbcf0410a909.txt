    70→// ISO → datetime-local girdi değeri (tarayıcı yerel saati)
    71→const toLocalInput = (iso: string) => {
    72→  const d = new Date(iso);
    73→  if (Number.isNaN(d.getTime())) return "";
    74→  const pad = (n: number) => String(n).padStart(2, "0");
    75→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    76→};
    77→// dosya → dataURL (boyut sınırı aşılırsa null)
    78→const fileToDataUrl = (file: File, maxBytes: number): Promise<string | null> =>
    79→  new Promise((resolve) => {
    80→    if (file.size > maxBytes) { resolve(null); return; }
    81→    const reader = new FileReader();
    82→    reader.onload = () => resolve(String(reader.result));
    83→    reader.onerror = () => resolve(null);
    84→    reader.readAsDataURL(file);
    85→  });

   121→export function ScientificView() {

   135→  const [subForm, setSubForm] = useState(emptySub);
   136→
   137→  // TASK-A F6: bildiriler imleçli load-more — 300 satırlık sessiz kesme kaldırıldı
   138→  const { data: subsPaged, error, reload, loading, more: subMore } = useApi<{ items: SubmissionRow[]; nextCursor?: string | null }>(
   139→    (cursor?: string) => listEntityPaged<SubmissionRow>("submissions", { editionId: currentEditionId ?? undefined, limit: 200 }, cursor),
   140→    [currentEditionId, refreshKey],
   141→    { append: true },
   142→  );
   143→  const subs = useMemo(() => subsPaged?.items ?? [], [subsPaged]);
   144→  const { data: tracks } = useApi<{ id: string; name: string }[]>(() => listEntity("tracks", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey]);
   145→
   146→  const byStatus = (s: string) => subs.filter((x) => x.status === s).length;
   147→  const overdue = subs.flatMap((s) => s.reviewAssignments).filter((r) => r.status === "OVERDUE").length;
   148→
   149→  const submitDecision = async () => {
   150→    if (!decideTarget) return;
   151→    setBusy(true);
   152→    try {
   153→      await apiSend("/api/decisions", "POST", { submissionId: decideTarget.id, decision, rationale, decidedBy: "Bilimsel Komite" });
   154→      const statusMap: Record<string, string> = { ACCEPT_ORAL: "ACCEPTED", ACCEPT_POSTER: "ACCEPTED", ACCEPT_E_POSTER: "ACCEPTED", ACCEPT_PANEL: "ACCEPTED", REJECT: "REJECTED", REVISION_REQUIRED: "REVISION_REQUIRED", WAITLIST: "WAITLIST", WITHDRAWN: "WITHDRAWN" };
   155→      await apiSend(`/api/submissions/${decideTarget.id}`, "PUT", { status: statusMap[decision] ?? decideTarget.status });
   156→      toast({ title: "Karar kaydedildi", description: `${decideTarget.code} → ${decision}. Sunum havuzu adayı oluştu; otomatik program slotu OLUŞTURULMAZ.` });
   157→      setDecideTarget(null); setRationale(""); reload(); bump();
   158→    } catch (e) {
   159→      toast({ title: "Karar kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   160→    } finally { setBusy(false); }
   161→  };
   162→
   163→  // ── R10-b: bildiri ekle/düzenle + hızlı durum değişimi ──
   164→  const openSubNew = () => { setEditingSub(null); setSubForm(emptySub); setSubOpen(true); };
   165→  const openSubEdit = (s: SubmissionRow) => {
   166→    setEditingSub(s);
   167→    setSubForm({
   168→      title: s.title, abstract: s.abstract ?? "", type: s.type, keywords: s.keywords ?? "",
   169→      presentingAuthorName: s.presentingAuthorName ?? "", status: s.status,
   170→      trackId: s.trackId ?? "none", fileUrl: s.fileUrl ?? "", posterNo: s.posterNo ?? "",
   171→      fileStatus: s.fileStatus ?? "MISSING", submittedAt: s.submittedAt ? toLocalInput(s.submittedAt) : "",
   172→    });
   173→    setSubOpen(true);
   174→  };
   175→  const changeStatus = async (s: SubmissionRow, status: string) => {
   176→    if (status === s.status || statusBusyId) return;
   177→    setStatusBusyId(s.id);
   178→    try {
   179→      await apiSend(`/api/submissions/${s.id}`, "PUT", { status });
   180→      toast({ title: "Bildiri durumu güncellendi", description: `${s.code} → ${label(SUBMISSION_STATUS, status)}` });
   181→      reload(); bump();
   182→    } catch (e) {
   183→      toast({ title: "Durum güncellenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   184→    } finally { setStatusBusyId(null); }
   185→  };

   195→        trackId: subForm.trackId === "none" ? null : subForm.trackId,
   196→        fileUrl: subForm.fileUrl || null, posterNo: subForm.posterNo || null,
   197→        fileStatus: subForm.fileStatus || null,
   198→        submittedAt: subForm.submittedAt ? new Date(subForm.submittedAt).toISOString() : null,
   199→      };
   200→      if (editingSub) await apiSend(`/api/submissions/${editingSub.id}`, "PUT", payload);
   201→      else await apiSend("/api/submissions", "POST", { editionId: currentEditionId, ...payload, code: `SUB-${Date.now().toString().slice(-4)}${Math.floor(Math.random() * 9)}` });
   202→      toast({ title: editingSub ? "Bildiri güncellendi" : "Bildiri kaydedildi", description: subForm.title.trim() });
   203→      setSubOpen(false); reload(); bump();
   204→    } catch (e) {
   205→      toast({ title: "Bildiri kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });

   214→        </Button>
   215→      </PageHeader>
   216→
   217→      <div className="grid grid-cols-2 gap-3 md:grid-cols-5">
   218→        <KpiCard label="Gönderilen" value={subs.filter((s) => s.status !== "DRAFT").length} sub="taslak hariç" icon={<Icons.FileText className="size-4" />} />
   219→        <KpiCard label="İncelemede" value={byStatus("UNDER_REVIEW")} sub={`geciken hakem: ${overdue}`} tone="amber" icon={<Icons.Hourglass className="size-4" />} />
   220→        <KpiCard label="Kabul" value={byStatus("ACCEPTED")} sub="sunum havuzunda" tone="emerald" icon={<Icons.CircleCheck className="size-4" />} />
   221→        <KpiCard label="Revizyon" value={byStatus("REVISION_REQUIRED")} sub="yazara geri döndü" tone="amber" />
