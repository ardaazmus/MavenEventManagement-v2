   126→const MAX_IMAGE_KB = 600; // SQLite satırı tavanı — sunucu tarafıyla aynı
   127→
   128→function LinkedPhotoUploader({
   129→  editionId, systemFolder, linkedType, linkedId, assetName, currentUrl, onApply, folderLabel, alt, fit = "cover",
   130→}: {
   131→  editionId: string | null;
   132→  systemFolder: string; // KISI_FOTOGRAF | KURUM_LOGO
   133→  linkedType: "PERSON" | "ORGANIZATION";
   134→  linkedId: string;
   135→  assetName: string;
   136→  currentUrl?: string | null;
   137→  onApply: (dataUrl: string) => Promise<void>;
   138→  folderLabel: string;
   139→  alt: string;
   140→  fit?: "cover" | "contain";
   141→}) {
   142→  const { toast } = useToast();
   143→  const fileRef = useRef<HTMLInputElement>(null);
   144→  const [busy, setBusy] = useState(false);
   145→  const [preview, setPreview] = useState<string | null>(currentUrl ?? null);
   146→  useEffect(() => { setPreview(currentUrl ?? null); }, [currentUrl]);
   147→
   148→  const pick = async (ev: React.ChangeEvent<HTMLInputElement>) => {
   149→    const file = ev.target.files?.[0];
   150→    ev.target.value = ""; // aynı dosya yeniden seçilebilsin
   151→    if (!file) return;
   152→    if (!file.type.startsWith("image/")) {
   153→      toast({ title: "Desteklenmeyen dosya", description: "Lütfen bir görsel dosyası seçin (image/*).", variant: "destructive" });
   154→      return;
   155→    }
   156→    if (file.size > MAX_IMAGE_KB * 1024) {
   157→      toast({ title: "Dosya çok büyük", description: `Görsel en fazla ${MAX_IMAGE_KB} KB olabilir — daha küçük bir dosya seçin.`, variant: "destructive" });
   158→      return;
   159→    }
   160→    setBusy(true);
   161→    try {
   162→      const dataUrl = await new Promise<string>((resolve, reject) => {
   163→        const fr = new FileReader();
   164→        fr.onload = () => resolve(String(fr.result));
   165→        fr.onerror = () => reject(new Error("Dosya okunamadı"));
   166→        fr.readAsDataURL(file);
   167→      });
   168→      const res = await apiSend<UploadLinkedResult>("/api/media/upload-linked", "POST", {
   169→        editionId, systemFolder, name: assetName, dataUrl, linkedType, linkedId,
   170→      });
   171→      const finalUrl = res.asset.dataUrl ?? dataUrl;
   172→      await onApply(finalUrl);
   173→      setPreview(finalUrl);
   174→      toast({ title: "Görsel yüklendi", description: `Medya Arşivi → ${folderLabel}: ${res.asset.name}` });
   175→    } catch (e) {
   176→      toast({ title: "Yükleme başarısız", description: e instanceof Error ? e.message : "Görsel yüklenemedi", variant: "destructive" });
   177→    } finally {
   178→      setBusy(false);
   179→    }
   180→  };
   181→
   182→  return (
   183→    <div className="flex items-start gap-3">
   184→      {preview ? (
   185→        <img src={preview} alt={alt} className={cn("size-14 shrink-0 rounded-lg border bg-background", fit === "contain" ? "object-contain p-0.5" : "object-cover")} />
   186→      ) : (
   187→        <span className="grid size-14 shrink-0 place-items-center rounded-lg border border-dashed bg-muted/30 text-muted-foreground" aria-hidden>
   188→          <Icons.ImageUp className="size-5" />
   189→        </span>
   190→      )}
   191→      <div className="min-w-0 flex-1">
   192→        <div className="flex flex-wrap items-center gap-2">
   193→          <Button type="button" size="sm" variant="outline" disabled={busy || !editionId} onClick={() => fileRef.current?.click()}>
   194→            {busy ? <Icons.Loader2 className="size-3.5 animate-spin" aria-hidden /> : <Icons.ImageUp className="size-3.5" />}
   195→            {preview ? "Değiştir" : "Görsel Seç"}
   196→          </Button>
   197→          <span className="text-[11px] text-muted-foreground">{preview ? "görsel bağlı" : "görsel yok"}</span>
   198→        </div>
   199→        <p className="mt-1 text-[11px] leading-snug text-muted-foreground">Medya Arşivi → {folderLabel} klasörüne benzersiz adla kaydedilir (≤ {MAX_IMAGE_KB} KB).</p>
   200→      </div>
   201→      <input ref={fileRef} type="file" accept="image/*" className="sr-only" onChange={pick} disabled={busy} aria-label="Görsel dosyası seç" />
   202→    </div>
   203→  );
   204→}

   228→function CvPanel({ personId, editionId }: { personId: string; editionId: string | null }) {
   229→  const { toast } = useToast();
   230→  const { data: cvs, error, reload, loading } = useApi<CvEntryRow[]>(
   231→    () => listEntity<CvEntryRow>("cv-entries", { personId, limit: 100 }),
   232→    [personId],
   233→  );
   234→  const [open, setOpen] = useState(false);
   235→  const [editing, setEditing] = useState<CvEntryRow | null>(null);
   236→  const [busy, setBusy] = useState(false);
   237→  const empty = { kind: "EXPERIENCE", title: "", organization: "", city: "", startDate: "", endDate: "", isCurrent: false, description: "", order: "0" };
   238→  const [form, setForm] = useState(empty);
   239→
   240→  const openNew = () => { setEditing(null); setForm(empty); setOpen(true); };
   241→  const openEdit = (c: CvEntryRow) => {
   242→    setEditing(c);
   243→    setForm({
   244→      kind: c.kind, title: c.title, organization: c.organization ?? "", city: c.city ?? "",
   245→      startDate: c.startDate ? c.startDate.slice(0, 10) : "", endDate: c.endDate ? c.endDate.slice(0, 10) : "",
   246→      isCurrent: c.isCurrent, description: c.description ?? "", order: String(c.order ?? 0),
   247→    });
   248→    setOpen(true);
   249→  };
   250→  const save = async () => {
   251→    if (!form.title.trim()) return;
   252→    setBusy(true);
   253→    try {
   254→      const payload = {
   255→        personId, editionId,
   256→        kind: form.kind, title: form.title.trim(), organization: form.organization || null, city: form.city || null,
   257→        startDate: form.startDate || null, endDate: form.isCurrent ? null : form.endDate || null,
   258→        isCurrent: form.isCurrent, description: form.description || null, order: Number(form.order) || 0,
   259→      };
   260→      if (editing) await apiSend(`/api/cv-entries/${editing.id}`, "PUT", payload);
   261→      else await apiSend("/api/cv-entries", "POST", payload);
   262→      toast({ title: editing ? "CV kaydı güncellendi" : "CV kaydı eklendi", description: form.title });
   263→      setOpen(false); reload();
   264→    } catch (e) {
   265→      toast({ title: "Hata", description: e instanceof Error ? e.message : "CV kaydı kaydedilemedi", variant: "destructive" });
   266→    } finally { setBusy(false); }
   267→  };
   268→  const remove = async (c: CvEntryRow) => {
   269→    try {
   270→      await apiSend(`/api/cv-entries/${c.id}`, "DELETE");
   271→      toast({ title: "CV kaydı silindi", description: c.title });
   272→      reload();
   273→    } catch (e) {
   274→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Silinemedi", variant: "destructive" });
   275→    }
   276→  };
   277→
   278→  const groups = CV_KIND_ORDER.map((k) => ({ kind: k, items: (cvs ?? []).filter((c) => c.kind === k).sort((a, b) => a.order - b.order) })).filter((g) => g.items.length > 0);
   279→  const canAdd = Boolean(editionId);
   280→
   281→  return (
   282→    <SectionCard
   283→      title="CV & Deneyimler"
   284→      desc="Eğitim, deneyim, ödül, dil, yayın ve sertifikalar — kişiye bağlı zaman çizelgesi"
   285→      action={
   286→        <Button size="sm" variant="outline" onClick={openNew} disabled={!canAdd} aria-label="CV kaydı ekle">
   287→          <Icons.Plus className="size-3.5" /> Ekle
   288→        </Button>
   289→      }
   290→    >
   291→      {!canAdd && <p className="mb-2 text-[11px] text-amber-600">CV kaydı için bir edisyon seçili olmalı (etkinlik izolasyonu).</p>}
   292→      {loading ? <Loading rows={2} /> : error ? <ErrorState message={error} onRetry={reload} /> : groups.length === 0 ? (
   293→        <p className="py-2 text-xs text-muted-foreground">Henüz CV kaydı yok — &quot;Ekle&quot; ile ilk kaydı girin.</p>
   294→      ) : (
   295→        <div className="maven-scroll max-h-96 space-y-4 overflow-y-auto pr-1">
   296→          {groups.map((g, gi) => (
   297→            <div key={g.kind} className="maven-stagger-item" style={{ animationDelay: `${gi * 60}ms` }}>
   298→              <p className="mb-2 flex items-center gap-1.5 text-xs font-semibold">
   299→                <Chip tone="teal">{label(CV_KIND, g.kind)}</Chip>
   300→                <span className="font-normal tabular-nums text-muted-foreground">{g.items.length} kayıt</span>
   301→              </p>
   302→              {/* zaman çizelgesi: sol hat + nokta düğümleri */}
   303→              <ol className="relative ml-1.5 space-y-3 border-l-2 border-teal-500/25 pl-4">
   304→                {g.items.map((c) => (
   305→                  <li key={c.id} className="group relative">
   306→                    <span className="absolute -left-[21.5px] top-1.5 size-2.5 rounded-full border-2 border-teal-500/60 bg-background" aria-hidden />
   307→                    <div className="rounded-lg border bg-muted/20 p-2.5 transition-colors hover:border-teal-500/30 hover:bg-muted/40">
   308→                      <div className="flex flex-wrap items-center justify-between gap-x-2 gap-y-1">
   309→                        <p className="min-w-0 text-sm font-medium leading-tight">{c.title}</p>

   501→function RolesPanel() {
   502→  const { currentEditionId, bump, refreshKey } = useApp();
   503→  const { toast } = useToast();
   504→
   505→  // özel rol tanımları — hiyerarşi seviyesine göre sıralı (registry orderBy)

   512→    () => listEntity<ParticipationLite>("participations", { editionId: currentEditionId ?? undefined, limit: 50 }),
   513→    [currentEditionId, refreshKey],
   514→  );
   515→
   516→  const sorted = (roles ?? []).slice().sort((a, b) => a.hierarchyLevel - b.hierarchyLevel);
   517→
   518→  // ── rol oluştur/düzenle diyaloğu ──
   519→  const [dialogOpen, setDialogOpen] = useState(false);
   520→  const [editing, setEditing] = useState<CustomRoleRow | null>(null);
   521→  const [busy, setBusy] = useState(false);
   522→  const emptyRole = { name: "", key: "", keyTouched: false, hierarchyLevel: 50, color: "teal", permissions: [] as string[], description: "" };
   523→  const [form, setForm] = useState(emptyRole);
   524→  const keyHint = form.keyTouched ? form.key : slugifyKey(form.name);
   525→  const keyClash = Boolean(keyHint && sorted.some((r) => r.key === keyHint && r.id !== editing?.id));
   526→
   527→  const openNew = () => { setEditing(null); setForm(emptyRole); setDialogOpen(true); };
   528→  const openEdit = (r: CustomRoleRow) => {
   529→    setEditing(r);
   530→    setForm({ name: r.name, key: r.key, keyTouched: true, hierarchyLevel: r.hierarchyLevel, color: r.color ?? "teal", permissions: parsePerms(r.permissions), description: r.description ?? "" });
   531→    setDialogOpen(true);
   532→  };
   533→  const saveRole = async () => {
   534→    if (!form.name.trim() || !keyHint || keyClash || !currentEditionId) return;
   535→    setBusy(true);
   536→    try {
   537→      const payload = {
   538→        editionId: currentEditionId, name: form.name.trim(), key: keyHint, color: form.color,
   539→        hierarchyLevel: form.hierarchyLevel, permissions: JSON.stringify(form.permissions), description: form.description || null,
   540→      };
   541→      if (editing) await apiSend(`/api/custom-roles/${editing.id}`, "PUT", payload);
   542→      else await apiSend("/api/custom-roles", "POST", payload);
   543→      toast({ title: editing ? "Rol güncellendi" : "Rol oluşturuldu", description: `${form.name} · seviye ${form.hierarchyLevel}` });
   544→      setDialogOpen(false); reload(); bump();
   545→    } catch (e) {
   546→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Rol kaydedilemedi", variant: "destructive" });
   547→    } finally { setBusy(false); }
   548→  };
   549→  const toggleActive = async (r: CustomRoleRow) => {
   550→    try {
   551→      await apiSend(`/api/custom-roles/${r.id}`, "PUT", { isActive: !r.isActive });
   552→      toast({ title: r.isActive ? "Rol pasifleştirildi" : "Rol aktifleştirildi", description: r.name });
   553→      reload();
   554→    } catch (e) {
   555→      toast({ title: "Hata", description: e instanceof Error ? e.message : "Durum değiştirilemedi", variant: "destructive" });
   556→    }
   557→  };
   558→  const deleteRole = async (r: CustomRoleRow) => {
   559→    try {
   560→      await apiSend(`/api/custom-roles/${r.id}`, "DELETE");
   561→      toast({ title: "Rol silindi", description: r.name });
   562→      reload(); bump();
   563→    } catch (e) {
   564→      toast({ title: "Silinemedi", description: e instanceof Error ? e.message : "Rola bağlı atamalar olabilir.", variant: "destructive" });

   584→      setAssignPart(""); setAssignRole(""); bump();
   585→    } catch (e) {
