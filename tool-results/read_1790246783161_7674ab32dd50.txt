    35→export function OnsiteView() {
    36→  const { currentEditionId, bump, refreshKey } = useApp();
    37→  const { toast } = useToast();
    38→  const [door, setDoor] = useState("Kapı A");
    39→  const [code, setCode] = useState("");
    40→  const [scanning, setScanning] = useState(false);
    41→  const [last, setLast] = useState<ScanResult | null>(null);
    42→  const [denyTarget, setDenyTarget] = useState<string | null>(null);
    43→  const [forceReason, setForceReason] = useState("");
    44→
    45→  const { data: scans, error, reload, loading } = useApi<ScanRow[]>(() => listEntity<ScanRow>("scan-events", { limit: 60 }), [currentEditionId, refreshKey]);
    46→
    47→  const todayStart = new Date(); todayStart.setHours(0, 0, 0, 0);
    48→  const todays = (scans ?? []).filter((s) => new Date(s.scannedAt) >= todayStart);
    49→  const uniqueArrived = new Set(todays.filter((s) => s.action === "ENTRY" && s.result === "ALLOWED").map((s) => s.participation?.id)).size;
    50→  const rescans = todays.filter((s) => s.result === "RESCAN_WARNING");
    51→  const denied = todays.filter((s) => s.result === "DENIED");
    52→
    53→  const scan = async (force?: string) => {
    54→    if (!code.trim()) return;
    55→    setScanning(true);
    56→    try {
    57→      const res = await apiSend<ScanResult>("/api/scan", "POST", { code: code.trim(), door, forceReason: force });
    58→      setLast(res);
    59→      if (res.result === "ALLOWED") setCode("");
    60→      if (force) setDenyTarget(null); setForceReason("");
    61→      toast({
    62→        title: res.result === "ALLOWED" ? "Giriş izin verildi" : res.result === "RESCAN_WARNING" ? "Tekrar tarama — sarı durum" : "Giriş reddedildi",
    63→        variant: res.result === "DENIED" ? "destructive" : "default",
    64→        description: res.person ? `${res.person.name}${res.reason ? ` — ${res.reason}` : ""}` : res.reason,
    65→      });
    66→      reload(); bump();
    67→    } catch (e) {
    68→      const msg = e instanceof Error ? e.message : "Tarama hatası";
    69→      if (msg.includes("bulunamadı")) {
    70→        setLast({ result: "DENIED", tone: "red", reason: msg });
    71→      }
    72→      toast({ title: "Tarama", description: msg, variant: "destructive" });
    73→    } finally {
    74→      setScanning(false);
    75→    }
    76→  };
    77→
    78→  return (
    79→    <div className="space-y-5">
    80→      <PageHeader title="Sahada — Canlı Kontrol" desc="Kapı masası: kimlik doğrulama, tekrar tarama ayrı sarı durum; tarama geçmişi silinmez" />
    81→
    82→      <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
    83→        <KpiCard label="Benzersiz Gelen (bugün)" value={uniqueArrived} sub="geçerli ilk giriş — oturum girişi sayılmaz" tone="emerald" icon={<Icons.UserCheck className="size-4" />} />
    84→        <KpiCard label="Tekrar Tarama" value={rescans.length} sub="ilk girişin yerine geçmez" tone="amber" icon={<Icons.RotateCcw className="size-4" />} />
    85→        <KpiCard label="Reddedilen" value={denied.length} sub="yetkiliye yönlendirme" tone="rose" icon={<Icons.UserX className="size-4" />} />
    86→        <KpiCard label="Toplam Olay" value={todays.length} sub="son 24 saat" icon={<Icons.Activity className="size-4" />} />
    87→      </div>
    88→
    89→      <div className="grid gap-4 lg:grid-cols-5">
    90→        <SectionCard title="Tarama Masası" desc={`${door} · görevli kapsamı: etkinlik girişi + oturum`} className="min-w-0 lg:col-span-2">
    91→          <div className="flex gap-2">
    92→            {["Kapı A", "Kapı B", "Gala", "VIP Lounge"].map((d) => (
    93→              <button key={d} onClick={() => setDoor(d)} className={cn("rounded-lg border px-3 py-1.5 text-xs font-medium transition", door === d ? "border-primary bg-primary/10 text-primary" : "text-muted-foreground hover:border-primary/40")}>
    94→                {d}
    95→              </button>
    96→            ))}
    97→          </div>
    98→          <div className="mt-3 flex gap-2">
    99→            <Input placeholder="QR kodu (örn. QR-0001) veya part_id…" value={code} onChange={(e) => setCode(e.target.value)} onKeyDown={(e) => e.key === "Enter" && scan()} className="font-mono" />
   100→            <Button onClick={() => scan()} disabled={scanning || !code.trim()}>
   101→              <Icons.ScanLine className="size-4" /> {scanning ? "Okutuluyor…" : "Tara"}
   102→            </Button>
   103→          </div>
   104→
   105→          {/* sonuç kartı */}
   106→          {last && (
   107→            <div className={cn("mt-3 rounded-lg border-2 p-3",
   108→              last.result === "ALLOWED" ? "border-emerald-300 bg-emerald-50/70" : last.result === "RESCAN_WARNING" ? "border-amber-300 bg-amber-50/70" : "border-rose-300 bg-rose-50/70")}>
   109→              <div className="flex items-center gap-2">
   110→                {last.result === "ALLOWED" ? <Icons.CircleCheck className="size-5 text-emerald-600" /> : last.result === "RESCAN_WARNING" ? <Icons.TriangleAlert className="size-5 text-amber-600" /> : <Icons.Ban className="size-5 text-rose-600" />}
   111→                <p className="text-sm font-bold">{last.result === "ALLOWED" ? "Girişe izin ver" : last.result === "RESCAN_WARNING" ? "Tekrar tarama" : "Yetkiliye yönlendir"}</p>
   112→              </div>
   113→              {last.person && (
   114→                <div className="mt-2 space-y-0.5 text-xs">
   115→                  <p className="font-semibold">{last.person.name} <span className="font-normal text-muted-foreground">{last.person.title ? `· ${last.person.title}` : ""}</span></p>
   116→                  <p className="text-muted-foreground">{last.person.company ?? "—"}</p>
   117→                </div>
   118→              )}
   119→              {last.registration && (
   120→                <div className="mt-2 flex flex-wrap gap-1 text-[11px]">
   121→                  <Chip tone="teal">{last.registration.category ?? "kategori yok"}</Chip>
   122→                  <Chip tone={last.registration.status === "CONFIRMED" ? "emerald" : "amber"}>kayıt: {last.registration.status}</Chip>
   123→                  {last.badge && <Chip tone="violet">yaka kartı: {last.badge.profile} {last.badge.status}</Chip>}
   124→                </div>
   125→              )}
   126→              {last.reason && <p className="mt-2 text-xs text-muted-foreground">{last.reason}</p>}
   127→              {last.result === "DENIED" && (
   128→                <Button size="sm" variant="outline" className="mt-2" onClick={() => setDenyTarget(last.person?.id ?? "")}>
   129→                  Manuel İstisna (gerekçe ile)
   130→                </Button>
   131→              )}
   132→            </div>
   133→          )}
   134→          <p className="mt-3 text-[11px] text-muted-foreground">Demo: QR-0001 … QR-0024 aktif yaka kartları; iptal/reddedilen katılımcı kodu yoktur → kırmızı durum.</p>
   135→        </SectionCard>
   136→
   137→        <SectionCard title="Canlı Tarama Akışı" desc="olay bazlı — ilk geçerli giriş ve tekrar tarama ayrı satır" className="min-w-0 lg:col-span-3" bodyClass="max-h-[420px] overflow-y-auto maven-scroll">
   138→          {loading ? <Loading rows={6} /> : error ? <ErrorState message={error} onRetry={reload} /> : (scans ?? []).length === 0 ? (
   139→            <EmptyState title="Bu kapı için giriş kaydı yok" desc="Doğru gün ve kapıyı seçtiğinizden emin olun." />
   140→          ) : (
   141→            <div className="space-y-1.5">
   142→              {(scans ?? []).map((s) => (
   143→                <div key={s.id} className={cn("flex flex-wrap items-center gap-2 rounded-lg border px-3 py-2 text-sm",
   144→                  s.result === "ALLOWED" ? "border-emerald-100 bg-emerald-50/40" : s.result === "RESCAN_WARNING" ? "border-amber-200 bg-amber-50/60" : "border-rose-200 bg-rose-50/60")}>
   145→                  <span className={cn("grid size-6 place-items-center rounded-full text-white",
   146→                    s.result === "ALLOWED" ? "bg-emerald-500" : s.result === "RESCAN_WARNING" ? "bg-amber-500" : "bg-rose-500")}>
   147→                    {s.result === "ALLOWED" ? <Icons.Check className="size-3.5" /> : s.result === "RESCAN_WARNING" ? <Icons.RotateCcw className="size-3.5" /> : <Icons.X className="size-3.5" />}
   148→                  </span>
   149→                  <span className="min-w-0 flex-1 truncate font-medium">
   150→                    {s.participation ? `${s.participation.person.firstName} ${s.participation.person.lastName}` : "Bilinmeyen kod"}
   151→                    <span className="ml-2 text-xs text-muted-foreground">{s.action} · {s.location}{s.doorName ? ` (${s.doorName})` : ""}</span>
   152→                  </span>
   153→                  {s.reason && <span className="hidden max-w-56 truncate text-xs text-muted-foreground md:inline">{s.reason}</span>}
   154→                  <span className="text-xs tabular-nums text-muted-foreground">{fmtDateTime(s.scannedAt)}</span>
   155→                  <span className="text-[11px] text-muted-foreground">{s.operator}</span>
   156→                </div>
   157→              ))}
   158→            </div>
   159→          )}
   160→        </SectionCard>
   161→      </div>
   162→
   163→      <Dialog open={Boolean(denyTarget)} onOpenChange={(o) => !o && setDenyTarget(null)}>
   164→        <DialogContent className="sm:max-w-sm">
   165→          <DialogHeader><DialogTitle>Manuel İstisna</DialogTitle><DialogDescription>Gerekçe zorunlu; karar tarama geçmişine DENIED + istisna olarak işlenir.</DialogDescription></DialogHeader>
   166→          <div><Label>Gerekçe</Label><Textarea value={forceReason} onChange={(e) => setForceReason(e.target.value)} placeholder="Örn. yaka kartı basımı sürüyor, kimlik ibraz edildi…" className="mt-1" /></div>
   167→          <DialogFooter>
   168→            <Button variant="outline" onClick={() => setDenyTarget(null)}>Vazgeç</Button>
   169→            <Button disabled={!forceReason} onClick={() => scan(forceReason)}>İstisna Uygula</Button>
   170→          </DialogFooter>
   171→        </DialogContent>
   172→      </Dialog>
   173→    </div>
   174→  );
   175→}

   473→  const fillFor = (tpl: string): string =>

   476→      .replace(/\{\{title\}\}/g, previewIssue?.participation?.person.title || "Ar-Ge Müdürü")

   512→    [els[idx], els[target]] = [els[target], els[idx]];

   545→    } else {
   546→      updateCertElement(d.id, { w: Math.max(1, certR1(d.origW + dx)), h: Math.max(0.4, certR1(d.origH + dy)) });
   547→    }
   548→  };

   554→    const el = certElements.find((x) => x.id === selElId);
   555→    if (!el) return;
   556→    const step = e.shiftKey ? 5 : 1;
   557→    if (e.key === "ArrowLeft") { updateCertElement(selElId, { x: certR1(Math.max(0, el.x - step)) }); e.preventDefault(); }
   558→    else if (e.key === "ArrowRight") { updateCertElement(selElId, { x: certR1(el.x + step) }); e.preventDefault(); }
   559→    else if (e.key === "ArrowUp") { updateCertElement(selElId, { y: certR1(Math.max(0, el.y - step)) }); e.preventDefault(); }
   560→    else if (e.key === "ArrowDown") { updateCertElement(selElId, { y: certR1(el.y + step) }); e.preventDefault(); }
   561→    else if (e.key === "Delete" || e.key === "Backspace") { removeCertElement(selElId); e.preventDefault(); }
   562→  };
   563→
   564→  const renderCertElement = (el: CertElement) => {

   567→    const style: React.CSSProperties = {
   568→      position: "absolute", left: el.x * CERT_PX_PER_MM, top: el.y * CERT_PX_PER_MM,

   578→    } else if (el.type === "line") {
   579→      content = <div className="size-full" style={{ background: color, opacity: 0.85, borderRadius: (el.radius ?? 0) * CERT_PX_PER_MM }} />;
   580→    } else if (el.type === "image") {
   581→      content = el.imageDataUrl
   582→        ? <img src={el.imageDataUrl} alt="Sertifika görseli" className="size-full object-cover" style={{ borderRadius: (el.radius ?? 0) * CERT_PX_PER_MM }} />
   583→        : <span className={cn("flex size-full items-center justify-center rounded border border-dashed px-1 text-center text-[10px] leading-tight", selected ? "border-teal-500 text-teal-700" : "border-slate-400/70 text-slate-500")}>{el.text || "Görsel alanı"}</span>;
   584→    } else {
   585→      content = (
   586→        <span className={cn("flex size-full items-center justify-center rounded-[2px] border border-dashed", certPreviewMode ? "border-teal-600/50 bg-teal-50/60" : "border-teal-500/70 bg-teal-50")}>
   587→          <Icons.QrCode className="text-teal-700" style={{ width: el.h * CERT_PX_PER_MM * 0.62, height: el.h * CERT_PX_PER_MM * 0.62 }} />
   588→        </span>
   589→      );
   590→    }
   591→    return (
   592→      <div
   593→        key={el.id}
   594→        style={style}
   595→        onPointerDown={(e) => startDrag(e, el, "move")}
   596→        onPointerMove={moveDrag}
   597→        onPointerUp={endDrag}
   598→        className={cn(
   599→          "cursor-move touch-none select-none",
   600→          selected && "ring-2 ring-teal-500 ring-offset-1 ring-offset-white",
   601→          el.type === "text" && "bg-slate-500/5",
   602→        )}
   603→        title={`${CERT_ELEMENT_LABELS[el.type]} — sürükleyerek taşı`}
   604→      >
   605→        {content}
   606→        {selected && (
   607→          <span
   608→            onPointerDown={(e) => startDrag(e, el, "resize")}
   609→            onPointerMove={moveDrag}
   610→            onPointerUp={endDrag}
   611→            className="absolute -bottom-1.5 -right-1.5 size-3 cursor-nwse-resize rounded-sm border border-teal-600 bg-teal-500"
   612→            aria-label="Boyutlandırma tutamacı"
   613→          />
   614→        )}
   615→      </div>
   616→    );
