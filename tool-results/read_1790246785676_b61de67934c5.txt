   128→   128→function LinkedPhotoUploader({
   129→   129→  editionId, systemFolder, linkedType, linkedId, assetName, currentUrl, onApply, folderLabel, alt, fit = "cover",
   130→   130→}: {
   131→   131→  editionId: string | null;
   132→   132→  systemFolder: string; // KISI_FOTOGRAF | KURUM_LOGO
   133→   133→  linkedType: "PERSON" | "ORGANIZATION";
   134→   134→  linkedId: string;
   135→   135→  assetName: string;
   136→   136→  currentUrl?: string | null;
   137→   137→  onApply: (dataUrl: string) => Promise<void>;
   138→   138→  folderLabel: string;
   139→   139→  alt: string;
   140→   140→  fit?: "cover" | "contain";
   141→   141→}) {
   142→   142→  const { toast } = useToast();
   143→   143→  const fileRef = useRef<HTMLInputElement>(null);
   144→   144→  const [busy, setBusy] = useState(false);
   145→   145→  const [preview, setPreview] = useState<string | null>(currentUrl ?? null);
   146→   146→  useEffect(() => { setPreview(currentUrl ?? null); }, [currentUrl]);
   147→   147→
   148→   148→  const pick = async (ev: React.ChangeEvent<HTMLInputElement>) => {
   149→   149→    const file = ev.target.files?.[0];
   150→   150→    ev.target.value = ""; // aynı dosya yeniden seçilebilsin
   151→   151→    if (!file) return;
   152→   152→    if (!file.type.startsWith("image/")) {
   153→   153→      toast({ title: "Desteklenmeyen dosya", description: "Lütfen bir görsel dosyası seçin (image/*).", variant: "destructive" });
   154→   154→      return;
   155→   155→    }
   156→   156→    if (file.size > MAX_IMAGE_KB * 1024) {
   157→   157→      toast({ title: "Dosya çok büyük", description: `Görsel en fazla ${MAX_IMAGE_KB} KB olabilir — daha küçük bir dosya seçin.`, variant: "destructive" });
   158→   158→      return;
   159→   159→    }
   160→   160→    setBusy(true);
   161→   161→    try {
   162→   162→      const dataUrl = await new Promise<string>((resolve, reject) => {
   163→   163→        const fr = new FileReader();
   164→   164→        fr.onload = () => resolve(String(fr.result));
   165→   165→        fr.onerror = () => reject(new Error("Dosya okunamadı"));
   166→   166→        fr.readAsDataURL(file);
   167→   167→      });
   168→   168→      const res = await apiSend<UploadLinkedResult>("/api/media/upload-linked", "POST", {
   169→   169→        editionId, systemFolder, name: assetName, dataUrl, linkedType, linkedId,
   170→   170→      });
   171→   171→      const finalUrl = res.asset.dataUrl ?? dataUrl;
   172→   172→      await onApply(finalUrl);
   173→   173→      setPreview(finalUrl);
   174→   174→      toast({ title: "Görsel yüklendi", description: `Medya Arşivi → ${folderLabel}: ${res.asset.name}` });
   175→   175→    } catch (e) {
   176→   176→      toast({ title: "Yükleme başarısız", description: e instanceof Error ? e.message : "Görsel yüklenemedi", variant: "destructive" });
   177→   177→    } finally {
   178→   178→      setBusy(false);
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  return (
   183→   183→    <div className="flex items-start gap-3">
   184→   184→      {preview ? (
   185→   185→        <img src={preview} alt={alt} className={cn("size-14 shrink-0 rounded-lg border bg-background", fit === "contain" ? "object-contain p-0.5" : "object-cover")} />
   186→   186→      ) : (
   187→   187→        <span className="grid size-14 shrink-0 place-items-center rounded-lg border border-dashed bg-muted/30 text-muted-foreground" aria-hidden>
   188→   188→          <Icons.ImageUp className="size-5" />
   189→   189→        </span>
   190→   190→      )}
   191→   191→      <div className="min-w-0 flex-1">
   192→   192→        <div className="flex flex-wrap items-center gap-2">
   193→   193→          <Button type="button" size="sm" variant="outline" disabled={busy || !editionId} onClick={() => fileRef.current?.click()}>
   194→   194→            {busy ? <Icons.Loader2 className="size-3.5 animate-spin" aria-hidden /> : <Icons.ImageUp className="size-3.5" />}
   195→   195→            {preview ? "Değiştir" : "Görsel Seç"}
   196→   196→          </Button>
   197→   197→          <span className="text-[11px] text-muted-foreground">{preview ? "görsel bağlı" : "görsel yok"}</span>
   198→   198→        </div>
   199→   199→        <p className="mt-1 text-[11px] leading-snug text-muted-foreground">Medya Arşivi → {folderLabel} klasörüne benzersiz adla kaydedilir (≤ {MAX_IMAGE_KB} KB).</p>
   200→   200→      </div>
   201→   201→      <input ref={fileRef} type="file" accept="image/*" className="sr-only" onChange={pick} disabled={busy} aria-label="Görsel dosyası seç" />
   202→   202→    </div>
   203→   203→  );
   204→   204→}
