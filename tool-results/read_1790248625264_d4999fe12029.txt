   108→   128→function LinkedPhotoUploader({
   109→   129→  editionId, systemFolder, linkedType, linkedId, assetName, currentUrl, onApply, folderLabel, alt, fit = "cover",
   110→   130→}: {
   111→   131→  editionId: string | null;
   112→   132→  systemFolder: string; // KISI_FOTOGRAF | KURUM_LOGO
   113→   133→  linkedType: "PERSON" | "ORGANIZATION";
   114→   134→  linkedId: string;
   115→   135→  assetName: string;
   116→   136→  currentUrl?: string | null;
   117→   137→  onApply: (dataUrl: string) => Promise<void>;
   118→   138→  folderLabel: string;
   119→   139→  alt: string;
   120→   140→  fit?: "cover" | "contain";
   121→   141→}) {
   122→   142→  const { toast } = useToast();
   123→   143→  const fileRef = useRef<HTMLInputElement>(null);
   124→   144→  const [busy, setBusy] = useState(false);
   125→   145→  const [preview, setPreview] = useState<string | null>(currentUrl ?? null);
   126→   146→  useEffect(() => { setPreview(currentUrl ?? null); }, [currentUrl]);
   127→   147→
   128→   148→  const pick = async (ev: React.ChangeEvent<HTMLInputElement>) => {
   129→   149→    const file = ev.target.files?.[0];
   130→   150→    ev.target.value = ""; // aynı dosya yeniden seçilebilsin
   131→   151→    if (!file) return;
   132→   152→    if (!file.type.startsWith("image/")) {
   133→   153→      toast({ title: "Desteklenmeyen dosya", description: "Lütfen bir görsel dosyası seçin (image/*).", variant: "destructive" });
   134→   154→      return;
   135→   155→    }
   136→   156→    if (file.size > MAX_IMAGE_KB * 1024) {
   137→   157→      toast({ title: "Dosya çok büyük", description: `Görsel en fazla ${MAX_IMAGE_KB} KB olabilir — daha küçük bir dosya seçin.`, variant: "destructive" });
   138→   158→      return;
   139→   159→    }
   140→   160→    setBusy(true);
   141→   161→    try {
   142→   162→      const dataUrl = await new Promise<string>((resolve, reject) => {
   143→   163→        const fr = new FileReader();
   144→   164→        fr.onload = () => resolve(String(fr.result));
   145→   165→        fr.onerror = () => reject(new Error("Dosya okunamadı"));
   146→   166→        fr.readAsDataURL(file);
   147→   167→      });
   148→   168→      const res = await apiSend<UploadLinkedResult>("/api/media/upload-linked", "POST", {
   149→   169→        editionId, systemFolder, name: assetName, dataUrl, linkedType, linkedId,
   150→   170→      });
   151→   171→      const finalUrl = res.asset.dataUrl ?? dataUrl;
   152→   172→      await onApply(finalUrl);
   153→   173→      setPreview(finalUrl);
   154→   174→      toast({ title: "Görsel yüklendi", description: `Medya Arşivi → ${folderLabel}: ${res.asset.name}` });
   155→   175→    } catch (e) {
   156→   176→      toast({ title: "Yükleme başarısız", description: e instanceof Error ? e.message : "Görsel yüklenemedi", variant: "destructive" });
   157→   177→    } finally {
   158→   178→      setBusy(false);
   159→   179→    }
   160→   180→  };
   161→   181→
   162→   182→  return (
   163→   183→    <div className="flex items-start gap-3">
   164→   184→      {preview ? (
   165→   185→        <img src={preview} alt={alt} className={cn("size-14 shrink-0 rounded-lg border bg-background", fit === "contain" ? "object-contain p-0.5" : "object-cover")} />
   166→   186→      ) : (
   167→   187→        <span className="grid size-14 shrink-0 place-items-center rounded-lg border border-dashed bg-muted/30 text-muted-foreground" aria-hidden>
   168→   188→          <Icons.ImageUp className="size-5" />
   169→   189→        </span>
   170→   190→      )}
   171→   191→      <div className="min-w-0 flex-1">
   172→   192→        <div className="flex flex-wrap items-center gap-2">
   173→   193→          <Button type="button" size="sm" variant="outline" disabled={busy || !editionId} onClick={() => fileRef.current?.click()}>
   174→   194→            {busy ? <Icons.Loader2 className="size-3.5 animate-spin" aria-hidden /> : <Icons.ImageUp className="size-3.5" />}
   175→   195→            {preview ? "Değiştir" : "Görsel Seç"}
   176→   196→          </Button>
   177→   197→          <span className="text-[11px] text-muted-foreground">{preview ? "görsel bağlı" : "görsel yok"}</span>
   178→   198→        </div>
   179→   199→        <p className="mt-1 text-[11px] leading-snug text-muted-foreground">Medya Arşivi → {folderLabel} klasörüne benzersiz adla kaydedilir (≤ {MAX_IMAGE_KB} KB).</p>
   180→   200→      </div>
   181→   201→      <input ref={fileRef} type="file" accept="image/*" className="sr-only" onChange={pick} disabled={busy} aria-label="Görsel dosyası seç" />
