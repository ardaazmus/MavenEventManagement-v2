     1→"use client";

    11→import { B2B_PLAN_STATUS, B2B_ASSIGNMENT_STATUS, B2B_ROLES, label } from "@/lib/constants";
    12→import { Button } from "@/components/ui/button";
    13→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from "@/components/ui/dialog";
    14→import { Input } from "@/components/ui/input";

    18→import { Switch } from "@/components/ui/switch";
    19→import { useToast } from "@/hooks/use-toast";
    20→import * as Icons from "lucide-react";
    21→import { cn } from "@/lib/utils";

    23→function toLocalInput(iso?: string | null): string {
    24→  if (!iso) return "";
    25→  const d = new Date(iso);
    26→  if (Number.isNaN(d.getTime())) return "";
    27→  const pad = (n: number) => String(n).padStart(2, "0");
    28→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    29→}

    31→interface B2bPerson { id: string; firstName: string; lastName: string; email?: string | null; company?: string | null; title?: string | null; photoUrl?: string | null }
    32→
    33→interface B2bAssignmentRow {
    34→  id: string; planId: string; personId: string; role: string;
    35→  status: string; organizerApproved: boolean; personApproved: boolean;
    36→  feedback?: string | null; feedbackAt?: string | null; respondedAt?: string | null;
    37→  person?: B2bPerson;
    38→}

    40→interface B2bPlanRow {
    41→  id: string; editionId: string; subject: string; description?: string | null;
    42→  startsAt?: string | null; endsAt?: string | null; venue?: string | null; location?: string | null;
    43→  isPrivate: boolean; status: string; notes?: string | null;
    44→  assignments?: B2bAssignmentRow[];
    45→  _count?: { assignments?: number };
    46→}

    48→const emptyForm = {
    49→  subject: "", description: "", startsAt: "", endsAt: "", venue: "", location: "",
    50→  isPrivate: true, status: "DRAFT", notes: "",
    51→};
    52→
    53→export function B2bView() {
    54→  const { currentEditionId, bump, refreshKey } = useApp();
    55→  const { toast } = useToast();
    56→
    57→  const { data: plans, reload } = useApi<B2bPlanRow[]>(
    58→    () => listEntity<B2bPlanRow>("b2b-plans", { editionId: currentEditionId ?? undefined, limit: 200 }),
    59→    [currentEditionId, refreshKey],
    60→  );
    61→
    62→  const { data: participations } = useApi<{ id: string; person: B2bPerson }[]>(
    63→    () => listEntity("participations", { editionId: currentEditionId ?? undefined, limit: 500 }),
    64→    [currentEditionId, refreshKey],
    65→  );
    66→
    67→  const [statusFilter, setStatusFilter] = useState<string>("ALL");
    68→  const [expandedId, setExpandedId] = useState<string | null>(null);
    69→
    70→  // ── plan formu (create + edit — çift tık) ──
    71→  const [formOpen, setFormOpen] = useState(false);
    72→  const [editingId, setEditingId] = useState<string | null>(null);
    73→  const [form, setForm] = useState({ ...emptyForm });
    74→  const [busy, setBusy] = useState(false);
    75→
    76→  // ── kişi atama diyaloğu ──
    77→  const [assignPlan, setAssignPlan] = useState<B2bPlanRow | null>(null);
    78→  const [assignSelected, setAssignSelected] = useState<string[]>([]);
    79→  const [assignRole, setAssignRole] = useState("PARTICIPANT");
    80→  const [assignSearch, setAssignSearch] = useState("");
    81→  const [assignBusy, setAssignBusy] = useState(false);
    82→
    83→  // ── mobil önizleme (Mobil App simülasyonu) ──
    84→  const [mobileOpen, setMobileOpen] = useState(false);
    85→  const [mobilePersonId, setMobilePersonId] = useState<string>("");
    86→  const [mobileFeedback, setMobileFeedback] = useState<Record<string, string>>({});
    87→
    88→  const personOptions = useMemo(() => {
    89→    const map = new Map<string, B2bPerson>();
    90→    for (const p of participations ?? []) if (p.person && !map.has(p.person.id)) map.set(p.person.id, p.person);
    91→    return Array.from(map.values());
    92→  }, [participations]);
    93→
    94→  const filtered = useMemo(() => (plans ?? []).filter((p) => statusFilter === "ALL" || p.status === statusFilter), [plans, statusFilter]);
    95→
    96→  const stats = useMemo(() => {
    97→    const all = plans ?? [];
    98→    return {
    99→      total: all.length,
   100→      active: all.filter((p) => p.status === "ACTIVE").length,
   101→      pending: all.filter((p) => p.status === "PENDING_APPROVAL" || p.status === "DRAFT").length,
   102→      assignments: all.reduce((s, p) => s + (p.assignments?.length ?? 0), 0),
   103→      accepted: all.reduce((s, p) => s + (p.assignments ?? []).filter((a) => a.status === "ACCEPTED").length, 0),
   104→      feedbacks: all.reduce((s, p) => s + (p.assignments ?? []).filter((a) => Boolean(a.feedback)).length, 0),
   105→    };
   106→  }, [plans]);
   107→
   108→  const openCreate = () => { setEditingId(null); setForm({ ...emptyForm }); setFormOpen(true); };
   109→
   110→  const openEdit = (p: B2bPlanRow) => {
   111→    setEditingId(p.id);
   112→    setForm({
   113→      subject: p.subject, description: p.description ?? "",
   114→      startsAt: toLocalInput(p.startsAt), endsAt: toLocalInput(p.endsAt),
   115→      venue: p.venue ?? "", location: p.location ?? "",
   116→      isPrivate: p.isPrivate, status: p.status, notes: p.notes ?? "",
   117→    });
   118→    setFormOpen(true);
   119→  };
   120→
   121→  const saveForm = async () => {
   122→    if (!currentEditionId) return;
   123→    if (!form.subject.trim()) { toast({ title: "Konu zorunlu", variant: "destructive" }); return; }
   124→    setBusy(true);
   125→    try {
   126→      const body = {
   127→        subject: form.subject.trim(),
   128→        description: form.description || null,
   129→        startsAt: form.startsAt ? new Date(form.startsAt).toISOString() : null,
   130→        endsAt: form.endsAt ? new Date(form.endsAt).toISOString() : null,
   131→        venue: form.venue || null, location: form.location || null,
   132→        isPrivate: form.isPrivate, status: form.status, notes: form.notes || null,
   133→        editionId: currentEditionId,
   134→      };
   135→      if (editingId) await apiSend(`/api/b2b-plans/${editingId}`, "PUT", body);
   136→      else await apiSend("/api/b2b-plans", "POST", body);
   137→      toast({ title: editingId ? "B2B planı güncellendi" : "B2B planı oluşturuldu", description: form.subject });
   138→      setFormOpen(false);
   139→      reload(); bump();
   140→    } catch (e) {
   141→      toast({ title: "Kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   142→    } finally { setBusy(false); }
   143→  };
   144→
   145→  const removePlan = async (p: B2bPlanRow) => {
   146→    if (!window.confirm(`"${p.subject}" B2B planı ve tüm atamaları silinsin mi?`)) return;
   147→    try {
   148→      await apiSend(`/api/b2b-plans/${p.id}`, "DELETE");
   149→      toast({ title: "Plan silindi" });
   150→      reload(); bump();
   151→    } catch (e) {
   152→      toast({ title: "Silinemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   153→    }
   154→  };
   155→
   156→  const openAssign = (p: B2bPlanRow) => {
   157→    setAssignPlan(p);
   158→    setAssignSelected([]);
   159→    setAssignRole("PARTICIPANT");
   160→    setAssignSearch("");
   161→  };
   162→
   163→  const sendAssignments = async () => {
   164→    if (!assignPlan || assignSelected.length === 0) return;
   165→    setAssignBusy(true);
   166→    try {
   167→      let n = 0;
   168→      for (const personId of assignSelected) {
   169→        await apiSend("/api/b2b-assignments", "POST", { planId: assignPlan.id, personId, role: assignRole, status: "ASSIGNED" });
   170→        n++;
   171→      }
   172→      toast({ title: "Atama tamamlandı", description: `${n} kişiye B2B planı atandı — mobil uygulamadan yanıt bekleniyor.` });
   173→      setAssignPlan(null);
   174→      reload(); bump();
   175→    } catch (e) {
   176→      toast({ title: "Atanamadı", description: e instanceof Error ? e.message : "Hata (aynı kişiye çift atama olabilir)", variant: "destructive" });
   177→    } finally { setAssignBusy(false); }
   178→  };
   179→
   180→  const removeAssignment = async (a: B2bAssignmentRow) => {
   181→    try {
   182→      await apiSend(`/api/b2b-assignments/${a.id}`, "DELETE");
   183→      reload(); bump();
   184→    } catch (e) {
   185→      toast({ title: "Atama kaldırılamadı", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   186→    }
   187→  };
   188→
   189→  const organizerApprove = async (a: B2bAssignmentRow, approved: boolean) => {
   190→    try {
   191→      const r = await apiSend<{ planActivated?: boolean }>("/api/flows", "POST", { action: "b2b.approve", assignmentId: a.id, approved });
   192→      toast({ title: approved ? "Organizatör onayı verildi" : "Organizatör onayı geri alındı", description: r.planActivated ? "Karşılıklı onay tamam — plan ETKİN duruma geçti!" : undefined });
   193→      reload(); bump();
   194→    } catch (e) {
   195→      toast({ title: "Onay işlenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   196→    }
   197→  };
   198→
   199→  const setPlanStatus = async (p: B2bPlanRow, status: string) => {
   200→    try {
   201→      await apiSend(`/api/b2b-plans/${p.id}`, "PUT", { status });
   202→      toast({ title: `Plan durumu: ${label(B2B_PLAN_STATUS, status)}` });
   203→      reload(); bump();
   204→    } catch (e) {
   205→      toast({ title: "Durum değiştirilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   206→    }
   207→  };
   208→
   209→  // ── mobil akış: kişi yanıtı (kabul/red + görüş) ──
   210→  const mobileAssignments = useMemo(() => {
   211→    const out: { plan: B2bPlanRow; a: B2bAssignmentRow }[] = [];
   212→    for (const p of plans ?? []) for (const a of p.assignments ?? []) if (a.personId === mobilePersonId) out.push({ plan: p, a });
   213→    return out;
   214→  }, [plans, mobilePersonId]);
   215→
   216→  const respondMobile = async (a: B2bAssignmentRow, accepted: boolean) => {
   217→    try {
   218→      const feedback = (mobileFeedback[a.id] ?? "").trim() || undefined;
   219→      const r = await apiSend<{ planActivated?: boolean }>("/api/flows", "POST", { action: "b2b.respond", assignmentId: a.id, accepted, feedback, respondedBy: "Mobil Uygulama" });
   220→      toast({ title: accepted ? "Plan kabul edildi ✓" : "Plan reddedildi", description: r.planActivated ? "Karşılıklı onay tamamlandı — plan etkin." : "Organizatör onayı bekleniyor." });
   221→      reload(); bump();
   222→    } catch (e) {
   223→      toast({ title: "Yanıt gönderilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   224→    }
   225→  };
   226→
   227→  const saveFeedbackMobile = async (a: B2bAssignmentRow) => {
   228→    try {
   229→      const feedback = (mobileFeedback[a.id] ?? "").trim();
   230→      if (!feedback) { toast({ title: "Görüş yazın", variant: "destructive" }); return; }
   231→      await apiSend(`/api/b2b-assignments/${a.id}`, "PUT", { feedback, feedbackAt: new Date().toISOString() });
   232→      toast({ title: "Görüş kaydedildi", description: "Organizatör görüşü panoda görebilir." });
   233→      reload(); bump();
   234→    } catch (e) {
   235→      toast({ title: "Görüş kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   236→    }
   237→  };
   238→
   239→  const mobilePersons = useMemo(() => {
   240→    const map = new Map<string, B2bPerson>();
   241→    for (const p of plans ?? []) for (const a of p.assignments ?? []) if (a.person && !map.has(a.person.id)) map.set(a.person.id, a.person);
   242→    return Array.from(map.values());
   243→  }, [plans]);
   244→
   245→  if (!currentEditionId) return <EmptyState title="Edisyon seçin" desc="B2B planları edisyona bağlıdır." />;
   246→
   247→  return (
   248→    <div className="space-y-5">
   249→      <PageHeader title="B2B Planı" desc="Konu · Saat · Etkinlik yeri · Konum girişli B2B planları — kişilere atanır, mobil uygulamadan kabul edilir, karşılıklı onayla etkinleşir">
   250→        <div className="flex gap-2">
   251→          <Button size="sm" variant="outline" onClick={() => { setMobilePersonId(""); setMobileOpen(true); }}>
   252→            <Icons.Smartphone className="size-4" /> Mobil Önizleme
   253→          </Button>
   254→          <Button size="sm" onClick={openCreate}><Icons.Plus className="size-4" /> Yeni B2B Planı</Button>
   255→        </div>
   256→      </PageHeader>
   257→
   258→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
   259→        <KpiCard label="Toplam Plan" value={stats.total} icon={<Icons.Briefcase className="size-4" />} />
   260→        <KpiCard label="Etkin" value={stats.active} tone="emerald" icon={<Icons.CheckCircle2 className="size-4" />} />
   261→        <KpiCard label="Onay Bekleyen" value={stats.pending} tone="amber" icon={<Icons.Hourglass className="size-4" />} />
   262→        <KpiCard label="Atama" value={stats.assignments} tone="violet" icon={<Icons.UserPlus className="size-4" />} />
   263→        <KpiCard label="Kabul" value={stats.accepted} tone="teal" icon={<Icons.ThumbsUp className="size-4" />} />
   264→        <KpiCard label="Görüş Bildiren" value={stats.feedbacks} tone="rose" icon={<Icons.MessageSquare className="size-4" />} />
   265→      </div>
   266→
   267→      <div className="flex flex-wrap items-center gap-2">
   268→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   269→          <SelectTrigger className="h-8 w-[210px]" aria-label="Durum filtresi"><SelectValue /></SelectTrigger>
   270→          <SelectContent>
   271→            <SelectItem value="ALL">Tüm durumlar</SelectItem>
   272→            {Object.entries(B2B_PLAN_STATUS).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}
   273→          </SelectContent>
   274→        </Select>
   275→        <Chip tone="teal">{filtered.length} plan</Chip>
   276→        <p className="text-xs text-muted-foreground">Karşılıklı onay: kişinin mobil kabulü + organizatör onayı → plan ETKİN olur.</p>
   277→      </div>
   278→
   279→      {filtered.length === 0 ? (
   280→        <EmptyState
   281→          title="Henüz B2B planı yok"
   282→          desc="Konu, saat, etkinlik yeri ve konum bilgisi girip planı kişilere atayın; mobil uygulamadan gelen kabul ve görüşleri buradan izleyin."
   283→          action={<Button onClick={openCreate}><Icons.Plus className="size-4" /> İlk B2B planını oluştur</Button>}
   284→        />
   285→      ) : (
   286→        <div className="grid gap-4 lg:grid-cols-2">
   287→          {filtered.map((p) => {
   288→            const asg = p.assignments ?? [];
   289→            const allAccepted = asg.length > 0 && asg.every((a) => a.status === "ACCEPTED");
   290→            return (
   291→              <SectionCard
   292→                key={p.id}
   293→                title={p.subject}
   294→                desc={[p.venue, p.location].filter(Boolean).join(" · ") || "Yer/konum girilmedi"}
   295→                action={<StatusBadge map={B2B_PLAN_STATUS} value={p.status} />}
   296→                className="transition hover:shadow-md"
   297→              >
   298→                <div onDoubleClick={() => openEdit(p)} title="Çift tıkla → düzenle">
   299→                  <div className="flex flex-wrap items-center gap-1.5">
   300→                    {p.isPrivate ? <Chip tone="violet"><span className="inline-flex items-center gap-1"><Icons.Lock className="size-3" />Kişiye özel</span></Chip> : <Chip tone="teal">Genel</Chip>}
   301→                    {p.startsAt && <Chip><span className="inline-flex items-center gap-1"><Icons.Clock className="size-3" />{new Date(p.startsAt).toLocaleString("tr-TR", { dateStyle: "short", timeStyle: "short" })}</span></Chip>}
   302→                    {p.location && <Chip><span className="inline-flex items-center gap-1"><Icons.MapPin className="size-3" />{p.location}</span></Chip>}
   303→                    <Chip tone={allAccepted && asg.some((a) => a.organizerApproved) ? "emerald" : "amber"}><Icons.Users className="mr-1 inline size-3" />{asg.length} kişi</Chip>
   304→                  </div>
   305→                  {p.description && <p className="mt-2 line-clamp-2 text-xs text-muted-foreground">{p.description}</p>}
   306→                  {p.notes && <p className="mt-1 line-clamp-1 text-xs italic text-muted-foreground"><Icons.StickyNote className="mr-1 inline size-3" />{p.notes}</p>}
   307→                </div>
   308→
   309→                {/* atama satırları */}
   310→                {expandedId === p.id && asg.length > 0 && (
   311→                  <div className="mt-3 max-h-64 space-y-1.5 overflow-y-auto rounded-lg border bg-muted/30 p-2 maven-scroll">
   312→                    {asg.map((a) => (
   313→                      <div key={a.id} className="rounded border bg-background px-2 py-1.5 text-xs">
   314→                        <div className="flex items-center gap-2">
   315→                          <Icons.User className="size-3.5 shrink-0 text-muted-foreground" />
   316→                          <span className="min-w-0 flex-1 truncate font-medium">{a.person ? `${a.person.firstName} ${a.person.lastName}` : "Kişi"}{a.person?.company ? ` — ${a.person.company}` : ""}</span>
   317→                          <Chip>{label(B2B_ROLES, a.role)}</Chip>
   318→                          {a.status === "ACCEPTED" ? <Chip tone="emerald">{label(B2B_ASSIGNMENT_STATUS, a.status)}</Chip>
   319→                            : a.status === "DECLINED" ? <Chip tone="rose">{label(B2B_ASSIGNMENT_STATUS, a.status)}</Chip>
   320→                            : <Chip tone="amber">{label(B2B_ASSIGNMENT_STATUS, a.status)}</Chip>}
   321→                          {a.organizerApproved ? (
   322→                            <Button size="icon" variant="ghost" className="size-6" aria-label="Organizatör onayını geri al" onClick={() => organizerApprove(a, false)} title="Onaylı — geri almak için tıkla"><Icons.BadgeCheck className="size-3.5 text-emerald-600" /></Button>
   323→                          ) : (
   324→                            <Button size="icon" variant="ghost" className="size-6" aria-label="Organizatör onayı ver" onClick={() => organizerApprove(a, true)} title="Organizatör onayı ver"><Icons.Circle className="size-3.5 text-muted-foreground" /></Button>
   325→                          )}
   326→                          <Button size="icon" variant="ghost" className="size-6" aria-label="Atamayı kaldır" onClick={() => removeAssignment(a)}><Icons.Trash2 className="size-3 text-muted-foreground" /></Button>
   327→                        </div>
   328→                        {a.feedback && (
   329→                          <p className="mt-1 flex items-start gap-1 rounded bg-teal-500/10 px-1.5 py-1 text-[11px] text-teal-700">
   330→                            <Icons.MessageSquare className="mt-0.5 size-3 shrink-0" /> Görüş: {a.feedback}
   331→                          </p>
   332→                        )}
   333→                      </div>
   334→                    ))}
   335→                  </div>
   336→                )}
   337→
   338→                <div className="mt-3 flex flex-wrap gap-2">
   339→                  <Button size="sm" onClick={() => openAssign(p)}><Icons.UserPlus className="size-3.5" /> Kişi Ata</Button>
   340→                  {asg.length > 0 && (
   341→                    <Button size="sm" variant="outline" onClick={() => setExpandedId(expandedId === p.id ? null : p.id)}>
   342→                      <Icons.List className="size-3.5" /> Atamalar ({asg.length})
   343→                    </Button>
   344→                  )}
   345→                  <Button size="sm" variant="ghost" onClick={() => { setMobilePersonId(""); setMobileOpen(true); }} aria-label="Mobil önizleme"><Icons.Smartphone className="size-3.5" /></Button>
   346→                  <Button size="sm" variant="ghost" onClick={() => openEdit(p)} aria-label="Düzenle"><Icons.Pencil className="size-3.5" /> Düzenle</Button>
   347→                  {p.status === "ACTIVE" && <Button size="sm" variant="ghost" onClick={() => setPlanStatus(p, "COMPLETED")}>Tamamlandı</Button>}
   348→                  <Button size="sm" variant="ghost" className="ml-auto text-rose-500 hover:text-rose-600" onClick={() => removePlan(p)} aria-label={`${p.subject} sil`}><Icons.Trash2 className="size-3.5" /></Button>
   349→                </div>
   350→              </SectionCard>
   351→            );
   352→          })}
   353→        </div>
   354→      )}
   355→
   356→      {/* Plan formu — create + edit */}
   357→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   358→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl maven-scroll">
   359→          <DialogHeader>
   360→            <DialogTitle>{editingId ? "B2B Planını Düzenle" : "Yeni B2B Planı"}</DialogTitle>
   361→            <DialogDescription>Konu, saat, etkinlik yeri ve konum girilir; plan kişiye atanır, mobil uygulamadan karşılıklı onayla etkinleşir.</DialogDescription>
   362→          </DialogHeader>
   363→          <div className="grid gap-3">
   364→            <div><Label>Konu *</Label><Input value={form.subject} onChange={(e) => setForm({ ...form, subject: e.target.value })} placeholder="Ölçüm altyapıları iş birliği görüşmesi" className="mt-1" /></div>
   365→            <div className="grid gap-3 sm:grid-cols-2">
