    24→    24→// datetime-local ↔ ISO yardımcıları
    25→    25→function toLocalInput(iso?: string | null): string {
    26→    26→  if (!iso) return "";
    27→    27→  const d = new Date(iso);
    28→    28→  if (Number.isNaN(d.getTime())) return "";
    29→    29→  const pad = (n: number) => String(n).padStart(2, "0");
    30→    30→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    31→    31→}
