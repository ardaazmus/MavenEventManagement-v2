    61→function relTime(d?: string | null): string {
    62→  if (!d) return "—";
    63→  const diff = Date.now() - new Date(d).getTime();
    64→  if (diff < 0) return "az önce";
    65→  if (diff < 60_000) return "az önce";
    66→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)} dk önce`;
    67→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)} sa önce`;
    68→  return `${Math.floor(diff / 86_400_000)} g önce`;
    69→}

    71→// authConfig JSON'u → { key, maskeli değer } — UI'da değerler asla düz gösterilmez
    72→function maskAuthConfig(configJson?: string | null): { key: string; masked: string }[] {
    73→  if (!configJson) return [];
    74→  try {
    75→    const obj = JSON.parse(configJson) as Record<string, unknown>;
    76→    return Object.entries(obj).map(([key, v]) => ({
    77→      key,
    78→      masked: typeof v === "string" && v.length > 0 ? "•".repeat(Math.max(6, Math.min(v.length, 16))) : String(v),
    79→    }));
    80→  } catch {
    81→    return [{ key: "authConfig", masked: "••••••" }];
    82→  }
    83→}
