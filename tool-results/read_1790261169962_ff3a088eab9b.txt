   126→const todayStr = (): string => {
   127→  const d = new Date();
   128→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);

   131→// benzersiz gider kodu: GSN-YYYY-XXX (sayaç = mevcut adet + 1; çakışırsa +1 dene)
   132→const nextExpenseCode = (existing: ExpenseRow[]): string => {
   133→  const year = new Date().getFullYear();
   134→  const taken = new Set(existing.map((e) => e.code));
   135→  let n = existing.length + 1;
   136→  let code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   137→  while (taken.has(code)) {
   138→    n += 1;
   139→    code = `GSN-${year}-${String(n).padStart(3, "0")}`;
   140→  }
   141→  return code;
   142→};
   143→
   144→// benzersiz gelir kodu: GLR-YYYY-XXX (giderdeki desenin aynası)
   145→const nextIncomeCode = (existing: IncomeRow[]): string => {
   146→  const year = new Date().getFullYear();
   147→  const taken = new Set(existing.map((e) => e.code));
   148→  let n = existing.length + 1;
   149→  let code = `GLR-${year}-${String(n).padStart(3, "0")}`;
   150→  while (taken.has(code)) {
   151→    n += 1;
   152→    code = `GLR-${year}-${String(n).padStart(3, "0")}`;
   153→  }
   154→  return code;
   155→};
