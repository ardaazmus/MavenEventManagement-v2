    66→  const [guestTick, setGuestTick] = useState(0);

    77→  const { data: companions, reload: reloadCompanions } = useApi<CompanionRow[]>(
    78→    () => listEntity<CompanionRow>("companions", { participationId: guestRes?.primaryGuest?.id ?? "__none__" }),

    82→  const nightsOf = (r: ReservationRow) => Math.max(1, Math.round((new Date(r.checkOut).getTime() - new Date(r.checkIn).getTime()) / 86400000));

    85→  const roomNightsSold = (reservations ?? []).filter((r) => ["RESERVED", "CONFIRMED", "CHECKED_IN"].includes(r.status)).reduce((s, r) => s + nightsSaved(r), 0);
    86→  const pendingRes = (reservations ?? []).filter((r) => ["REQUESTED", "WAITLIST"].includes(r.status));
    87→  const confirmedCount = (reservations ?? []).filter((r) => ["CONFIRMED", "CHECKED_IN"].includes(r.status)).length;
    88→  const noShowList = (reservations ?? []).filter((r) => r.noShow);
    89→  const noShowFeeTotal = noShowList.reduce((s, r) => s + (r.noShowFee ?? 0), 0);
    90→  const occupancyBreakdown = (["SINGLE", "DOUBLE", "FAMILY_SHARED"] as const).map((t) => ({
    91→    type: t,
    92→    count: (reservations ?? []).filter((r) => r.occupancyType === t && !r.noShow && r.status !== "CANCELLED").length,
    93→  }));

    98→      await apiSend("/api/flows", "POST", { action: "reservation.confirm", reservationId: r.id });

   100→      reload(); bump();
   101→    } catch (e) {
   102→      toast({ title: "Teyit engellendi", description: e instanceof Error ? e.message : "Bir gece eksikse 'teyitli' görünmez.", variant: "destructive" });
   103→    } finally { setBusyId(null); }
   104→  };

   107→  const openEdit = (r: ReservationRow) => {
   108→    setEditForm({
   109→      occupancyType: r.occupancyType ?? "__none__",
   110→      ratePerNight: fromMinor(r.ratePerNight ?? 0), // F6: kuruş→₺ girdi

   112→      noShow: r.noShow ?? false,
   113→      noShowFee: fromMinor(r.noShowFee ?? 0), // F6: kuruş→₺ girdi

   119→    if (!editRes) return;
   120→    setBusy(true);
   121→    try {
   122→      await apiSend(`/api/reservations/${editRes.id}`, "PUT", {
   123→        occupancyType: editForm.occupancyType === "__none__" ? null : editForm.occupancyType,
   124→        ratePerNight: toMinor(Number(editForm.ratePerNight) || 0), // F6: ₺ girdi→kuruş
   125→        nights: Math.max(0, Math.round(Number(editForm.nights) || 0)),
   126→        noShow: editForm.noShow,
   127→        noShowFee: editForm.noShow ? toMinor(Number(editForm.noShowFee) || 0) : 0, // F6: ₺→kuruş
   128→      });
   129→      toast({ title: "Rezervasyon güncellendi", description: `${editRes.guestName} · oda tipi/fiyat/gece kaydedildi.` });
   130→      setEditRes(null);
   131→      reloadRes(); bump();
   132→    } catch (e) {
   133→      toast({ title: "Güncellenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   134→    } finally { setBusy(false); }

   138→  const openNoShow = (r: ReservationRow) => {
   139→    setNoShowFee((r.ratePerNight ?? 0) > 0 ? (r.ratePerNight ?? 0) * nightsSaved(r) : 0);
   140→    setNoShowRes(r);
   141→  };
   142→
   143→  const markNoShow = async () => {
   144→    if (!noShowRes) return;
   145→    setBusy(true);
   146→    try {
   147→      await apiSend(`/api/reservations/${noShowRes.id}`, "PUT", { noShow: true, noShowFee: toMinor(Number(noShowFee) || 0) }); // F6
   148→      toast({ title: "No-show işaretlendi", description: `${noShowRes.guestName} sistemden düşüldü — ücret faturaya kalem olarak yansır.` });
   149→      setNoShowRes(null);
   150→      reloadRes(); bump();
   151→    } catch (e) {
   152→      toast({ title: "İşaretlenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   153→    } finally { setBusy(false); }
   154→  };
   155→
   156→  // ── misafir hiyerarşisi: bağımlı kişi + refakatçi + konuk slotu
   157→  const dependentsOf = (personId: string) => (people ?? []).filter((p) => p.parentPersonId === personId);
   158→  const primaryDependents = guestRes?.primaryGuest ? dependentsOf(guestRes.primaryGuest.person.id) : [];
   159→
   160→  const openAddGuest = () => {
   161→    setGuestForm({ firstName: "", lastName: "", relationType: "SPOUSE", ageType: "ADULT", makeCompanion: true, addSlot: true });
   162→    setAddGuestOpen(true);
   163→  };
   164→
   165→  const addGuestProfile = async () => {
   166→    const r = guestRes;
   167→    if (!r?.primaryGuest || !tenant?.id || !currentEditionId || !guestForm.firstName.trim() || !guestForm.lastName.trim()) return;
   168→    setBusy(true);
   169→    try {
   170→      // 1) Person — self-referencing parentPersonId ile (Parent_ID kuralı)
   171→      const person = await apiSend<{ id: string }>("/api/people", "POST", {
   172→        tenantId: tenant.id,
   173→        firstName: guestForm.firstName.trim(),
   174→        lastName: guestForm.lastName.trim(),
   175→        parentPersonId: r.primaryGuest.person.id,
   176→        relationType: guestForm.relationType,
   177→      });
   178→      // 2) Katılım (bu edisyonda yoksa) — @@unique([editionId, personId]) güvencesi
   179→      const existing = await listEntity<{ id: string }>("participations", { editionId: currentEditionId, personId: person.id });
   180→      let participationId = existing[0]?.id ?? null;
   181→      let participationNote = "mevcut katılım";
   182→      if (!participationId) {
   183→        const created = await apiSend<{ id: string }>("/api/participations", "POST", { editionId: currentEditionId, personId: person.id });
   184→        participationId = created.id;
   185→        participationNote = "yeni katılım";
   186→      }
   187→      // 3) Refakatçi kaydı (ana konuğun katılımına)
   188→      if (guestForm.makeCompanion) {
   189→        await apiSend("/api/companions", "POST", {
   190→          participationId: r.primaryGuest.id,
   191→          name: `${guestForm.firstName.trim()} ${guestForm.lastName.trim()}`,
   192→          type: guestForm.ageType,
   193→        });
   194→      }
   195→      // 4) Rezervasyon konuk slotu
   196→      if (guestForm.addSlot && participationId) {
   197→        await apiSend("/api/occupancy-slots", "POST", {
   198→          reservationId: r.id,
   199→          participationId,
   200→          guestName: `${guestForm.firstName.trim()} ${guestForm.lastName.trim()}`,
   201→        });
   202→      }
   203→      toast({
   204→        title: "Misafir profili eklendi",
   205→        description: `Kişi + ${participationNote}${guestForm.makeCompanion ? " + refakatçi" : ""}${guestForm.addSlot ? " + konuk slotu" : ""} kuruldu.`,
   206→      });
   207→      setAddGuestOpen(false);
   208→      reloadPeople(); reloadCompanions(); reloadRes(); setGuestTick((t) => t + 1); bump();
   209→    } catch (e) {
   210→      toast({ title: "Misafir eklenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   211→    } finally { setBusy(false); }
   212→  };
   213→
   214→  const openCreateHotel = () => {
   215→    setHotelForm({ name: "", city: "", district: "", address: "", starRating: "__none__", checkInNote: "", contactName: "", contactPhone: "", email: "", website: "", notes: "", logoUrl: "", imageUrl: "" });
   216→    setHotelPending({});
   217→    setHotelDialog({ target: null });
   218→  };
   219→
   220→  const openEditHotel = (h: HotelRow) => {
   221→    setHotelForm({
   222→      name: h.name, city: h.city ?? "", district: h.district ?? "", address: h.address ?? "",
   223→      starRating: h.starRating ? String(h.starRating) : "__none__", checkInNote: h.checkInNote ?? "",
   224→      contactName: h.contactName ?? "", contactPhone: h.contactPhone ?? "", email: h.email ?? "",
   225→      website: h.website ?? "", notes: h.notes ?? "", logoUrl: h.logoUrl ?? "", imageUrl: h.imageUrl ?? "",
   226→    });
   227→    setHotelPending({});
   228→    setHotelDialog({ target: h });
   229→  };
   230→
   231→  // Medya Arşivi → Otel Görselleri klasörüne benzersiz adla yükle + otel kaydına yaz
   232→  const uploadHotelMedia = async (hotelId: string, kind: "logo" | "cover", dataUrl: string) => {
   233→    const hotelName = hotelForm.name.trim() || "otel";
   234→    const r = await apiSend<{ asset: { id: string; dataUrl: string | null } }>("/api/media/upload-linked", "POST", {
   235→      editionId: currentEditionId,
   236→      systemFolder: "OTEL",
   237→      linkedType: "HOTEL",
   238→      linkedId: hotelId,
   239→      name: `${hotelName}-${kind === "logo" ? "logosu" : "kapak"}`,
   240→      dataUrl,
   241→    });
   242→    const url = r.asset.dataUrl ?? dataUrl;
   243→    await apiSend(`/api/hotels/${hotelId}`, "PUT", kind === "logo" ? { logoUrl: url } : { imageUrl: url });
   244→  };
   245→
   246→  const onHotelMediaPick = (kind: "logo" | "cover", file: File) => {
   247→    if (file.size > 600 * 1024) {
   248→      toast({ title: "Dosya 600KB sınırı aşılıyor", description: "Daha küçük bir görsel seçin — arşive gömme tavanı 600KB'dir.", variant: "destructive" });
   249→      return;
   250→    }
   251→    const reader = new FileReader();
   252→    reader.onload = () => {
   253→      const dataUrl = typeof reader.result === "string" ? reader.result : "";
   254→      if (!dataUrl) return;
   255→      setHotelForm((prev) => ({ ...prev, [kind === "logo" ? "logoUrl" : "imageUrl"]: dataUrl })); // anında önizleme
   256→      if (hotelDialog?.target) {
   257→        // düzenleme modu: hemen yükle ve kayda yaz
   258→        setHotelMediaBusy(kind);
   259→        uploadHotelMedia(hotelDialog.target.id, kind, dataUrl)
   260→          .then(() => {
   261→            toast({ title: kind === "logo" ? "Logo yüklendi" : "Kapak görseli yüklendi", description: "Medya Arşivi → Otel Görselleri klasörüne benzersiz adla kaydedildi." });
   262→            reload(); bump();
   263→          })
   264→          .catch((e) => toast({ title: "Görsel yüklenemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" }))
   265→          .finally(() => setHotelMediaBusy(null));
   266→      } else {
   267→        // yeni otel: POST id döndükten sonra yüklenecek (saveHotel içinde)
   268→        setHotelPending((prev) => ({ ...prev, [kind]: { dataUrl } }));
   269→      }
   270→    };
   271→    reader.readAsDataURL(file);
   272→  };
   273→
   274→  const saveHotel = async () => {
   275→    if (!currentEditionId || !hotelForm.name.trim()) return;
   276→    setBusy(true);
   277→    try {
   278→      // yalnızca skaler alanlar — registry include'u roomTypes/blocks döndürür, asla gönderilmez
   279→      const scalars = {
   280→        name: hotelForm.name.trim(),
   281→        city: hotelForm.city.trim(),
   282→        district: hotelForm.district.trim(),
   283→        address: hotelForm.address.trim(),
   284→        starRating: hotelForm.starRating === "__none__" ? null : Number(hotelForm.starRating),
   285→        checkInNote: hotelForm.checkInNote.trim(),
   286→        contactName: hotelForm.contactName.trim(),
   287→        contactPhone: hotelForm.contactPhone.trim(),
   288→        email: hotelForm.email.trim(),
   289→        website: hotelForm.website.trim(),
   290→        notes: hotelForm.notes.trim(),
   291→      };
   292→      let hotelId = hotelDialog?.target?.id ?? null;
   293→      if (hotelId) {
   294→        // düzenleme: görsel alanlarını da taşı ("" → null ile kaldırma desteklenir)
   295→        await apiSend(`/api/hotels/${hotelId}`, "PUT", { ...scalars, logoUrl: hotelForm.logoUrl || null, imageUrl: hotelForm.imageUrl || null });
   296→      } else {
   297→        const created = await apiSend<{ id: string }>("/api/hotels", "POST", { editionId: currentEditionId, ...scalars });
   298→        hotelId = created.id;
   299→      }
   300→      // bekleyen logo/kapak — yeni otelde id döndükten sonra yüklenir
   301→      for (const kind of ["logo", "cover"] as const) {
   302→        const pm = hotelPending[kind];
   303→        if (pm?.dataUrl) await uploadHotelMedia(hotelId, kind, pm.dataUrl);
   304→      }
   305→      toast({
   306→        title: hotelDialog?.target ? "Otel güncellendi" : "Otel eklendi",
   307→        description: hotelDialog?.target
   308→          ? `${scalars.name} — iletişim ve kontrat ayrıntıları kaydedildi.`
   309→          : `${scalars.name} · logo/kapak görselleri Medya Arşivi → Otel Görselleri klasörüne bağlandı.`,
   310→      });
   311→      setHotelDialog(null);
   312→      reload(); bump();
   313→    } catch (e) {
   314→      toast({ title: "Otel kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   315→    } finally { setBusy(false); }
   316→  };
   317→
   318→  // diyaloğa gömülü medya seçici (logo / kapak)
   319→  const renderHotelMediaPicker = (kind: "logo" | "cover") => {
   320→    const isLogo = kind === "logo";
   321→    const preview = isLogo ? hotelForm.logoUrl : hotelForm.imageUrl;
   322→    const inputId = isLogo ? "hotel-logo-input" : "hotel-cover-input";
   323→    return (
   324→      <div className="space-y-2">
   325→        <p className="flex items-center gap-1.5 text-xs font-semibold"><Icons.Image className="size-3.5 text-muted-foreground" /> {isLogo ? "Otel logosu" : "Kapak görseli"}</p>
   326→        <div className="flex items-center gap-2.5">
   327→          <span className={cn("grid size-12 shrink-0 place-items-center overflow-hidden rounded-lg border", preview ? "bg-white" : "bg-muted")}>
   328→            {preview ? <img src={preview} alt={isLogo ? "Logo önizleme" : "Kapak önizleme"} className={cn("size-full", isLogo ? "object-contain p-1" : "object-cover")} /> : <Icons.Building2 className="size-5 text-muted-foreground/50" />}
   329→          </span>
   330→          <div className="min-w-0 space-y-1">
   331→            <label htmlFor={inputId} className="inline-flex h-7 cursor-pointer items-center gap-1 rounded-md border bg-card px-2 text-[11px] font-medium transition-colors hover:bg-muted">
   332→              {hotelMediaBusy === kind ? <Icons.Loader2 className="size-3 animate-spin" /> : <Icons.Upload className="size-3" />}
   333→              {preview ? "Değiştir" : "Görsel seç"}
   334→            </label>
   335→            {preview && (
   336→              <button type="button" className="block text-[10px] text-rose-600 hover:underline" onClick={() => { setHotelForm((p) => ({ ...p, [isLogo ? "logoUrl" : "imageUrl"]: "" })); setHotelPending((p) => ({ ...p, [kind]: undefined })); }}>
   337→                Kaldır
   338→              </button>
   339→            )}
   340→          </div>
   341→        </div>
   342→        <Input
   343→          id={inputId}
   344→          type="file"
   345→          accept="image/*"
   346→          className="hidden"
   347→          aria-label={isLogo ? "Otel logosu yükle" : "Kapak görseli yükle"}
   348→          onChange={(e) => { const f = e.target.files?.[0]; if (f) onHotelMediaPick(kind, f); e.target.value = ""; }}
   349→        />
   350→      </div>
   351→    );
   352→  };
   353→
   354→  return (
   355→    <div className="space-y-5">
   356→      <PageHeader title="Konaklama & Seyahat" desc="Oda stoğu gün bazlıdır — 1 oda × 3 gece = 3 oda-gece; bir gece eksikse teyit engellenir" />
   357→
