   111→// JSON dizi alanını güvenle çöz (spamReasons, çoklu seçim yanıtları)
   112→function parseJsonArray(v?: string | null): string[] {
   113→  if (!v) return [];
   114→  try {
   115→    const p: unknown = JSON.parse(v);
   116→    if (Array.isArray(p)) return p.map(String);
   117→  } catch {
   118→    /* JSON değilse virgülle ayrılmış */
   119→  }
   120→  return v.split(",").map((s) => s.trim()).filter(Boolean);
   121→}
   122→// apiSend PATCH metotunu kapsamıyor → gönderi inceleme aksiyonları için yerel yardımcı (göreli yol)
   123→async function apiPatch<T>(path: string, body: unknown): Promise<T> {
   124→  const res = await fetch(path, {
   125→    method: "PATCH",
   126→    headers: { "Content-Type": "application/json" },
   127→    body: JSON.stringify(body),
   128→  });
   129→  const data = await res.json().catch(() => ({}));
   130→  if (!res.ok) throw new Error((data as { error?: string }).error ?? t("forms.actionFailed"));
   131→  return data as T;
   132→}
