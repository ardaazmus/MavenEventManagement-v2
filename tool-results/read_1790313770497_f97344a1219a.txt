   111→   111→// JSON dizi alanını güvenle çöz (spamReasons, çoklu seçim yanıtları)
   112→   112→function parseJsonArray(v?: string | null): string[] {
   113→   113→  if (!v) return [];
   114→   114→  try {
   115→   115→    const p: unknown = JSON.parse(v);
   116→   116→    if (Array.isArray(p)) return p.map(String);
   117→   117→  } catch {
   118→   118→    /* JSON değilse virgülle ayrılmış */
   119→   119→  }
   120→   120→  return v.split(",").map((s) => s.trim()).filter(Boolean);
   121→   121→}

   123→   123→async function apiPatch<T>(path: string, body: unknown): Promise<T> {
   124→   124→  const res = await fetch(path, {
   125→   125→    method: "PATCH",
   126→   126→    headers: { "Content-Type": "application/json" },
   127→   127→    body: JSON.stringify(body),
   128→   128→  });
   129→   129→  const data = await res.json().catch(() => ({}));
   130→   130→  if (!res.ok) throw new Error((data as { error?: string }).error ?? t("forms.actionFailed"));
   131→   131→  return data as T;
   132→   132→}
