    69→function SignaturePad({ value, onChange }: { value: string; onChange: (v: string) => void }) {
    70→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    71→  const drawing = useRef(false);
    72→
    73→  const pos = (e: React.PointerEvent<HTMLCanvasElement>) => {
    74→    const cv = canvasRef.current;
    75→    if (!cv) return { x: 0, y: 0 };
    76→    const r = cv.getBoundingClientRect();
    77→    return { x: ((e.clientX - r.left) / r.width) * cv.width, y: ((e.clientY - r.top) / r.height) * cv.height };
    78→  };
    79→  const start = (e: React.PointerEvent<HTMLCanvasElement>) => {
    80→    e.preventDefault();
    81→    const ctx = canvasRef.current?.getContext("2d");
    82→    if (!ctx) return;
    83→    drawing.current = true;
    84→    const p = pos(e);
    85→    ctx.beginPath();
    86→    ctx.moveTo(p.x, p.y);
    87→  };
    88→  const move = (e: React.PointerEvent<HTMLCanvasElement>) => {
    89→    if (!drawing.current) return;
    90→    e.preventDefault();
    91→    const ctx = canvasRef.current?.getContext("2d");
    92→    if (!ctx) return;
    93→    const p = pos(e);
    94→    ctx.strokeStyle = "#0f766e";
    95→    ctx.lineWidth = 2.2;
    96→    ctx.lineCap = "round";
    97→    ctx.lineTo(p.x, p.y);
    98→    ctx.stroke();
    99→  };
   100→  const end = () => {
   101→    if (!drawing.current) return;
   102→    drawing.current = false;
   103→    const cv = canvasRef.current;
   104→    if (cv) onChange(cv.toDataURL("image/png"));
   105→  };
   106→  const clear = () => {
   107→    const cv = canvasRef.current;
   108→    const ctx = cv?.getContext("2d");
   109→    if (cv && ctx) ctx.clearRect(0, 0, cv.width, cv.height);
   110→    onChange("");
   111→  };
   112→
   113→  return (
   114→    <div className="grid gap-1.5">
   115→      <canvas
   116→        ref={canvasRef}
   117→        width={560}
   118→        height={150}
   119→        role="img"
   120→        aria-label={t("forms.signatureAria")}
   121→        className="h-36 w-full cursor-crosshair touch-none rounded-lg border border-dashed bg-white"
   122→        onPointerDown={start}
   123→        onPointerMove={move}
   124→        onPointerUp={end}
   125→        onPointerLeave={end}
   126→      />
   127→      <div className="flex items-center justify-between">
   128→        <p className="text-[11px] text-muted-foreground">{t("forms.signatureHint")}</p>
   129→        {value && (
   130→          <Button type="button" size="sm" variant="ghost" className="h-7 text-xs" onClick={clear}>
   131→            <RotateCcw className="size-3" /> {t("forms.signatureClear")}
   132→          </Button>
   133→        )}
   134→      </div>
   135→    </div>
   136→  );
   137→}

   141→function RankingInput({ opts, value, onChange }: { opts: string[]; value: string; onChange: (v: string) => void }) {
   142→  const [order, setOrder] = useState<string[]>(() => {
   143→    const saved = value ? (parseMaybeJson(value) as unknown) : null;
   144→    if (Array.isArray(saved) && (saved as string[]).length === opts.length && opts.every((o) => (saved as string[]).includes(o))) {
   145→      return saved as string[];
   146→    }
   147→    return [...opts];
   148→  });
   149→  useEffect(() => {
   150→    onChange(JSON.stringify(order));
   151→    // yalnız sıralama değişince yaz
   152→  }, [order]);
   153→
   154→  const move = (i: number, dir: -1 | 1) => {
   155→    setOrder((o) => {
   156→      const next = [...o];
   157→      const j = i + dir;
   158→      if (j < 0 || j >= next.length) return o;
   159→      [next[i], next[j]] = [next[j], next[i]];
   160→      return next;
   161→    });
   162→  };
   163→
   164→  return (
   165→    <div className="grid gap-1">
   166→      {order.map((opt, i) => (
   167→        <div key={opt} className="flex items-center gap-2 rounded-md border bg-muted/30 px-2.5 py-1.5 text-sm">
   168→          <span className="grid size-5 shrink-0 place-items-center rounded-full bg-teal-100 text-[11px] font-bold text-teal-800">
   169→            {i + 1}
   170→          </span>
   171→          <span className="min-w-0 flex-1 truncate">{opt}</span>
   172→          <span className="flex shrink-0">
   173→            <button type="button" aria-label={`${opt}: ${t("forms.moveUp")}`} className="rounded px-1.5 py-0.5 hover:bg-accent" onClick={() => move(i, -1)} disabled={i === 0}>▲</button>
   174→            <button type="button" aria-label={`${opt}: ${t("forms.moveDown")}`} className="rounded px-1.5 py-0.5 hover:bg-accent" onClick={() => move(i, 1)} disabled={i === order.length - 1}>▼</button>
   175→          </span>
   176→        </div>
   177→      ))}
   178→    </div>
   179→  );
   180→}

   184→export function PublicFormPage({ idOrSlug, embed = false }: { idOrSlug: string; embed?: boolean }) {
   185→  useLang();
   186→  const [form, setForm] = useState<PublicForm | null>(null);
   187→  const [loadError, setLoadError] = useState<string | null>(null);
   188→  const [loading, setLoading] = useState(true);
   189→  const [ans, setAns] = useState<Record<string, string>>({});
   190→  const [visitor, setVisitor] = useState({ name: "", email: "", phone: "", organization: "" });
   191→  const [honeypot, setHoneypot] = useState("");
   192→  const [captchaAnswer, setCaptchaAnswer] = useState("");
   193→  const [commsOptIn, setCommsOptIn] = useState(false);
   194→  const [challenge, setChallenge] = useState<{ question: string; token: string } | null>(null);
   195→  const [payMethod, setPayMethod] = useState("ONLINE_CARD");
   196→  const [pay, setPay] = useState({ cardHolder: "", cardNumber: "", expiry: "", cvc: "" });
   197→  const [payBusy, setPayBusy] = useState(false);
   198→  const [payOutcome, setPayOutcome] = useState<{ outcome: string; message: string; payment?: { id: string; status: string; reference?: string | null } } | null>(null);
   199→  const [busy, setBusy] = useState(false);
   200→  const [result, setResult] = useState<SubmitResult | null>(null);
   201→  const [results, setResults] = useState<PublicResults | null>(null);
   202→  const [error, setError] = useState<string | null>(null);
   203→  const startRef = useRef<number>(Date.now());
   204→
   205→  const load = async () => {
   206→    setLoading(true);
   207→    setLoadError(null);
   208→    try {
   209→      const res = await fetch(`/api/public-forms/${encodeURIComponent(idOrSlug)}`);
   210→      const data = (await res.json()) as PublicForm & { error?: string };
   211→      if (!res.ok) throw new Error(data.error ?? t("forms.publicNotFound"));
   212→      setForm(data);
   213→      setChallenge(data.challenge ?? null);
   214→      startRef.current = Date.now();
   215→    } catch (e) {
   216→      setLoadError(e instanceof Error ? e.message : t("forms.publicNotFound"));
   217→    } finally {
   218→      setLoading(false);
   219→    }
   220→  };
   221→  useEffect(() => {
   222→    load();
   223→  }, [idOrSlug]);
   224→
   225→  const labelIndex = useMemo(
   226→    () => new Map((form?.fields ?? []).map((f) => [f.label, f.id])),
   227→    [form],
   228→  );
   229→  const visible = useMemo(() => {
   230→    if (!form) return new Set<string>();
   231→    const s = new Set<string>();
   232→    for (const f of form.fields) {
   233→      if (isFieldVisible(f, ans, labelIndex)) s.add(f.id);
   234→    }
   235→    return s;
   236→  }, [form, ans, labelIndex]);
   237→
   238→  const set = (fid: string, v: string) => setAns((a) => ({ ...a, [fid]: v }));
   239→  const toggleIn = (fid: string, opt: string, on: boolean) =>
   240→    setAns((a) => {
   241→      const cur: string[] = a[fid] ? (Array.isArray(parseMaybeJson(a[fid])) ? (parseMaybeJson(a[fid]) as string[]) : []) : [];
   242→      return { ...a, [fid]: JSON.stringify(on ? [...cur, opt] : cur.filter((x) => x !== opt)) };
   243→    });
   244→
   245→  const submit = async () => {
   246→    if (!form) return;
   247→    setError(null);
   248→    if (!visitor.email.trim()) {
   249→      setError(t("forms.publicEmailRequired"));
   250→      return;
   251→    }
   252→    // görünür zorunlu alan denetimi (istemci tarafı — sunucu da denetler)
   253→    const missing = form.fields.filter(
   254→      (f) => f.required === "ALWAYS" && f.type !== "SECTION" && visible.has(f.id) &&
   255→        !(ans[f.id] ?? "").trim() && isFieldVisible(f, ans, labelIndex),
   256→    );
   257→    if (missing.length > 0) {
   258→      setError(`${t("forms.publicMissingFields")}: ${missing.map((f) => f.label).join(", ")}`);
   259→      return;
   260→    }
   261→    if (form.captchaEnabled && !captchaAnswer.trim()) {
   262→      setError(t("forms.publicCaptchaRequired"));
   263→      return;
   264→    }
   265→    setBusy(true);
   266→    try {
   267→      const res = await apiSend<SubmitResult>("/api/public-register", "POST", {
   268→        formId: idOrSlug, // slug veya id — sunucu ikisini de çözer
   269→        respondentName: visitor.name,
   270→        respondentEmail: visitor.email,
   271→        phone: visitor.phone || undefined,
   272→        organization: visitor.organization || undefined,
   273→        answers: ans,
   274→        honeypotValue: honeypot,
   275→        elapsedSeconds: Math.round((Date.now() - startRef.current) / 1000),
   276→        source: "WEB_PUBLIC",
   277→        paymentMethod: payMethod,
   278→        challengeToken: challenge?.token,
   279→        challengeAnswer: captchaAnswer,
   280→        commsOptIn,
   281→      });
   282→      setResult(res);
   283→      if (form.hasPublicResults && res.status !== "SPAM") {
   284→        fetch(`/api/public-forms/${encodeURIComponent(idOrSlug)}/results`)
   285→          .then((r) => (r.ok ? r.json() : null))
   286→          .then((d: PublicResults | null) => d && setResults(d))
   287→          .catch(() => undefined);
   288→      }
   289→    } catch (e) {
   290→      setError(e instanceof Error ? e.message : t("forms.toastSubmitError"));
   291→    } finally {
   292→      setBusy(false);
   293→    }
   294→  };
   295→
   296→  // Kayıt formu + online ödeme: gönderim başarılıysa ödeme simülasyonu (sandbox)
   297→  const processPayment = async () => {
   298→    if (!result?.payment) return;
   299→    setPayBusy(true);
   300→    try {
   301→      const res = await apiSend<{ outcome: string; message: string; payment?: { id: string; status: string; reference?: string | null } }>(
   302→        `/api/payments/${result.payment.id}/process`, "POST",
   303→        { cardHolder: pay.cardHolder, cardNumber: pay.cardNumber, expiry: pay.expiry, cvc: pay.cvc },
   304→      );
   305→      setPayOutcome(res);
   306→    } catch (e) {
   307→      setError(e instanceof Error ? e.message : t("forms.toastPayError"));
   308→    } finally {
   309→      setPayBusy(false);
   310→    }
   311→  };
   312→
   313→  const again = () => {
   314→    setResult(null);
   315→    setResults(null);
   316→    setPayOutcome(null);
   317→    setPay({ cardHolder: "", cardNumber: "", expiry: "", cvc: "" });
   318→    setPayMethod("ONLINE_CARD");
   319→    setAns({});
   320→    setVisitor({ name: "", email: "", phone: "", organization: "" });
   321→    setCaptchaAnswer("");
   322→    setHoneypot("");
   323→    startRef.current = Date.now();
   324→    if (form?.captchaEnabled) load(); // tek-kullanım challenge tazele
   325→  };
   326→
   327→  // ─── yüklenme / hata ekranları ───
   328→  if (loading) {
   329→    return (
   330→      <div className={embed ? "grid min-h-[240px] place-items-center" : "grid min-h-screen place-items-center bg-background"} role="status" aria-label={t("common.loading")}>
   331→        <Loader2 className="size-7 animate-spin text-teal-600" />
   332→      </div>
   333→    );
   334→  }
   335→  if (loadError || !form) {
   336→    return (
   337→      <div className={embed ? "p-6" : "grid min-h-screen place-items-center bg-background p-6"}>
   338→        <div className="max-w-sm rounded-xl border bg-card p-6 text-center shadow-sm">
   339→          <AlertTriangle className="mx-auto size-8 text-amber-500" />
   340→          <h1 className="mt-3 text-lg font-semibold">{t("forms.publicNotFoundTitle")}</h1>
   341→          <p className="mt-1 text-sm text-muted-foreground">{loadError ?? t("forms.publicNotFound")}</p>
   342→        </div>
   343→      </div>
   344→    );
   345→  }
   346→
   347→  const wrap = (children: ReactNode) =>
   348→    embed ? <div className="mx-auto w-full max-w-2xl p-3">{children}</div> : (
   349→      <div className="min-h-screen bg-gradient-to-b from-teal-50/60 via-background to-background px-4 py-8">
   350→        <div className="mx-auto w-full max-w-2xl">{children}</div>
   351→      </div>
   352→    );
   353→
   354→  // ─── sonuç ekranı ───
   355→  if (result) {
   356→    const spam = result.status === "SPAM";
   357→    return wrap(
   358→      <div className="space-y-4">
   359→        <div className={`rounded-xl border p-6 text-center ${spam ? "border-rose-200 bg-rose-50" : result.status === "APPROVED" ? "border-emerald-200 bg-emerald-50" : "border-amber-200 bg-amber-50"}`}>
   360→          {spam ? <AlertTriangle className="mx-auto size-10 text-rose-500" /> : <CheckCircle2 className="mx-auto size-10 text-emerald-600" />}
   361→          <h1 className="mt-3 text-xl font-semibold">
   362→            {spam ? t("forms.toastSpamFlagged") : result.status === "APPROVED" ? t("forms.publicApproved") : t("forms.publicReceived")}
   363→          </h1>
   364→          {!spam && form.successMessage && <p className="mt-2 text-sm text-muted-foreground">{form.successMessage}</p>}
   365→          {result.registration?.confirmationNo && (
   366→            <p className="mt-3 inline-flex items-center gap-2 rounded-lg border bg-card px-3 py-1.5 text-sm">
   367→              {t("forms.publicRegNo")}: <span className="font-mono font-semibold">{result.registration.confirmationNo}</span>
   368→            </p>
   369→          )}
   370→          {result.quizScore != null && (
   371→            <p className="mt-3 text-sm font-medium text-teal-800">
   372→              {t("forms.quizScoreLine", { score: Math.round(result.quizScore), correct: result.quizCorrect ?? 0, total: result.quizTotal ?? 0 })}
   373→            </p>
   374→          )}
   375→        </div>
   376→
   377→        {/* Online kart ödemesi (simülasyon) — ONLINE_CARD seçildiyse */}
   378→        {!spam && payMethod === "ONLINE_CARD" && result.payment && result.payment.status !== "SUCCEEDED" && (
   379→          <div className="rounded-xl border bg-card p-5 shadow-sm">
   380→            <h2 className="flex items-center gap-2 text-sm font-semibold">{t("forms.payCardTitle")}</h2>
   381→            <div className="mt-3 grid gap-2 sm:grid-cols-2">
   382→              <div className="grid gap-1 sm:col-span-2">
   383→                <Label className="text-xs">{t("forms.payCardHolder")}</Label>
   384→                <Input value={pay.cardHolder} onChange={(e) => setPay({ ...pay, cardHolder: e.target.value })} autoComplete="cc-name" />
   385→              </div>
   386→              <div className="grid gap-1 sm:col-span-2">
   387→                <Label className="text-xs">{t("forms.payCardNumber")}</Label>
   388→                <Input value={pay.cardNumber} placeholder="4242 4242 4242 4242" onChange={(e) => setPay({ ...pay, cardNumber: e.target.value })} autoComplete="cc-number" inputMode="numeric" />
   389→              </div>
   390→              <div className="grid gap-1">
   391→                <Label className="text-xs">{t("forms.payExpiry")}</Label>
   392→                <Input placeholder="AA/YY" value={pay.expiry} onChange={(e) => setPay({ ...pay, expiry: e.target.value })} autoComplete="cc-exp" />
   393→              </div>
   394→              <div className="grid gap-1">
   395→                <Label className="text-xs">CVC</Label>

   397→              </div>
   398→            </div>
   399→            <div className="mt-3 flex flex-wrap items-center gap-2">

   401→                {t("forms.payComplete")}

   404→            </div>
   405→            {payOutcome?.outcome === "SUCCEEDED" && (
   406→              <p className="mt-3 rounded-lg border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-700">
   407→                {t("forms.payOkLine", { ref: payOutcome.payment?.reference ?? "—" })}
   408→              </p>
   409→            )}
   410→            {payOutcome && payOutcome.outcome !== "SUCCEEDED" && (
   411→              <p className="mt-3 rounded-lg border border-rose-200 bg-rose-50 p-3 text-sm text-rose-700">
   412→                {payOutcome.message || t("forms.toastPayError")}
   413→              </p>
   414→            )}
   415→          </div>
   416→        )}
   417→        {results && <ResultsPanel results={results} />}
   418→        <div className="text-center">
   419→          <Button variant="outline" onClick={again}>
   420→            <RefreshCw className="size-4" /> {t("forms.publicAgain")}
   421→          </Button>
   422→        </div>
   423→      </div>,
   424→    );
   425→  }
   426→
   427→  // ─── form ekranı ───
   428→  const isReg = form.type === "REGISTRATION";
   429→  return wrap(
   430→    <div className="space-y-4">
   431→      <header className="rounded-xl border bg-card p-5 shadow-sm">
   432→        <div className="flex flex-wrap items-center gap-2">
   433→          <h1 className="text-xl font-bold">{form.name}</h1>
   434→          {isReg && <Badge className="bg-teal-600">{t("forms.badgeRegistration")}</Badge>}
   435→          {(form.type === "SURVEY" || form.type === "FEEDBACK") && <Badge variant="outline">{t("forms.badgeSurvey")}</Badge>}

   437→        {form.description && <p className="mt-1.5 text-sm text-muted-foreground">{form.description}</p>}

   440→      <div className="rounded-xl border bg-card p-5 shadow-sm">

   456→          </div>
   457→
   458→          {/* ziyaretçi kimliği */}
   459→          <div className="grid gap-3 rounded-lg border bg-muted/20 p-3 sm:grid-cols-2">
   460→            <div className="grid gap-1">

   464→            <div className="grid gap-1">
   465→              <Label className="text-xs">{t("forms.publicEmail")}<span className="text-rose-500"> *</span></Label>
   466→              <Input type="email" required value={visitor.email} autoComplete="email" onChange={(e) => setVisitor({ ...visitor, email: e.target.value })} />
   467→            </div>
   468→            {isReg && (
   469→              <>
   470→                <div className="grid gap-1">
   471→                  <Label className="text-xs">{t("forms.publicPhone")}</Label>
   472→                  <Input type="tel" value={visitor.phone} autoComplete="tel" onChange={(e) => setVisitor({ ...visitor, phone: e.target.value })} />
   473→                </div>
   474→                <div className="grid gap-1">
   475→                  <Label className="text-xs">{t("forms.publicOrg")}</Label>
   476→                  <Input value={visitor.organization} autoComplete="organization" onChange={(e) => setVisitor({ ...visitor, organization: e.target.value })} />
   477→                </div>
   478→              </>
   479→            )}
   480→          </div>
   481→
   482→          {/* KVKK rıza */}
   483→          <label className="flex cursor-pointer items-start gap-2 rounded-lg border p-3 text-xs">
   484→            <Checkbox checked={commsOptIn} onCheckedChange={(c) => setCommsOptIn(c === true)} className="mt-0.5" />
   485→            <span className="text-muted-foreground">{t("forms.publicConsent")}</span>
   486→          </label>
   487→
   488→          {/* Ödeme yöntemi — sadece online ödemeli kayıt formu */}
   489→          {isReg && form.enableOnlinePayment && (
   490→            <div className="grid gap-1 rounded-lg border bg-muted/20 p-3">
   491→              <Label className="text-xs">{t("forms.payMethodLabel")}</Label>
   492→              <div className="grid gap-1.5">
   493→                {[
   494→                  { value: "ONLINE_CARD", label: t("forms.payOnlineCard") },
   495→                  { value: "BANK_TRANSFER", label: t("forms.payBankTransfer") },
   496→                  { value: "PAYMENT_LINK", label: t("forms.payLink") },
   497→                ].map((m) => (
   498→                  <label key={m.value} className="flex cursor-pointer items-center gap-2 text-sm">
   499→                    <input type="radio" name="payMethod" checked={payMethod === m.value} onChange={() => setPayMethod(m.value)} className="size-4 accent-teal-700" />
   500→                    {m.label}
   501→                  </label>
   502→                ))}
   503→              </div>
   504→            </div>
   505→          )}
   506→
   507→          {/* insan doğrulaması */}
   508→          {form.captchaEnabled && challenge && (
   509→            <div className="rounded-lg border border-teal-200 bg-teal-50/50 p-3">
   510→              <Label className="flex items-center gap-1.5 text-xs font-medium text-teal-900">
   511→                <ShieldCheck className="size-3.5" /> {t("forms.publicCaptchaTitle")}
   512→              </Label>
   513→              <div className="mt-2 flex items-center gap-2">
   514→                <span className="rounded-md border bg-white px-3 py-1.5 font-mono text-sm font-bold tracking-wide text-teal-900" aria-live="polite">
   515→                  {challenge.question} = ?
   516→                </span>
   517→                <Input
   518→                  className="w-24"
   519→                  inputMode="numeric"
   520→                  value={captchaAnswer}
   521→                  placeholder={t("forms.publicCaptchaPh")}
   522→                  aria-label={t("forms.publicCaptchaTitle")}
   523→                  onChange={(e) => setCaptchaAnswer(e.target.value)}
   524→                />
   525→              </div>
   526→              <p className="mt-1 text-[11px] text-muted-foreground">{t("forms.publicCaptchaHint")}</p>
   527→            </div>
   528→          )}
   529→
   530→          {/* honeypot — gizli alan */}
   531→          {form.honeypotEnabled && (
   532→            <div className="absolute -left-[9999px] top-auto h-px w-px overflow-hidden" aria-hidden>
   533→              <Input value={honeypot} name="website" autoComplete="off" tabIndex={-1} onChange={(e) => setHoneypot(e.target.value)} />
   534→            </div>
   535→          )}
   536→
   537→          {error && (
   538→            <p role="alert" className="rounded-lg border border-rose-200 bg-rose-50 p-3 text-sm text-rose-700">{error}</p>
   539→          )}
   540→
   541→          <Button className="w-full" disabled={busy} onClick={submit}>
   542→            {busy ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   543→            {isReg ? t("forms.publicSubmitReg") : t("forms.publicSubmit")}
   544→          </Button>
   545→          <p className="text-center text-[11px] text-muted-foreground">
   546→            <ShieldCheck className="mr-1 inline size-3" />
   547→            {t("forms.publicSpamNote")}
   548→          </p>
   549→        </div>
   550→      </div>
   551→    </div>,
   552→  );
   553→}

   557→function FieldInput({
   558→  f, ans, set, toggleIn,
   559→}: {
   560→  f: PublicField;
   561→  ans: Record<string, string>;
   562→  set: (fid: string, v: string) => void;
   563→  toggleIn: (fid: string, opt: string, on: boolean) => void;
   564→}) {
   565→  const value = ans[f.id] ?? "";
   566→  const reqStar = f.required === "ALWAYS" ? <span className="text-rose-500"> *</span> : null;
   567→  const opts = optsOf(f);
   568→  const cols = colsOf(f);
   569→  const multiSelected: string[] = value ? (Array.isArray(parseMaybeJson(value)) ? (parseMaybeJson(value) as string[]) : []) : [];
   570→
   571→  if (f.type === "SECTION") {
   572→    return (
   573→      <div className="border-b pb-1.5">
   574→        <p className="text-sm font-bold tracking-wide text-teal-900">{f.label}</p>
   575→        {f.helpText && <p className="mt-0.5 text-xs text-muted-foreground">{f.helpText}</p>}
   576→      </div>
   577→    );
   578→  }
   579→
   580→  const body = (() => {
   581→    switch (f.type) {
   582→      case "LONGTEXT":
   583→        return <Textarea rows={3} value={value} placeholder={f.placeholder ?? ""} onChange={(e) => set(f.id, e.target.value)} />;
   584→      case "NUMBER":
   585→        return <Input type="number" value={value} placeholder={f.placeholder ?? ""} onChange={(e) => set(f.id, e.target.value)} />;
   586→      case "EMAIL":
   587→        return <Input type="email" value={value} placeholder={f.placeholder ?? "ad@ornek.com"} onChange={(e) => set(f.id, e.target.value)} />;
   588→      case "PHONE":
   589→        return <Input type="tel" value={value} placeholder={f.placeholder ?? "+90 5xx xxx xx xx"} onChange={(e) => set(f.id, e.target.value)} />;
   590→      case "URL":
   591→        return <Input type="url" value={value} placeholder="https://…" onChange={(e) => set(f.id, e.target.value)} />;
   592→      case "DATE":
   593→        return <Input type="date" value={value} onChange={(e) => set(f.id, e.target.value)} />;
   594→      case "TIME":
   595→        return <Input type="time" value={value} onChange={(e) => set(f.id, e.target.value)} />;
   596→      case "COUNTRY":
   597→        return <Input value={value} placeholder={f.placeholder ?? t("forms.phCountry")} onChange={(e) => set(f.id, e.target.value)} />;
   598→      case "FILE":
   599→        return <Input type="file" onChange={() => undefined} className="text-xs" aria-label={f.label} />;
   600→      case "SINGLE_CHOICE":
   601→      case "QA_QUIZ":
   602→      case "VOTE":
   603→        return (
   604→          <div className={`grid gap-1.5 ${f.type === "VOTE" ? "rounded-lg border p-2.5" : ""}`} role="radiogroup" aria-label={f.label}>
   605→            {opts.map((o) => (
   606→              <label key={o} className="flex cursor-pointer items-center gap-2 text-sm">
   607→                <input
   608→                  type="radio"
   609→                  name={f.id}
   610→                  value={o}
   611→                  checked={value === o}
   612→                  onChange={() => set(f.id, o)}
   613→                  className="size-4 accent-teal-700"
   614→                />
   615→                {o}
   616→              </label>
   617→            ))}
   618→            {opts.length === 0 && <p className="text-xs text-muted-foreground">{t("forms.noOptions")}</p>}
   619→          </div>
   620→        );
   621→      case "MULTI_CHOICE":
   622→        return (
   623→          <div className="grid gap-1.5 rounded-lg border p-2.5">
   624→            {opts.map((o) => (
   625→              <label key={o} className="flex cursor-pointer items-center gap-2 text-sm">
   626→                <Checkbox checked={multiSelected.includes(o)} onCheckedChange={(c) => toggleIn(f.id, o, c === true)} />
   627→                {o}
   628→              </label>
   629→            ))}
   630→            {opts.length === 0 && <p className="text-xs text-muted-foreground">{t("forms.noOptions")}</p>}
   631→          </div>
   632→        );
   633→      case "CHECKBOX":
   634→      case "YESNO":
   635→      case "TERMS":
   636→        return (
   637→          <label className="flex cursor-pointer items-center gap-2 text-sm">
   638→            <Switch checked={value === "true"} onCheckedChange={(c) => set(f.id, c ? "true" : "false")} aria-label={f.label} />
   639→            {f.type === "YESNO" && (
   640→              <span className="text-muted-foreground">
   641→                {value === "true" ? t("forms.yes") : value === "false" ? t("forms.no") : t("forms.chooseHint")}
   642→              </span>
   643→            )}
   644→          </label>
   645→        );
   646→      case "RATING":
   647→        return (
   648→          <div className="flex items-center gap-1">
   649→            {[1, 2, 3, 4, 5].map((n) => (
   650→              <button key={n} type="button" aria-label={t("forms.stars", { n })} aria-pressed={Number(value) >= n}
   651→                onClick={() => set(f.id, String(n))}
   652→                className="rounded-md p-1 transition hover:bg-amber-50">
   653→                <svg viewBox="0 0 24 24" className={`size-6 ${Number(value) >= n ? "fill-amber-400 text-amber-500" : "text-muted-foreground/40"}`} stroke="currentColor" strokeWidth="1.5">
   654→                  <path d="M12 2.5l2.9 5.9 6.5.9-4.7 4.6 1.1 6.5L12 17.3l-5.8 3.1 1.1-6.5L2.6 9.3l6.5-.9L12 2.5z" />
   655→                </svg>
   656→              </button>
   657→            ))}
   658→            {value && <span className="ml-1 text-xs text-muted-foreground">{value}/5</span>}
   659→          </div>
   660→        );
   661→      case "NPS":
   662→        return (
   663→          <div className="grid gap-1">
   664→            <div className="grid grid-cols-11 gap-1">
   665→              {Array.from({ length: 11 }, (_, n) => (
   666→                <button key={n} type="button" aria-pressed={Number(value) === n} aria-label={String(n)}
   667→                  onClick={() => set(f.id, String(n))}
   668→                  className={`rounded-md border py-1.5 text-xs font-medium transition ${Number(value) === n ? "border-teal-600 bg-teal-600 text-white" : "hover:bg-accent"}`}>
   669→                  {n}
   670→                </button>
   671→              ))}
   672→            </div>
   673→            <div className="flex justify-between text-[10px] text-muted-foreground">
   674→              <span>{t("forms.npsLow")}</span><span>{t("forms.npsHigh")}</span>
   675→            </div>
   676→          </div>
   677→        );
   678→      case "RANKING":
   679→        return opts.length > 0 ? <RankingInput opts={opts} value={value} onChange={(v) => set(f.id, v)} /> : <p className="text-xs text-muted-foreground">{t("forms.noOptions")}</p>;
   680→      case "MATRIX":
   681→        return (
   682→          <div className="overflow-x-auto rounded-lg border">
   683→            <table className="w-full min-w-[420px] text-sm">
   684→              <thead>
   685→                <tr className="border-b bg-muted/40">
   686→                  <th className="p-2 text-left text-xs font-medium" />
   687→                  {cols.map((c) => <th key={c} className="p-2 text-center text-xs font-medium">{c}</th>)}
   688→                </tr>
   689→              </thead>
   690→              <tbody>
   691→                {opts.map((r) => {
   692→                  const rowState: Record<string, string> = value ? ((parseMaybeJson(value) as Record<string, string>) ?? {}) : {};
   693→                  return (
   694→                    <tr key={r} className="border-b last:border-0">
   695→                      <td className="p-2 text-xs font-medium">{r}</td>
   696→                      {cols.map((c) => (
   697→                        <td key={c} className="p-2 text-center">
   698→                          <input
   699→                            type="radio"
   700→                            name={`${f.id}-${r}`}
   701→                            aria-label={`${r}: ${c}`}
   702→                            checked={rowState[r] === c}
   703→                            onChange={() => set(f.id, JSON.stringify({ ...rowState, [r]: c }))}
   704→                            className="size-4 accent-teal-700"
   705→                          />
   706→                        </td>
   707→                      ))}
   708→                    </tr>
   709→                  );
   710→                })}
   711→              </tbody>
   712→            </table>
   713→          </div>
   714→        );
   715→      case "SIGNATURE":
   716→        return <SignaturePad value={value} onChange={(v) => set(f.id, v)} />;
   717→      default:
   718→        return <Input value={value} placeholder={f.placeholder ?? ""} onChange={(e) => set(f.id, e.target.value)} />;
   719→    }
   720→  })();
   721→
   722→  return (
   723→    <div className="grid gap-1.5">
   724→      <Label className="text-xs font-medium">
   725→        {f.label}
   726→        {reqStar}
   727→        {f.points != null && f.type === "QA_QUIZ" && (
   728→          <span className="ml-1.5 rounded bg-emerald-100 px-1.5 py-0.5 text-[10px] font-semibold text-emerald-700">{f.points} {t("forms.pts")}</span>
   729→        )}
   730→        {f.mobileInteractive && <Badge variant="outline" className="ml-1.5 text-[10px]">{t("forms.mobileItem")}</Badge>}
   731→      </Label>
   732→      {body}
   733→      {f.helpText && f.type !== "CHECKBOX" && f.type !== "YESNO" && f.type !== "TERMS" && (
   734→        <p className="text-xs text-muted-foreground">{f.helpText}</p>
   735→      )}
   736→      {f.type === "TERMS" && f.helpText && (
   737→        <p className="rounded-md bg-muted/40 p-2.5 text-xs text-muted-foreground">{f.helpText}</p>
   738→      )}
   739→    </div>
   740→  );
   741→}
