   105→function useNow(intervalMs = 30_000) {
   106→  const [now, setNow] = useState(() => Date.now());
   107→  useEffect(() => {
   108→    const t = setInterval(() => setNow(Date.now()), intervalMs);
   109→    return () => clearInterval(t);
   110→  }, [intervalMs]);
   111→  return now;
   112→}

   114→function offerCountdown(expiresAt?: string | null, now = Date.now()) {
   115→  if (!expiresAt) return null;
   116→  const diff = new Date(expiresAt).getTime() - now;
   117→  if (diff <= 0) return { text: "Süresi doldu", expired: true as const, hours: 0 };
   118→  const h = Math.floor(diff / 3_600_000);
   119→  const m = Math.floor((diff % 3_600_000) / 60_000);
   120→  return { text: `${h} sa ${m} dk kaldı`, expired: false as const, hours: h };
   121→}

   123→// portal durumu adımları: Katılım → Kayıt → Onay → Yaka Kartı → Giriş
   124→function buildSteps(p: NonNullable<ParticipantData["participation"]>, regs: ParticipantData["registrations"]) {
   125→  const activeReg = regs.find((r) => r.status !== "CANCELLED") ?? regs[0];
   126→  const badge = p.badges[0];
   127→  const steps: { key: string; title: string; state: "done" | "current" | "pending" | "error"; hint?: string }[] = [
   128→    { key: "part", title: "Katılım", state: "done", hint: label(REG_SOURCES, p.source) },
   129→    activeReg
   130→      ? {
   131→          key: "reg", title: t("portal.stepReg"),
   132→          state: ["REJECTED", "CANCELLED"].includes(activeReg.status) ? "error" : "done",
   133→          hint: activeReg.confirmationNo,
   134→        }
   135→      : { key: "reg", title: t("portal.stepReg"), state: "pending", hint: "Henüz kayıt yok" },
   136→    activeReg && ["CONFIRMED"].includes(activeReg.status)
   137→      ? { key: "appr", title: "Onay", state: "done", hint: activeReg.decidedAt ? fmtDate(activeReg.decidedAt) : undefined }
   138→      : activeReg && ["SUBMITTED", "PENDING_APPROVAL"].includes(activeReg.status)
   139→        ? { key: "appr", title: "Onay", state: "current", hint: "İncelemede" }
   140→        : { key: "appr", title: "Onay", state: activeReg ? "pending" : "pending" },
   141→    badge
   142→      ? { key: "badge", title: "Yaka Kartı", state: ["READY", "PRINTED", "REPRINTED", "ISSUED"].includes(badge.status) ? "done" : "current", hint: badge.badgeNo }
   143→      : { key: "badge", title: "Yaka Kartı", state: "pending" },
   144→    p.attendance === "CHECKED_IN" || p.attendance === "CHECKED_OUT"
   145→      ? { key: "in", title: "Giriş", state: "done", hint: label(ATTENDANCE_STATUS, p.attendance) }
   146→      : { key: "in", title: "Giriş", state: "pending", hint: label(ATTENDANCE_STATUS, p.attendance) },
   147→  ];
   148→  return steps;
   149→}

   256→function OrderBlock({ orders, onPayLink, busyId }: { orders: PortalOrder[]; onPayLink: (o: PortalOrder) => void; busyId?: string | null }) {
   257→  if (orders.length === 0) return <p className="text-xs text-muted-foreground">Sipariş bulunmuyor.</p>;
   258→  return (
   259→    <div className="space-y-3">
   260→      {orders.map((o) => (
   261→        <div key={o.id} className="rounded-lg border bg-background/60 p-3 transition-colors hover:bg-background">
   262→          <div className="flex flex-wrap items-center justify-between gap-2">
   263→            <div className="flex items-center gap-2">
   264→              <span className="font-mono text-xs font-semibold text-foreground">{o.orderNo}</span>
   265→              <StatusBadge map={ORDER_STATUS as unknown as Record<string, string>} value={o.status} />
   266→            </div>
   267→            <span className="text-sm font-semibold tabular-nums">{fmtMoney(o.totalAmount, o.currency)}</span>
   268→          </div>
   269→          {o.lines.length > 0 && (
   270→            <ul className="mt-2 space-y-0.5 text-xs text-muted-foreground">
   271→              {o.lines.map((l) => (
   272→                <li key={l.id} className="flex justify-between gap-2">
   273→                  <span className="truncate">{l.quantity}× {l.description}{l.personName ? ` — ${l.personName}` : ""}</span>
   274→                  <span className="tabular-nums">{fmtMoney(l.total, o.currency)}</span>
   275→                </li>
   276→              ))}
   277→            </ul>
   278→          )}
   279→          <div className="mt-2 flex flex-wrap items-center justify-between gap-2 border-t border-dashed pt-2">
   280→            <div className="flex flex-wrap gap-1.5">
   281→              {o.payments.slice(0, 3).map((p) => {
   282→                const I = PAY_SOURCE_ICON[p.source] ?? Icons.CreditCard;
   283→                return (
   284→                  <span key={p.id} className={cn(
   285→                    "inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[10px]",
   286→                    p.status === "SUCCEEDED" && "border-emerald-200 bg-emerald-50 text-emerald-700",
   287→                    p.status === "PENDING" && "border-amber-200 bg-amber-50 text-amber-700",
   288→                    p.status === "FAILED" && "border-rose-200 bg-rose-50 text-rose-700",
   289→                  )}>
   290→                    <I className="size-3" aria-hidden /> {label(PAYMENT_METHODS, p.source)} · {fmtMoney(p.amount, o.currency)}
   291→                  </span>
   292→                );
   293→              })}
   294→            </div>
   295→            <div className="flex items-center gap-2">
   296→              <span className={cn("text-xs font-medium tabular-nums", o.remaining > 0 ? "text-amber-600" : "text-emerald-600")}>
   297→                {o.remaining > 0 ? `Kalan: ${fmtMoney(o.remaining, o.currency)}` : "Bakiye kapalı"}
   298→              </span>
   299→              {o.remaining > 0 && (
   300→                <Button size="sm" variant="outline" className="h-7 gap-1.5 text-xs" disabled={busyId === o.id} onClick={() => onPayLink(o)}>
   301→                  {busyId === o.id ? <Icons.Loader2 className="size-3.5 animate-spin" /> : <Icons.Link2 className="size-3.5" />} Ödeme bağlantısı
   302→                </Button>
   303→              )}
   304→            </div>
   305→          </div>
   306→        </div>
   307→      ))}
   308→    </div>
   309→  );
   310→}

   324→// payloadJson güvenli ayrıştırma — dize/nesne/dizi/sahte biçimlerde boş nesneye düşer
   325→function parseBlockPayload(raw?: string | null): BlockPayload {
   326→  if (!raw) return {};
   327→  try {
   328→    const p = JSON.parse(raw) as unknown;
   329→    return p && typeof p === "object" && !Array.isArray(p) ? (p as BlockPayload) : {};
   330→  } catch {
   331→    return {};
   332→  }
   333→}
