    69→    69→function readImage(file: File, maxBytes: number): Promise<string> {
    70→    70→  return new Promise((resolve, reject) => {
    71→    71→    if (file.size > maxBytes) {
    72→    72→      reject(new RangeError("IMG_TOO_LARGE"));
    73→    73→      return;
    74→    74→    }
    75→    75→    const r = new FileReader();
    76→    76→    r.onload = () => resolve(String(r.result));
    77→    77→    r.onerror = () => reject(new RangeError("IMG_READ_FAILED"));
    78→    78→    r.readAsDataURL(file);
    79→    79→  });
    80→    80→}

    82→    82→function parseWidgetsJson(raw: string | null): WidgetRow[] {
    83→    83→  const defaults: WidgetRow[] = WIDGET_ORDER.map((w, i) => ({ key: w.key, enabled: true, visibility: w.key === "b2b" ? "AUTH" : "ALL", order: i }));
    84→    84→  if (!raw) return defaults;
    85→    85→  try {
    86→    86→    const p = JSON.parse(raw) as Record<string, { enabled?: boolean; visibility?: string; order?: number }>;
    87→    87→    return WIDGET_ORDER.map((w, i) => {
    88→    88→      const o = p[w.key];
    89→    89→      if (!o) return defaults[i];
    90→    90→      return {
    91→    91→        key: w.key,
    92→    92→        enabled: o.enabled === undefined ? true : Boolean(o.enabled),
    93→    93→        visibility: w.key === "b2b" ? "AUTH" : o.visibility === "AUTH" ? "AUTH" : "ALL",
    94→    94→        order: Number.isFinite(o.order) ? Number(o.order) : i,
    95→    95→      };
    96→    96→    });
    97→    97→  } catch {
    98→    98→    return defaults;
    99→    99→  }
   100→   100→}

   102→   102→function parseJsonObj<T>(raw: string | null, fallback: T): T {
   103→   103→  if (!raw) return fallback;
   104→   104→  try {
   105→   105→    const p = JSON.parse(raw) as unknown;
   106→   106→    return p && typeof p === "object" && !Array.isArray(p) ? (p as T) : fallback;
   107→   107→  } catch {
   108→   108→    return fallback;
   109→   109→  }
   110→   110→}
   111→   111→function parseJsonArr(raw: string | null): string[] {
   112→   112→  if (!raw) return [];
   113→   113→  try {
   114→   114→    const p = JSON.parse(raw) as unknown;
   115→   115→    return Array.isArray(p) ? p.filter((x): x is string => typeof x === "string") : [];
   116→   116→  } catch {
   117→   117→    return [];
   118→   118→  }
   119→   119→}
