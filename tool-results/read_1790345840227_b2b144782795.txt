   120→// portal POST yardımcısı — oturum başlığı taşır (apiSend header kabul etmediği için)
   121→async function portalSend(path: string, body: unknown, sessionKey: string | null): Promise<unknown> {
   122→  const res = await fetch(path, {
   123→    method: "POST",
   124→    headers: { "Content-Type": "application/json", ...(sessionKey ? { "x-portal-session": sessionKey } : {}) },
   125→    body: JSON.stringify(body),
   126→  });
   127→  const data = await res.json().catch(() => ({}));
   128→  if (!res.ok) throw new Error((data as { error?: string }).error ?? `İşlem başarısız (${res.status})`);
   129→  return data;
   130→}

   132→function loadStoredSession(slug: string): string | null {
   133→  try {
   134→    return localStorage.getItem(sessionKeyStorage(slug));
   135→  } catch {
   136→    return null;
   137→  }
   138→}
   139→function storeSession(slug: string, key: string) {
   140→  try {
   141→    localStorage.setItem(sessionKeyStorage(slug), key);
   142→  } catch {
   143→    /* yoksay */
   144→  }
   145→}
   146→function clearSession(slug: string) {
   147→  try {
   148→    localStorage.removeItem(sessionKeyStorage(slug));
   149→  } catch {
   150→    /* yoksay */
   151→  }
   152→}
   153→type MarkedReminder = { key: string; fireAt: number; title: string; body: string };
   154→function loadFired(slug: string): Set<string> {
   155→  try {
   156→    return new Set(JSON.parse(localStorage.getItem(firedStorage(slug)) ?? "[]") as string[]);
   157→  } catch {
   158→    return new Set();
   159→  }
   160→}
   161→function saveFired(slug: string, fired: Set<string>) {
   162→  try {
   163→    // faz büyümesini önle — son 200 kayıt yeter
   164→    localStorage.setItem(firedStorage(slug), JSON.stringify([...fired].slice(-200)));
   165→  } catch {
   166→    /* yoksay */
   167→  }
   168→}
   169→function loadReminders(slug: string): MarkedReminder[] {
   170→  try {
   171→    return JSON.parse(localStorage.getItem(remindersStorage(slug)) ?? "[]") as MarkedReminder[];
   172→  } catch {
   173→    return [];
   174→  }
   175→}
   176→function saveReminders(slug: string, items: MarkedReminder[]) {
   177→  try {
   178→    localStorage.setItem(remindersStorage(slug), JSON.stringify(items));
   179→  } catch {
   180→    /* yoksay */
   181→  }
   182→}

   194→// ─── ana bileşen ────────────────────────────────────────────────────────────
   195→export function PortalApp({ editionSlug, magicToken }: { editionSlug: string; magicToken?: string }) {
   196→  const { t } = useLang();
   197→  const { toast } = useToast();
   198→  const [phase, setPhase] = useState<Phase>("LOADING");
   199→  const [content, setContent] = useState<PortalContent | null>(null);
   200→  const [kind, setKind] = useState<"GUEST" | "AUTH" | null>(null);
   201→  const [screen, setScreen] = useState("home");
   202→  const [fatal, setFatal] = useState<string | null>(null);
   203→  const [sessionKey, setSessionKey] = useState<string | null>(null);
   204→  const sessionRef = useRef<string | null>(null);
   205→
   206→  const fetchContent = useCallback(
   207→    async (sessionKey: string | null) => {
   208→      const headers: Record<string, string> = sessionKey ? { "x-portal-session": sessionKey } : {};
   209→      const res = await fetch(`/api/portal/content?slug=${encodeURIComponent(editionSlug)}`, { headers, cache: "no-store" });
   210→      const data = (await res.json()) as PortalContent & { error?: string };
   211→      if (!res.ok) throw new Error(data.error ?? "CONTENT_FAILED");
   212→      return data;
   213→    },
   214→    [editionSlug],
   215→  );
   216→
   217→  const bootstrap = useCallback(async () => {
   218→    try {
   219→      // 1) URL magic token — otomatik AUTH girişi (?portal=slug&t=pt_...)
   220→      if (magicToken) {
   221→        try {
   222→          const acc = await apiSend<{ sessionKey: string; kind: "GUEST" | "AUTH" }>("/api/portal/access", "POST", {
   223→            editionSlug,
   224→            mode: "TOKEN",
   225→            token: magicToken,
   226→          });
   227→          storeSession(editionSlug, acc.sessionKey);
   228→          sessionRef.current = acc.sessionKey;
   229→          setKind(acc.kind);
   230→          // token'ı URL'den sil — paylaşımda/refresh'te sızmasın
   231→          window.history.replaceState({}, "", `/?portal=${encodeURIComponent(editionSlug)}`);
   232→        } catch {
   233→          /* geçersiz token → normal giriş ekranına düş */
   234→        }
   235→      }
   236→      // 2) saklı oturum doğrulaması
   237→      let sessionKey = sessionRef.current ?? loadStoredSession(editionSlug);
   238→      if (sessionKey) {
   239→        try {
   240→          const st = await apiGet<{ valid: boolean; kind?: "GUEST" | "AUTH" }>("/api/portal/access", {
   241→            headers: { "x-portal-session": sessionKey },
   242→          });
   243→          if (!st.valid) {
   244→            clearSession(editionSlug);
   245→            sessionKey = null;
   246→          } else {
   247→            setKind(st.kind ?? "GUEST");
   248→          }
   249→        } catch {
   250→          sessionKey = null;
   251→        }
   252→      }
   253→      sessionRef.current = sessionKey;
   254→      // 3) içerik
   255→      const data = await fetchContent(sessionKey);
   256→      setContent(data);
   257→      setSessionKey(sessionKey); // await sonrası — senkron setState yok
   258→      if (data.phase === "ACTIVE" && sessionKey) {
   259→        setPhase("ACTIVE");
   260→        void portalSend("/api/portal/interact", { action: "VISIT" }, sessionKey).catch(() => undefined);
   261→      } else if (data.phase === "ACTIVE") {
   262→        // oturum yok ama portal aktif → giriş ekranı
   263→        setPhase("LOGIN");
   264→      } else if (data.phase === "DISABLED") {
   265→        setPhase("DISABLED");
   266→      } else {
   267→        setPhase("LOGIN");
   268→      }
   269→    } catch (e) {
   270→      setFatal(e instanceof Error ? e.message : "PORTAL_LOAD_FAILED");
   271→      setPhase("ERROR");
   272→    }
   273→  }, [editionSlug, magicToken, fetchContent]);
   274→
   275→  useEffect(() => {
   276→    // bootstrap asenkron akıştır — microtask'ta başlat (senkron setState yok)
   277→    void Promise.resolve().then(() => bootstrap());
   278→  }, []);
   279→
   280→  const cfg = content?.config;
   281→
   282→  // ── PWA: service worker + install prompt ──
   283→  const [deferredPrompt, setDeferredPrompt] = useState<{ prompt: () => void } | null>(null);
   284→  useEffect(() => {
   285→    if (phase !== "ACTIVE") return;
   286→    if (cfg?.pwaEnabled && "serviceWorker" in navigator) {
   287→      navigator.serviceWorker.register("/sw.js").catch(() => undefined);
   288→    }
   289→    const onBip = (e: Event) => {
   290→      e.preventDefault();
   291→      setDeferredPrompt(e as unknown as { prompt: () => void });
   292→    };
   293→    window.addEventListener("beforeinstallprompt", onBip);
   294→    return () => window.removeEventListener("beforeinstallprompt", onBip);
   295→  }, [phase, cfg?.pwaEnabled]);
   296→
   297→  const installApp = async () => {
   298→    if (!deferredPrompt) return;
   299→    deferredPrompt.prompt();
   300→    try {
   301→      await portalSend("/api/portal/interact", { action: "PWA_INSTALL" }, sessionKey);
   302→      toast({ title: t("portalApp.install.doneTitle"), description: t("portalApp.install.doneDesc") });
   303→    } catch {
   304→      /* analitik fire-and-forget */
   305→    }
   306→    setDeferredPrompt(null);
   307→  };
   308→
   309→  // ── canlı duyuru akışı (§5.4) — 20 sn polling ──
   310→  const lastPollRef = useRef<string>(new Date().toISOString());
   311→  useEffect(() => {
   312→    if (phase !== "ACTIVE" || !sessionRef.current) return;
   313→    const poll = async () => {
   314→      try {
   315→        const res = await apiGet<{ items: Announcement[] }>("/api/portal/announcements", {
   316→          headers: { "x-portal-session": sessionRef.current! },
   317→        });
   318→        const fresh = res.items.filter((a) => new Date(a.createdAt).getTime() > new Date(lastPollRef.current).getTime());
   319→        if (res.items.length > 0) {
   320→          lastPollRef.current = res.items[0].createdAt;
   321→        }
   322→        for (const a of fresh.reverse()) {
   323→          toast({ title: a.title, description: a.message });
   324→          if (typeof Notification !== "undefined" && Notification.permission === "granted") {
   325→            try {
   326→              new Notification(a.title, { body: a.message, icon: "/portal-icon-192.png" });
   327→            } catch {
   328→              /* yoksay */
   329→            }
   330→          }
   331→        }
   332→      } catch {
   333→        /* sessiz — polling */
   334→      }
   335→    };
   336→    void poll();
   337→    const id = setInterval(poll, 20_000);
   338→    return () => clearInterval(id);
   339→  }, [phase, toast]);
   340→
   341→  // ── hatırlatıcı motoru (§4.2): B2B randevuları + işaretli oturumlar ──
   342→  useEffect(() => {
   343→    if (phase !== "ACTIVE" || !cfg?.notifications?.enabled || !content) return;
   344→    const offsets = cfg.notifications.offsets.length ? cfg.notifications.offsets : [60, 30, 10];
   345→    const tick = () => {
   346→      const now = Date.now();
   347→      const fired = loadFired(editionSlug);
   348→      const targets: MarkedReminder[] = [];
   349→      // B2B randevuları (AUTH)
   350→      for (const m of content.b2b ?? []) {
   351→        if (!m.startsAt || m.status === "DECLINED" || m.status === "CANCELLED") continue;
   352→        const at = new Date(m.startsAt).getTime();
   353→        if (at <= now) continue;
   354→        targets.push({
   355→          key: `b2b:${m.assignmentId}`,
   356→          fireAt: at,
   357→          title: t("portalApp.reminder.b2bTitle"),
   358→          body: `${m.subject} — ${fmtDateTime(m.startsAt)}`,
   359→        });
   360→      }
   361→      // takvime eklenen (hatırlatması işaretlenen) oturumlar
   362→      for (const r of loadReminders(editionSlug)) targets.push(r);
   363→      for (const target of targets) {
   364→        for (const off of offsets) {
   365→          const fireAt = target.fireAt - off * 60_000;
   366→          const key = `${target.key}@${off}`;
   367→          if (now >= fireAt && now - fireAt < 10 * 60_000 && !fired.has(key)) {
   368→            fired.add(key);
   369→            const body = off >= 60 ? `${t("portalApp.reminder.in")} ${Math.round(off / 60)} ${t("portalApp.reminder.hours")}: ${target.body}` : `${t("portalApp.reminder.in")} ${off} ${t("portalApp.reminder.minutes")}: ${target.body}`;
   370→            toast({ title: target.title, description: body });
   371→            if (typeof Notification !== "undefined" && Notification.permission === "granted") {
   372→              try {
   373→                new Notification(target.title, { body, icon: "/portal-icon-192.png" });
   374→              } catch {
   375→                /* yoksay */
   376→              }
   377→            }
   378→          }
   379→        }
   380→      }
   381→      saveFired(editionSlug, fired);
   382→    };
   383→    void tick();
   384→    const id = setInterval(tick, 30_000);
   385→    return () => clearInterval(id);
   386→  }, [phase, cfg?.notifications?.enabled, content, editionSlug, toast, t]);
   387→
   388→  // ekranı bozmadan içeriği tazele (Q&A gönderimi sonrası vb.)
   389→  const refreshContent = useCallback(async () => {
   390→    try {
   391→      const data = await fetchContent(sessionRef.current);
   392→      setContent(data);
   393→    } catch {
   394→      /* sessiz — mevcut içerik kalır */
   395→    }
   396→  }, [fetchContent]);
   397→
   398→  const openForm = async (formIdOrSlug: string | null) => {
   399→    if (!formIdOrSlug) return;
   400→    try {
   401→      void portalSend("/api/portal/interact", { action: "FORM_OPEN" }, sessionKey);
   402→    } catch {
   403→      /* fire-and-forget */
   404→    }
   405→    window.location.href = `/?form=${encodeURIComponent(formIdOrSlug)}`;
   406→  };
   407→
   408→  const trackClick = (widgetKey: string) => {
   409→    void portalSend("/api/portal/interact", { action: "WIDGET_CLICK", widgetKey }, sessionKey).catch(() => undefined);
   410→  };
   411→
   412→  const gotoScreen = (s: string) => {
   413→    trackClick(s);
   414→    setScreen(s);
   415→    window.scrollTo({ top: 0 });
   416→  };
   417→
   418→  // ─── render dalları ───
   419→  if (phase === "LOADING") {
   420→    return (
   421→      <div className="grid min-h-screen place-items-center bg-background">
   422→        <div className="flex flex-col items-center gap-3" role="status" aria-label={t("portalApp.loading")}>
   423→          <img src="/portal-icon-192.png" alt="" className="size-16 rounded-2xl shadow-sm" />
   424→          <div className="flex items-center gap-2 text-sm text-muted-foreground">
   425→            <Icons.Loader2 className="size-4 animate-spin" /> {t("portalApp.loading")}
   426→          </div>
   427→        </div>
   428→      </div>
   429→    );
   430→  }
   431→
   432→  if (phase === "ERROR") {
   433→    return (
   434→      <div className="grid min-h-screen place-items-center bg-background p-6">
   435→        <div className="max-w-sm text-center">
   436→          <div className="mx-auto grid size-12 place-items-center rounded-full bg-red-50 text-red-600">
   437→            <Icons.AlertTriangle className="size-6" />
   438→          </div>
   439→          <h1 className="mt-3 text-lg font-semibold">{t("portalApp.error.title")}</h1>
   440→          <p className="mt-1 text-sm text-muted-foreground">
   441→            {fatal === "CONTENT_FAILED"
   442→              ? t("portalApp.error.content")
   443→              : fatal === "PORTAL_LOAD_FAILED"
   444→                ? t("portalApp.error.title")
   445→                : fatal}
   446→          </p>

   707→// ─── PASİF PORTAL: bakım / geri sayım (§5.1) ────────────────────────────────
   708→function DisabledScreen({ content }: { content: PortalContent }) {
   709→  const { t } = useLang();
   710→  const [left, setLeft] = useState<string>("");
   711→  useEffect(() => {
   712→    if (!content.countdownTo) return;
   713→    const tick = () => {
   714→      const diff = new Date(content.countdownTo!).getTime() - Date.now();
   715→      if (diff <= 0) {
   716→        setLeft(t("portalApp.countdown.started"));
   717→        return;
   718→      }
   719→      const d = Math.floor(diff / 86_400_000);
   720→      const h = Math.floor((diff % 86_400_000) / 3_600_000);
   721→      const m = Math.floor((diff % 3_600_000) / 60_000);
   722→      const s = Math.floor((diff % 60_000) / 1000);
   723→      setLeft(`${d > 0 ? `${d}${t("portalApp.countdown.dayUnit")} ` : ""}${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`);
   724→    };
   725→    void tick();
   726→    const id = setInterval(tick, 1000);
   727→    return () => clearInterval(id);
   728→  }, [content.countdownTo, t]);
   729→
   730→  return (
   731→    <div className="grid min-h-screen place-items-center bg-gradient-to-b from-teal-50 to-background p-6">
   732→      <div className="w-full max-w-sm text-center">
   733→        {content.edition.logoUrl ? (
   734→          <img src={content.edition.logoUrl} alt="" className="mx-auto size-16 rounded-2xl border bg-white object-contain p-1.5 shadow-sm" />
   735→        ) : (
   736→          <div className="mx-auto grid size-16 place-items-center rounded-2xl bg-teal-600 text-white shadow-sm">
   737→            <Icons.CalendarRange className="size-7" />
   738→          </div>
   739→        )}
   740→        <h1 className="mt-4 text-lg font-bold">{content.edition.name}</h1>
   741→        {content.countdownTo ? (
   742→          <>
   743→            <p className="mt-1 text-sm text-muted-foreground">{t("portalApp.countdown.desc")}</p>
   744→            <div className="mt-4 rounded-xl border bg-white p-4 font-mono text-2xl font-bold tabular-nums text-teal-700 shadow-sm dark:bg-card" aria-live="polite">
   745→              {left || "--:--:--"}
   746→            </div>
   747→          </>
   748→        ) : (
   749→          <p className="mt-2 text-sm text-muted-foreground">{content.maintenanceMessage || t("portalApp.countdown.maintenance")}</p>
   750→        )}
   751→        <p className="mt-6 text-[11px] text-muted-foreground">{content.tenant.name}</p>
   752→      </div>
   753→    </div>
   754→  );
   755→}

  1059→      const day = fmtDate(s.startTime);
  1060→      const arr = map.get(day) ?? [];
  1061→      arr.push(s);
  1062→      map.set(day, arr);

  1066→

  1068→    const key = `session:${s.id}`;
  1069→    const cur = loadReminders(slug);
  1070→    const existing = cur.find((r) => r.key === key);
  1071→    if (existing) {
  1072→      saveReminders(slug, cur.filter((r) => r.key !== key));
  1073→      toast({ title: t("portalApp.reminder.off") });

  1077→      toast({ title: t("portalApp.reminder.disabledTitle"), description: t("portalApp.reminder.disabledDesc") });
