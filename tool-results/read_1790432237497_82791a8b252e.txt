     1→"use client";
     2→// Kayıt & Katılımcılar — çok eksenli durum (kayıt × ödeme × katılım ayrı), onay akışı, LCV, bekleme listesi
     3→import { useMemo, useState } from "react";
     4→import { listEntity, listEntityPaged, apiSend, apiGet } from "@/lib/client";
     5→import { useApp } from "@/lib/store";
     6→import { SectionCard, EmptyState, Loading, ErrorState, useApi, PageHeader, StatusBadge, Chip, KpiCard } from "../bits";
     7→import { REGISTRATION_STATUS, WAITLIST_STATUS, REG_SOURCES, FUNDING_SOURCES, ATTENDANCE_STATUS, INVITATION_STATUS, fmtDateTime, fmtDate, label } from "@/lib/constants";
     8→import { Input } from "@/components/ui/input";
     9→import { Button } from "@/components/ui/button";
    10→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    11→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from "@/components/ui/dialog";
    12→import { Textarea } from "@/components/ui/textarea";
    13→import { Label } from "@/components/ui/label";
    14→import { useToast } from "@/hooks/use-toast";
    15→import * as Icons from "lucide-react";
    16→import { cn } from "@/lib/utils";
    17→import { useLang } from "@/lib/i18n";
    18→
    19→interface RegRow {
    20→  id: string; confirmationNo: string; status: string; source: string; fundingSource: string; submittedAt?: string | null; decidedAt?: string | null; notes?: string | null;
    21→  category?: { id: string; name: string; basePrice: number; currency: string } | null;
    22→  participation: {
    23→    id: string; attendance: string;
    24→    person: { id: string; firstName: string; lastName: string; email?: string | null; phone?: string | null; title?: string | null; company?: string | null; city?: string | null; country?: string | null };
    25→    roleAssignments?: { role: string }[];
    26→  };
    27→  entitlementClaims?: { id: string; status: string; entitlement: { label: string } }[];
    28→}
    29→interface CategoryRow { id: string; name: string; code: string; basePrice: number; currency: string; requiresApproval: boolean; capacity?: number | null; paymentInstruction?: string | null; isActive: boolean; _count?: { registrations: number } }
    30→interface InvitationRow { id: string; fullName: string; email: string; status: string; sentAt?: string | null; respondedAt?: string | null; organization?: { name: string } | null }
    31→
    32→// ── Bekleme listesi tipleri (/api/waitlist sözleşmesi) ──
    33→interface WaitlistPerson { id: string; firstName: string; lastName: string; email?: string | null; company?: string | null; title?: string | null; status: string }
    34→interface WaitlistEntryRow {
    35→  id: string; personId: string; priority: number; status: string; offeredAt?: string | null; offerExpiresAt?: string | null; respondedAt?: string | null; notes?: string | null; createdAt: string; categoryId?: string | null;
    36→  person: WaitlistPerson;
    37→  category?: { id: string; name: string; code: string; capacity?: number | null } | null;
    38→  convertedRegistration?: { id: string; confirmationNo: string; status: string } | null;
    39→}
    40→interface WaitlistCategory { id: string; name: string; code: string; capacity: number | null; taken: number; seatsLeft: number | null; waitingCount: number; offeredCount: number; fillPercent: number | null }
    41→interface WaitlistData {
    42→  summary: { waiting: number; offered: number; converted: number; declinedExpired: number; fullCategories: number; categoriesWithQueue: number };
    43→  categories: WaitlistCategory[];
    44→  entries: WaitlistEntryRow[];
    45→}
    46→interface PersonLite { id: string; firstName: string; lastName: string; email?: string | null; company?: string | null }
    47→
    48→export function RegistrationsView() {
    49→  const { currentEditionId, bump, refreshKey } = useApp();
    50→  const { toast } = useToast();
    51→  const { t } = useLang(); // dil değişiminde re-render (F9-R-b)
    52→  const [statusFilter, setStatusFilter] = useState("ALL");
    53→  const [q, setQ] = useState("");
    54→  const [decideTarget, setDecideTarget] = useState<{ reg: RegRow; decision: "CONFIRMED" | "REJECTED" | "CANCELLED" } | null>(null);
    55→  const [reason, setReason] = useState("");
    56→  const [busy, setBusy] = useState(false);
    57→  const [tab, setTab] = useState<"list" | "waitlist" | "lcv">("list");
    58→
    59→  // ── R10-a: çift tıkla tam durum düzenleme — kişi + katılım + kayıt tek diyaloğda ──
    60→  const [editOpen, setEditOpen] = useState(false);
    61→  const [editRow, setEditRow] = useState<RegRow | null>(null);
    62→  const [editBusy, setEditBusy] = useState(false);
    63→  const emptyEdit = { firstName: "", lastName: "", email: "", phone: "", title: "", company: "", city: "", country: "", attendance: "NOT_ARRIVED", categoryId: "", status: "DRAFT", fundingSource: "SELF_PAID", notes: "" };
    64→  const [editForm, setEditForm] = useState(emptyEdit);
    65→
    66→  const openFullEdit = (r: RegRow) => {
    67→    setEditRow(r);
    68→    setEditForm({
    69→      firstName: r.participation.person.firstName, lastName: r.participation.person.lastName,
    70→      email: r.participation.person.email ?? "", phone: r.participation.person.phone ?? "",
    71→      title: r.participation.person.title ?? "", company: r.participation.person.company ?? "",
    72→      city: r.participation.person.city ?? "", country: r.participation.person.country ?? "",
    73→      attendance: r.participation.attendance,
    74→      categoryId: r.category?.id ?? "", status: r.status, fundingSource: r.fundingSource, notes: r.notes ?? "",
    75→    });
    76→    setEditOpen(true);
    77→  };
    78→
    79→  const saveFullEdit = async () => {
    80→    if (!editRow) return;
    81→    setEditBusy(true);
    82→    try {
    83→      // 1) Kişi — yalnız skaler alanlar (registry sanitize: "" → null; asla iç içe nesne yok)
    84→      await apiSend(`/api/people/${editRow.participation.person.id}`, "PUT", {
    85→        firstName: editForm.firstName, lastName: editForm.lastName,
    86→        email: editForm.email || null, phone: editForm.phone || null, title: editForm.title || null,
    87→        company: editForm.company || null, city: editForm.city || null, country: editForm.country || null,
    88→      });
    89→      // 2) Katılım — katılım durumu
    90→      await apiSend(`/api/participations/${editRow.participation.id}`, "PUT", { attendance: editForm.attendance });
    91→      // 3) Kayıt — kategori + durum + fon kaynağı + not
    92→      await apiSend(`/api/registrations/${editRow.id}`, "PUT", {
    93→        categoryId: editForm.categoryId || null, status: editForm.status,
    94→        fundingSource: editForm.fundingSource, notes: editForm.notes || null,
    95→      });
    96→      toast({ title: "Değişiklikler kaydedildi", description: t("registrations.updatedAll") });
    97→      setEditOpen(false); setEditRow(null);
    98→      reload(); bump();
    99→    } catch (e) {
   100→      toast({ title: "Kaydedilemedi", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   101→    } finally {
   102→      setEditBusy(false);
   103→    }
   104→  };
   105→
   106→  // TASK-A F6: sunucu-taraflı relation-aware arama (teyit no + kategori + kişi ad/e-posta)
   107→  // + imleçli load-more — 400 satırlık sessiz kesme KALDIRILDI; tüm kayıtlar erişilebilir.
   108→  const regsLoader = (cursor?: string) =>
   109→    listEntityPaged<RegRow>("registrations", {
   110→      editionId: currentEditionId ?? undefined,
   111→      status: statusFilter === "ALL" ? undefined : statusFilter,
   112→      q: q.trim() || undefined,
   113→      limit: 200,
   114→    }, cursor);
   115→  const { data: regsPaged, error, reload, loading, more } = useApi<{ items: RegRow[]; nextCursor?: string | null }>(regsLoader, [currentEditionId, statusFilter, q, refreshKey], { append: true });
   116→  const registrations = useMemo(() => regsPaged?.items ?? [], [regsPaged]);
   117→  const { data: categories } = useApi<CategoryRow[]>(() => listEntity<CategoryRow>("registration-categories", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey]);
   118→  const { data: invitations } = useApi<InvitationRow[]>(() => listEntity<InvitationRow>("invitations", { editionId: currentEditionId ?? undefined }), [currentEditionId, refreshKey, tab]);
   119→
   120→  const decide = async () => {
   121→    if (!decideTarget) return;
   122→    setBusy(true);
   123→    try {
   124→      if (decideTarget.decision === "CANCELLED") {
   125→        const res = await apiSend<{ waitlistOffered?: { personName: string }[] }>("/api/flows", "POST", { action: "registration.cancel", registrationId: decideTarget.reg.id, reason });
   126→        const offers = res.waitlistOffered ?? [];
   127→        toast({
   128→          title: t("registrations.cancelled"),
   129→          description: offers.length > 0
   130→            ? `Koltuk boşaldı — bekleme listesinden teklif gönderildi: ${offers.map((o) => o.personName).join(", ")}`
   131→            : "Yaka kartı ve haklar etkilendi; etki önizlemesi kayıtta görülür.",
   132→        });
   133→      } else {
   134→        await apiSend("/api/flows", "POST", { action: "registration.decide", registrationId: decideTarget.reg.id, decision: decideTarget.decision });
   135→        toast({
   136→          title: decideTarget.decision === "CONFIRMED" ? t("registrations.approved") : t("registrations.rejected"),
   137→          description: decideTarget.decision === "CONFIRMED"
   138→            ? "Hak claim'i CONSUMED'a geçti, yaka kartı READY — ödeme durumu ayrı hesaplanır."
   139→            : "Yaka kartı ve haklar etkilendi; etki önizlemesi kayıtta görülür.",
   140→        });
   141→      }
   142→      setDecideTarget(null); setReason(""); reload(); bump();
   143→    } catch (e) {
   144→      toast({ title: "İşlem başarısız", description: e instanceof Error ? e.message : "Hata", variant: "destructive" });
   145→    } finally {
   146→      setBusy(false);
   147→    }
   148→  };
   149→
   150→  const counts = (s: string) => (s === "ALL" ? registrations.length : registrations.filter((r) => r.status === s).length);
   151→
   152→  return (
   153→    <div>
   154→      <PageHeader title={t("registrations.title")} desc="Kategori → form → onay akışı; kayıt/ödeme/katılım üç ayrı eksen">
   155→        <div className="flex rounded-lg border p-0.5">
   156→          <button onClick={() => setTab("list")} className={cn("rounded-md px-3 py-1.5 text-xs font-medium", tab === "list" ? "bg-primary text-primary-foreground" : "text-muted-foreground")}>Kayıtlar</button>
   157→          <button onClick={() => setTab("waitlist")} className={cn("rounded-md px-3 py-1.5 text-xs font-medium", tab === "waitlist" ? "bg-primary text-primary-foreground" : "text-muted-foreground")}>Bekleme</button>
   158→          <button onClick={() => setTab("lcv")} className={cn("rounded-md px-3 py-1.5 text-xs font-medium", tab === "lcv" ? "bg-primary text-primary-foreground" : "text-muted-foreground")}>LCV / Davetler</button>
   159→        </div>
   160→      </PageHeader>
   161→
   162→      {tab === "list" ? (
   163→        <>
   164→          {/* Kategori kartları — doluluk */}
   165→          <div className="mb-4 grid grid-cols-2 gap-3 md:grid-cols-4 xl:grid-cols-7">
   166→            {(categories ?? []).map((c) => (
   167→              <KpiCard
   168→                key={c.id}
   169→                label={c.name}
   170→                value={`${c._count?.registrations ?? 0}${c.capacity ? ` / ${c.capacity}` : ""}`}
   171→                sub={c.basePrice > 0 ? `${(c.basePrice / 100).toLocaleString("tr-TR")} ${c.currency}` : "ücretsiz"} // F6: kuruş→₺
   172→                tone={c.requiresApproval ? "amber" : "teal"}
   173→              />
   174→            ))}
   175→          </div>
   176→
   177→          <div className="mb-3 flex flex-wrap items-center gap-2">
   178→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   179→              <SelectTrigger className="h-9 w-44"><SelectValue /></SelectTrigger>
   180→              <SelectContent>
   181→                <SelectItem value="ALL">Tüm durumlar ({counts("ALL")})</SelectItem>
   182→                {Object.entries(REGISTRATION_STATUS).map(([k, v]) => (
   183→                  <SelectItem key={k} value={k}>{v} ({counts(k)})</SelectItem>
   184→                ))}
   185→              </SelectContent>
   186→            </Select>
   187→            <Input placeholder="Ad / e-posta / kayıt no…" value={q} onChange={(e) => setQ(e.target.value)} className="h-9 w-64" />
   188→            <Button variant="ghost" size="sm" onClick={reload}><Icons.RefreshCw className="size-4" /></Button>
   189→            <span className="ml-auto flex items-center gap-2">
   190→              <Chip tone="neutral">
   191→                <span className="inline-flex items-center gap-1"><Icons.MousePointerClick className="size-3" aria-hidden />Çift tıklama ile de açılır</span>
   192→              </Chip>
   193→              <span className="text-xs text-muted-foreground">{registrations.length} kayıt listelendi</span>
   194→            </span>
   195→          </div>
   196→
   197→          {loading ? <Loading /> : error ? <ErrorState message={error} onRetry={reload} /> : registrations.length === 0 ? (
   198→            <EmptyState title="Bu filtrelerle eşleşen kayıt yok" desc="Henüz kayıt yoksa kayıt bağlantısını paylaşın veya ilk kişiyi ekleyin." />
   199→          ) : (
   200→            <div className="overflow-hidden rounded-xl border bg-card">
   201→              <div className="overflow-x-auto maven-scroll">
   202→                <table className="w-full min-w-[900px] text-sm">
   203→                  <thead>
   204→                    <tr className="border-b bg-muted/50 text-left text-xs text-muted-foreground">
   205→                      <th className="px-3 py-2.5 font-medium">Kişi</th>
   206→                      <th className="px-3 py-2.5 font-medium">Kayıt No</th>
   207→                      <th className="px-3 py-2.5 font-medium">Kategori</th>
   208→                      <th className="px-3 py-2.5 font-medium">Kayıt Durumu</th>
   209→                      <th className="px-3 py-2.5 font-medium">Fon Kaynağı</th>
   210→                      <th className="px-3 py-2.5 font-medium">Kaynak</th>
   211→                      <th className="px-3 py-2.5 font-medium">Katılım</th>
   212→                      <th className="px-3 py-2.5 font-medium text-right">Eylem</th>
   213→                    </tr>
   214→                  </thead>
   215→                  <tbody>
   216→                    {registrations.map((r) => (
   217→                      <tr
   218→                        key={r.id}
   219→                        className="cursor-pointer border-b transition hover:bg-muted/40 last:border-0"
   220→                        onDoubleClick={() => openFullEdit(r)}
   221→                        title="Çift tıkla: tüm durumları gör/düzenle"
   222→                      >
   223→                        <td className="px-3 py-2.5">
   224→                          <p className="font-medium leading-tight">{r.participation.person.firstName} {r.participation.person.lastName}</p>
   225→                          <p className="text-xs text-muted-foreground">{r.participation.person.company ?? "—"}</p>
   226→                        </td>
   227→                        <td className="px-3 py-2.5 font-mono text-xs">{r.confirmationNo}</td>
   228→                        <td className="px-3 py-2.5">
   229→                          {r.category?.name ?? "—"}
   230→                          {r.entitlementClaims && r.entitlementClaims.length > 0 && (
   231→                            <Chip tone="violet">{r.entitlementClaims[0].entitlement.label.slice(0, 22)}</Chip>
   232→                          )}
   233→                        </td>
   234→                        <td className="px-3 py-2.5"><StatusBadge map={REGISTRATION_STATUS} value={r.status} /></td>
   235→                        <td className="px-3 py-2.5 text-xs">{label(FUNDING_SOURCES, r.fundingSource)}</td>
   236→                        <td className="px-3 py-2.5 text-xs">{label(REG_SOURCES, r.source)}</td>
   237→                        <td className="px-3 py-2.5"><StatusBadge map={ATTENDANCE_STATUS} value={r.participation.attendance} /></td>
   238→                        <td className="px-3 py-2.5">
   239→                          <div className="flex justify-end gap-1">
   240→                            <Button
   241→                              size="sm" variant="ghost" className="h-7 w-7 p-0 text-muted-foreground"
   242→                              onClick={() => openFullEdit(r)}
   243→                              aria-label={`${r.participation.person.firstName} ${r.participation.person.lastName} kaydının tüm durumlarını gör/düzenle`}
   244→                              title="Tüm durumları gör/düzenle"
   245→                            >
   246→                              <Icons.Pencil className="size-3.5" />
   247→                            </Button>
   248→                            {r.status === "PENDING_APPROVAL" && (
   249→                              <>
   250→                                <Button size="sm" variant="outline" className="h-7 border-emerald-200 text-emerald-700 hover:bg-emerald-50" onClick={() => setDecideTarget({ reg: r, decision: "CONFIRMED" })}>
   251→                                  <Icons.Check className="size-3.5" /> Onayla
   252→                                </Button>
   253→                                <Button size="sm" variant="outline" className="h-7 border-rose-200 text-rose-700 hover:bg-rose-50" onClick={() => setDecideTarget({ reg: r, decision: "REJECTED" })}>
   254→                                  <Icons.X className="size-3.5" /> Reddet
   255→                                </Button>
   256→                              </>
   257→                            )}
   258→                            {["CONFIRMED", "PENDING_APPROVAL", "SUBMITTED"].includes(r.status) && (
   259→                              <Button size="sm" variant="ghost" className="h-7 text-muted-foreground" onClick={() => setDecideTarget({ reg: r, decision: "CANCELLED" })}>
   260→                                İptal
   261→                              </Button>
   262→                            )}
   263→                          </div>
   264→                        </td>
   265→                      </tr>
   266→                    ))}
   267→                  </tbody>
   268→                </table>
   269→              </div>
   270→              {/* TASK-A F6: kesintisiz yükleme — 200'er sayfalık imleç yürüyüşü, sessiz kesme yok */}
   271→              {more?.hasMore && (
   272→                <div className="flex items-center justify-center border-t bg-muted/20 p-3">
   273→                  <Button variant="outline" size="sm" className="gap-1.5 text-xs" disabled={more.loading} onClick={more.next}>
   274→                    {more.loading ? <Icons.Loader2 className="size-3.5 animate-spin" /> : <Icons.ChevronsDown className="size-3.5" />}
   275→                    Daha fazla yükle
   276→                  </Button>
   277→                </div>
   278→              )}
   279→            </div>
   280→          )}
   281→        </>
   282→      ) : tab === "waitlist" ? (
   283→        <WaitlistTab editionId={currentEditionId} categories={categories ?? []} onChanged={() => { bump(); }} />
   284→      ) : (
   285→        <SectionCard title="LCV — Davet Listesi Yönetimi" desc="Davet bir kayıt yerine geçmez; 'gelecek' yanıtı kayıt yoluna yönlenir">
   286→          {(invitations ?? []).length === 0 ? (
   287→            <EmptyState title="Henüz davet oluşturmadınız" desc="Davetli ekleyin veya davet listesi yükleyin." />
   288→          ) : (
   289→            <div className="grid gap-2">
   290→              {(invitations ?? []).map((inv) => (
   291→                <div key={inv.id} className="flex flex-wrap items-center gap-3 rounded-lg border p-3">
   292→                  <div className="min-w-0 flex-1">
   293→                    <p className="text-sm font-medium">{inv.fullName}</p>
   294→                    <p className="text-xs text-muted-foreground">{inv.email}{inv.organization ? ` · ${inv.organization.name}` : ""}</p>
   295→                  </div>
   296→                  <div className="text-xs text-muted-foreground">
   297→                    {inv.sentAt ? `gönderim ${fmtDate(inv.sentAt)}` : "gönderilmedi"}
   298→                    {inv.respondedAt ? ` · yanıt ${fmtDate(inv.respondedAt)}` : ""}
   299→                  </div>
   300→                  <StatusBadge map={INVITATION_STATUS} value={inv.status} />
   301→                </div>
   302→              ))}
   303→            </div>
   304→          )}
   305→        </SectionCard>
   306→      )}
   307→
   308→      {/* Onay/ret/iptal dialogu — etki önizlemesi ile */}
   309→      <Dialog open={Boolean(decideTarget)} onOpenChange={(o) => !o && setDecideTarget(null)}>
   310→        <DialogContent className="sm:max-w-md">
   311→          <DialogHeader>
   312→            <DialogTitle>
   313→              {decideTarget?.decision === "CONFIRMED" ? "Kaydı onayla" : decideTarget?.decision === "REJECTED" ? "Kaydı reddet" : "Kaydı iptal et"}
   314→            </DialogTitle>
   315→            <DialogDescription>
   316→              {decideTarget?.reg.participation.person.firstName} {decideTarget?.reg.participation.person.lastName} — {decideTarget?.reg.confirmationNo}
   317→            </DialogDescription>
   318→          </DialogHeader>
   319→          <div className="rounded-lg border bg-muted/40 p-3 text-xs leading-relaxed text-muted-foreground">
   320→            <p className="mb-1 font-semibold text-foreground">Etki önizlemesi</p>
   321→            {decideTarget?.decision === "CONFIRMED" && (
   322→              <ul className="list-disc space-y-0.5 pl-4">
   323→                <li>Ayrılmış hak claim'i <b>CONSUMED</b> olur (davette ayır, onayda kullan)</li>
   324→                <li>Yaka kartı <b>READY</b> durumuna geçer</li>
   325→                <li>Ödeme gereksinimi ayrıca hesaplanır — fon kaynağı: {label(FUNDING_SOURCES, decideTarget?.reg.fundingSource)}</li>
   326→              </ul>
   327→            )}
   328→            {decideTarget?.decision === "CANCELLED" && (
   329→              <ul className="list-disc space-y-0.5 pl-4">
   330→                <li>Yaka kartı <b>VOID</b> olur ve erişim hakkı etkilenir</li>
   331→                <li>Sponsor hakları geri yüklenir (politika gereği ayrıca karar)</li>
   332→                <li>İade ayrı bir finansal işlemdir — burada otomatik yapılmaz</li>
   333→                <li>Kategoride bekleme listesi varsa koltuk sıradakine <b>otomatik teklif edilir</b></li>
   334→              </ul>
   335→            )}
   336→            {decideTarget?.decision === "REJECTED" && <ul className="list-disc space-y-0.5 pl-4"><li>Katılımcıya gerekçe ile bildirim gider</li></ul>}
   337→          </div>
   338→          {decideTarget?.decision !== "CONFIRMED" && (
   339→            <div>
   340→              <Label>Gerekçe</Label>
   341→              <Textarea value={reason} onChange={(e) => setReason(e.target.value)} placeholder="Karar gerekçesi…" className="mt-1" />
   342→            </div>
   343→          )}
   344→          <DialogFooter>
   345→            <Button variant="outline" onClick={() => setDecideTarget(null)}>Vazgeç</Button>
   346→            <Button
   347→              onClick={decide}
   348→              disabled={busy || (decideTarget?.decision !== "CONFIRMED" && !reason)}
   349→              variant={decideTarget?.decision === "CONFIRMED" ? "default" : "destructive"}
   350→            >
   351→              {busy ? "İşleniyor…" : "Onayla"}
   352→            </Button>
   353→          </DialogFooter>
   354→        </DialogContent>
   355→      </Dialog>

   466→function WaitlistTab({ editionId, categories, onChanged }: { editionId: string | null; categories: CategoryRow[]; onChanged: () => void }) {
   467→  const { refreshKey } = useApp();
   468→  const { toast } = useToast();

   479→  const loader = async (): Promise<WaitlistData | null> => {
   480→    if (!editionId) return null;
   481→    return apiGet<WaitlistData>(`/api/waitlist?editionId=${encodeURIComponent(editionId)}`);
   482→  };
   483→  const { data, error, reload, loading } = useApi<WaitlistData | null>(loader, [editionId, refreshKey]);
   484→
   485→  const { data: people } = useApi<PersonLite[]>(() => listEntity<PersonLite>("people", { limit: 300 }), [editionId]);
   486→
   487→  const entries = (data?.entries ?? []).filter((e) => catFilter === "ALL" || (catFilter === "GENERAL" ? !e.categoryId : e.categoryId === catFilter));
   488→  const summary = data?.summary;

   490→  const isFullCategory = (categoryId?: string | null) => {
   491→    if (!categoryId) return false;
   492→    const c = (data?.categories ?? []).find((x) => x.id === categoryId);
   493→    // açık teklifler de koltuk tutar — boş koltuk başına tek teklif kuralı

   501→    return sameQueue.findIndex((x) => x.id === e.id) + 1;
   502→  };

   504→  const offerCountdown = (e: WaitlistEntryRow): { text: string; expired: boolean } | null => {
   505→    if (e.status !== "OFFERED" || !e.offerExpiresAt) return null;
   506→    const ms = new Date(e.offerExpiresAt).getTime() - Date.now();
   507→    if (ms <= 0) return { text: "Süresi doldu", expired: true };
   508→    const hours = Math.floor(ms / 3600000);
   509→    const minutes = Math.floor((ms % 3600000) / 60000);

   716→                                {cd.expired ? "Süresi doldu" : `${fmtDateTime(e.offeredAt)} — ${cd.text}`}
   717→                              </span>
   718→                            ) : "—"}
   719→                          </td>
   720→                          <td className="px-3 py-2.5">
   721→                            <div className="flex justify-end gap-1">
   722→                              {e.status === "WAITING" && (

   753→                            </div>
   754→                          </td>
   755→                        </tr>

   757→                    })}
   758→                  </tbody>
   759→                </table>
   760→              </div>
   761→            </div>
   762→
   763→            {/* geçmiş (terminal durumlar) */}
   764→            {closedEntries.length > 0 && (

   775→                      <span className="ml-auto text-muted-foreground">{e.respondedAt ? `yanıt ${fmtDate(e.respondedAt)}` : `eklenme ${fmtDate(e.createdAt)}`}</span>

   779→                </div>
   780→              </div>
   781→            )}
   782→          </>
   783→        )}
   784→      </SectionCard>
   785→
   786→      {/* Listeye ekle dialogu */}
   787→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   788→        <DialogContent className="sm:max-w-md">
   789→          <DialogHeader>
   790→            <DialogTitle>Bekleme listesine ekle</DialogTitle>
   791→            <DialogDescription>Kişi koltuk boşaldığında öncelik sırasına göre otomatik teklif alır.</DialogDescription>
   792→          </DialogHeader>
   793→          <div className="space-y-3">
   794→            <div>
   795→              <Label>Kişi *</Label>
   796→              <Select value={addPersonId} onValueChange={setAddPersonId}>
   797→                <SelectTrigger className="mt-1"><SelectValue placeholder={t("registrations.selectPersonPh")} /></SelectTrigger>
   798→                <SelectContent>
   799→                  {(people ?? []).map((p) => (
   800→                    <SelectItem key={p.id} value={p.id}>{p.firstName} {p.lastName}{p.company ? ` — ${p.company}` : ""}</SelectItem>
   801→                  ))}
   802→                </SelectContent>
   803→              </Select>
   804→            </div>
   805→            <div>
   806→              <Label>Kategori</Label>
   807→              <Select value={addCategoryId} onValueChange={setAddCategoryId}>
   808→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   809→                <SelectContent>
   810→                  <SelectItem value="GENERAL">Genel (kategorisiz sıra)</SelectItem>
   811→                  {categories.map((c) => <SelectItem key={c.id} value={c.id}>{c.name} ({c.code})</SelectItem>)}
   812→                </SelectContent>
   813→              </Select>
   814→            </div>
   815→            <div className="grid grid-cols-2 gap-3">
   816→              <div>
   817→                <Label>Öncelik</Label>
   818→                <Input type="number" min={1} value={addPriority} onChange={(e) => setAddPriority(e.target.value)} placeholder="otomatik: sıra sonu" className="mt-1" />
   819→              </div>
   820→              <div className="flex items-end">
   821→                <p className="text-xs text-muted-foreground">Küçük sayı önce teklif alır. Boş bırakılırsa sıranın sonuna eklenir.</p>
   822→              </div>
   823→            </div>
   824→            <div>
   825→              <Label>Not</Label>
   826→              <Textarea value={addNotes} onChange={(e) => setAddNotes(e.target.value)} placeholder="Örn. sponsor ile görüştü…" className="mt-1" rows={2} />
   827→            </div>
   828→          </div>
   829→          <DialogFooter>
   830→            <Button variant="outline" onClick={() => setAddOpen(false)}>Vazgeç</Button>
   831→            <Button onClick={addEntry} disabled={busyAll || !addPersonId}>
   832→              {busyId === "add" ? <Icons.Loader2 className="size-4 animate-spin" /> : <Icons.UserPlus className="size-4" />}
   833→              Listeye Ekle
   834→            </Button>
   835→          </DialogFooter>
   836→        </DialogContent>
   837→      </Dialog>
   838→    </div>
   839→  );
