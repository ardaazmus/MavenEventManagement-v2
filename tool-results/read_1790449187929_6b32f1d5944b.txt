   344→function ImportParticipantsDialog({ open, onOpenChange, onImported }: { open: boolean; onOpenChange: (o: boolean) => void; onImported: () => void }) {
   345→  const { t } = useLang();
   346→  const { toast } = useToast();
   347→  const { currentEditionId } = useApp();
   348→  const [tag, setTag] = useState("");
   349→  const [categoryIds, setCategoryIds] = useState<string[]>([]);
   350→  const [busy, setBusy] = useState(false);
   351→  const [result, setResult] = useState<{ created: number; merged: number; skipped: number; total: number } | null>(null);
   352→
   353→  const loadCategories = useCallback(async () => {
   354→    if (!currentEditionId) return [];
   355→    try {
   356→      return await listEntity<CategoryLite>("registration-categories", { editionId: currentEditionId, limit: 100 });
   357→    } catch {
   358→      return [];
   359→    }
   360→  }, [currentEditionId]);
   361→
   362→  const [categories, setCategories] = useState<CategoryLite[]>([]);
   363→  if (open && categories.length === 0) {
   364→    void loadCategories().then(setCategories);
   365→  }
   366→
   367→  const run = async () => {
   368→    if (!currentEditionId) return;
   369→    setBusy(true);
   370→    setResult(null);
   371→    try {
   372→      const r = await apiSend<{ created: number; merged: number; skipped: number; total: number }>(
   373→        "/api/customer-contacts/import-participants", "POST",
   374→        { editionId: currentEditionId, tag: tag.trim() || undefined, categoryIds: categoryIds.length ? categoryIds : undefined },
   375→      );
   376→      setResult(r);
   377→      toast({ title: t("commsCrm.customer.importDone"), description: t("commsCrm.customer.importDoneDesc", { created: r.created, merged: r.merged, skipped: r.skipped }) });
   378→      onImported();
   379→    } catch (e) {
   380→      toast({ title: t("commsCrm.customer.importFail"), description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   381→    } finally {
   382→      setBusy(false);

   434→function SingleSendDialog({ target, onOpenChange, onSent }: { target: CustomerRow | null; onOpenChange: (o: boolean) => void; onSent: () => void }) {
