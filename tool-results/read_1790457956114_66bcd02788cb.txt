   469→// F6: fmtMoney MINOR unit (kuruş) alır — ₺1.234,56 biçiminde gösterir.
   470→// Form GİRDİLERİ (₺ major) için fmtMoney(toMinor(x)) kullanın; ham ₺ için fmtMoneyMajor.
   471→export function fmtMoney(v: number | null | undefined, currency = "TRY"): string {
   472→  const major = (v ?? 0) / 100;
   473→  return new Intl.NumberFormat("tr-TR", { style: "currency", currency, minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(major);
   474→}
   475→
   476→// F6: major (₺) değer doğrudan gösterimi — yalnız kullanıcı girdisi yankıları için
   477→export function fmtMoneyMajor(v: number | null | undefined, currency = "TRY"): string {
   478→  const n = v ?? 0;
   479→  return new Intl.NumberFormat("tr-TR", { style: "currency", currency, maximumFractionDigits: 0 }).format(n);
   480→}

   482→export function fmtDate(d?: string | Date | null, withTime = false): string {
   483→  if (!d) return "—";
   484→  const date = typeof d === "string" ? new Date(d) : d;
   485→  return date.toLocaleDateString("tr-TR", withTime
   486→    ? { day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" }
   487→    : { day: "2-digit", month: "short", year: "numeric" });
   488→}
   489→
   490→export function fmtDateTime(d?: string | Date | null): string {
   491→  return fmtDate(d, true);
   492→}
