   469→   469→// F6: fmtMoney MINOR unit (kuruş) alır — ₺1.234,56 biçiminde gösterir.
   470→   470→// Form GİRDİLERİ (₺ major) için fmtMoney(toMinor(x)) kullanın; ham ₺ için fmtMoneyMajor.
   471→   471→export function fmtMoney(v: number | null | undefined, currency = "TRY"): string {
   472→   472→  const major = (v ?? 0) / 100;
   473→   473→  return new Intl.NumberFormat("tr-TR", { style: "currency", currency, minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(major);
   474→   474→}

   476→   476→// F6: major (₺) değer doğrudan gösterimi — yalnız kullanıcı girdisi yankıları için
   477→   477→export function fmtMoneyMajor(v: number | null | undefined, currency = "TRY"): string {
   478→   478→  const n = v ?? 0;
   479→   479→  return new Intl.NumberFormat("tr-TR", { style: "currency", currency, maximumFractionDigits: 0 }).format(n);
   480→   480→}

   482→   482→export function fmtDate(d?: string | Date | null, withTime = false): string {
   483→   483→  if (!d) return "—";
   484→   484→  const date = typeof d === "string" ? new Date(d) : d;
   485→   485→  return date.toLocaleDateString("tr-TR", withTime
   486→   486→    ? { day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" }
   487→   487→    : { day: "2-digit", month: "short", year: "numeric" });
   488→   488→}
   489→   489→
   490→   490→export function fmtDateTime(d?: string | Date | null): string {
   491→   491→  return fmtDate(d, true);
