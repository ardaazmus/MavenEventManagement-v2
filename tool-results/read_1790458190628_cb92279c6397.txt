     9→type AnyDelegate = {
    10→  findMany: (args?: Record<string, unknown>) => Promise<unknown[]>;
    11→  findUnique: (args: Record<string, unknown>) => Promise<unknown>;
    12→  findFirst: (args?: Record<string, unknown>) => Promise<unknown>;
    13→  create: (args: Record<string, unknown>) => Promise<unknown>;
    14→  update: (args: Record<string, unknown>) => Promise<unknown>;
    15→  delete: (args: Record<string, unknown>) => Promise<unknown>;
    16→  count: (args?: Record<string, unknown>) => Promise<number>;
    17→};
    18→
    19→export interface EntityConfig {
    20→  delegate: AnyDelegate;
    21→  include?: Record<string, unknown>;
    22→  searchFields?: string[];          // serbest metin arama alanları (contains)
    23→  relationSearch?: (q: string) => Record<string, unknown>; // ilişki-uzanan serbest arama (P2: registrations vb.)
    24→  filterFields?: string[];          // ?field=value eşitlik filtreleri
    25→  defaultWhere?: Record<string, unknown>; // tüm listeye uygulanan taban filtre (ör. MERGED kişileri gizle)
    26→  orderBy?: Record<string, "asc" | "desc">;
    27→  auditType?: string;               // aktivite günlüğü tipi
    28→  auditMessage?: (data: Record<string, unknown>, action: "create" | "update" | "delete") => string;
    29→  // S3: sır koruma — okumada maskeleme (password sızması yasak), yazmada şifreleme
    30→  readMask?: (row: Record<string, unknown>) => Record<string, unknown>;
    31→  writeTransform?: (data: Record<string, unknown>, isUpdate: boolean) => Promise<Record<string, unknown>> | Record<string, unknown>;
    32→  // DÜZELTME (server-side validation): varlık-bazlı yazım sözleşmesi — null döner (geçerli)
    33→  // ya da hata mesajı döner (400). UI devre dışı butonu güvenlik kontrolü DEĞİLDİR.
    34→  validate?: (data: Record<string, unknown>, isUpdate: boolean) => string | null;
    35→  // Eşzamanlılık kontrolü kancası (B2B randevu çakışması): create/update ÖNCESİ
    36→  // async çakışma denetimi — hata mesajı dönerse 409 ile reddedilir. Inside the
    37→  // hook: withLock(anahtar) + çakışan-aralık sorgusu (check-then-write serileştirme).
    38→  beforeWrite?: (data: Record<string, unknown>, isUpdate: boolean, existingId?: string) => Promise<string | null>;
    39→}

   724→// YASAK). POST etkilenmez (meşru ilk-yazımlar korunur); yalnız güncelleme yolu.
   725→const IMMUTABLE_ON_UPDATE: Record<string, Set<string>> = {
   726→  payments: new Set(["status", "amount", "currency", "orderId", "paidAt"]),
   727→  registrations: new Set(["status", "orderId", "participationId", "categoryId"]),

   732→export function sanitize(data: Record<string, unknown>): Record<string, unknown> {
   733→  const out: Record<string, unknown> = {};
   734→  for (const [k, v] of Object.entries(data)) {
   735→    if (FORBIDDEN.has(k)) continue;
   736→    if (v === undefined) continue;
   737→    out[k] = v === "" ? null : v;
   738→  }
   739→  return out;
   740→}

   742→// P2: PUT yolu — sanitize + entity'ye özel durum-makinesi alan düşmesi.
   743→// Düşülen alan varsa hata DÖNMEZ (mevcut UI sessizce status gönderiyor olabilir);
   744→// alan yazılmaz ve yanıt meta'sında bildirilir (davranış-koruma + şeffaflık).
   745→export function sanitizeForUpdate(entity: string, data: Record<string, unknown>): { data: Record<string, unknown>; dropped: string[] } {
   746→  const base = sanitize(data);
   747→  const immutable = IMMUTABLE_ON_UPDATE[entity];
   748→  if (!immutable) return { data: base, dropped: [] };
   749→  const dropped: string[] = [];
   750→  const out: Record<string, unknown> = {};
   751→  for (const [k, v] of Object.entries(base)) {
   752→    if (immutable.has(k)) {
   753→      dropped.push(k);
   754→      continue;
   755→    }
   756→    out[k] = v;
   757→  }
   758→  return { data: out, dropped };
   759→}
